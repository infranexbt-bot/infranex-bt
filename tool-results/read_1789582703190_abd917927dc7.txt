   504→// ---------------------------------------------------------------------------
   505→// TIER2 — Deployment config revisions (spec §20/§25 rollback). Every config-
   506→// mutating action snapshots the PREVIOUS config here before writing, so any
   507→// change (initial deploy, drift remediation, runtime optimization, a rollback
   508→// itself) can be undone: restore the JSON, push apply_config via the daemon.
   509→// The escalation ladder points at the newest revision that differs from the
   510→// current config as the "last known-good" rollback target.
   511→// ~20 rows per deployment (pruned oldest-first).
   512→// ---------------------------------------------------------------------------
   513→
   514→model DeploymentRevision {
   515→  id           String   @id @default(cuid())
   516→  deploymentId String
   517→  // Monotonic per-deployment revision number (r1 = initial deploy config).
   518→  rev          Int
   519→  // "deploy" | "drift" | "runtime-opt" | "rollback-backup"
   520→  cause        String
   521→  // Full DeploymentConfig JSON that was active BEFORE the next change.
   522→  configJson   String
   523→  // Requirements profile snapshot at that time (when one existed).
   524→  requirementsJson String?
   525→  // Human note: what changed, or which rollback this backup was taken for.
   526→  note         String   @default("")
   527→  // "engine" (automation) or the operator userId for manual changes.
   528→  actor        String   @default("engine")
   529→  createdAt    DateTime @default(now())
   530→
   531→  @@unique([deploymentId, rev])
   532→  @@index([deploymentId, createdAt])
