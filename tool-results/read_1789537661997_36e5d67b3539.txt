     1→// ---------------------------------------------------------------------------
     2→// Miner's Ledger scoring v2 — models how an experienced GPU miner actually
     3→// picks a subnet. Research-backed (Bittensor docs, taostats miner guide,
     4→// miner-economics deep dives, dTAO analyst frameworks):
     5→//
     6→//   Q1 FIT        — can my hardware run this? (subnet type → GPU tier/cost)
     7→//   Q2 TOP LINE   — what does it pay? (emission × 41% miner share ÷ EARNING
     8→//                   miners, priced in alpha → TAO → USD)
     9→//   Q3 BOTTOM LINE— what do I keep? (minus GPU rental + infra → net monthly)
    10→//   Q4 SEAT SAFETY— can I keep the slot? (saturation, reward concentration,
    11→//                   burn cost, immunity runway)
    12→//   Q5 HOLD VALUE — will revenue persist? (alpha price trend, pool liquidity,
    13→//                   sell-pressure slippage, owner identity, maturity)
    14→//
    15→// Key constants from the chain: per-tempo split is 18% owner / 41% miners /
    16→// 41% validators+stakers. "Top 10-20% of miners take 60-80% of emissions;
    17→// median miners hover near breakeven" — so the ledger scores NET numbers,
    18→// not gross, and treats reward concentration as a first-class risk.
    19→// ---------------------------------------------------------------------------
    20→

    24→
    25→const clampScore = (v: number, lo: number, hi: number) =>
    26→  Math.min(hi, Math.max(lo, v));

    28→// ---------------------------------------------------------------------------
    29→// Score primitives — 5 pillars, miner-first weights

    31→
    32→export interface ScoreComponents {
    33→  /** Net ROI after GPU rental + infra costs (the bottom line). */
    34→  net_roi: number;
    35→  /** Seat safety — free slots, reward spread, burn cost, immunity runway. */
    36→  seat_safety: number;
    37→  /** Alpha economics — 24h price trend, pool liquidity, sell slippage. */
    38→  alpha_economics: number;
    39→  /** Earning reality — rewarded ratio, concentration, newcomer ramp penalty. */
    40→  earning_reality: number;
    41→  /** Fit & feasibility — emission on, validator coverage, identity, maturity. */
    42→  fit_feasibility: number;
    43→}
    44→
    45→export const SCORE_WEIGHTS: Record<keyof ScoreComponents, number> = {
    46→  net_roi: 0.3,
    47→  seat_safety: 0.2,
    48→  alpha_economics: 0.2,
    49→  earning_reality: 0.15,
    50→  fit_feasibility: 0.15,
    51→};
    52→
    53→const FACTOR_LABELS: Record<keyof ScoreComponents, string> = {
    54→  net_roi: "Net ROI (after GPU cost)",
    55→  seat_safety: "Seat Safety",
    56→  alpha_economics: "Alpha Economics",
    57→  earning_reality: "Earning Reality",
    58→  fit_feasibility: "Fit & Feasibility",
    59→};
    60→
    61→const FACTOR_DESC: Record<keyof ScoreComponents, string> = {
    62→  net_roi:
    63→    "Newcomer-adjusted mid-pack monthly USD (per-registered slot × reward spread) minus GPU rental + infra at market rates — the miner's bottom line, not gross emission.",
    64→  seat_safety:
    65→    "Free UID slots, reward spread (top-10% concentration), registration burn level and immunity runway — can you keep the seat?",
    66→  alpha_economics:
    67→    "dTAO hold value: alpha price 24h trend, TAO-side pool depth (exit liquidity) and slippage from converting daily earnings.",
    68→  earning_reality:
    69→    "Share of registered miners actually rewarded, top-10% emission concentration and the bond-EMA ramp penalty for newcomers.",
    70→  fit_feasibility:
    71→    "Emission enabled, validator coverage, on-chain identity/GitHub and subnet maturity — operational basics before committing hardware.",
    72→};
    73→
    74→export function deriveFactors(c: ScoreComponents): OpportunityFactor[] {
    75→  return (Object.keys(c) as (keyof ScoreComponents)[]).map((k) => {
    76→    const raw = c[k];
    77→    const weight = SCORE_WEIGHTS[k];
    78→    return {
    79→      name: FACTOR_LABELS[k],
    80→      value: Math.round(raw * weight * 100) / 100,
    81→      weight,
    82→      raw,
    83→      impact: raw >= 60 ? "positive" : raw <= 40 ? "negative" : "neutral",
    84→      description: FACTOR_DESC[k],
    85→    };
    86→  });
    87→}
    88→
    89→export function totalScore(c: ScoreComponents): number {
    90→  return Math.round(
    91→    (Object.keys(c) as (keyof ScoreComponents)[]).reduce(
    92→      (acc, k) => acc + c[k] * SCORE_WEIGHTS[k],
    93→      0
    94→    ) * 10
    95→  ) / 10;
    96→}
    97→
    98→export function riskLevel(score: number): "low" | "medium" | "high" {
    99→  if (score >= 60) return "low";
   100→  if (score >= 40) return "medium";
   101→  return "high";
   102→}

   152→/** Deterministic GPU tier from the reward stream — fallback for subnets
   153→ *  with no registered identity. What per-miner revenue can plausibly fund. */
   154→export function estimateGpuTierFromRevenue(monthlyUsdPerMiner: number): GpuTier {
   155→  if (monthlyUsdPerMiner >= 1200) return GPU_TIERS.h200;
   156→  if (monthlyUsdPerMiner >= 600) return GPU_TIERS.h100;
   157→  if (monthlyUsdPerMiner >= 300) return GPU_TIERS.a100;
   158→  if (monthlyUsdPerMiner >= 140) return GPU_TIERS.a6000;
   159→  if (monthlyUsdPerMiner >= 60) return GPU_TIERS.consumer24;
   160→  if (monthlyUsdPerMiner >= 25) return GPU_TIERS.vram16;
   161→  if (monthlyUsdPerMiner >= 8) return GPU_TIERS.entry;
   162→  return GPU_TIERS.cpu;
   163→}

   165→interface CategoryRule {
   166→  keywords: RegExp;
   167→  category: string;
   168→  tier: GpuTier;
   169→  infra?: number;
   170→}
   171→
   172→// Ordered — first match wins. Derived from how subnet teams describe their
   173→// incentive mechanism (on-chain SubnetIdentitiesV3 descriptions).
   174→const CATEGORY_RULES: CategoryRule[] = [
   175→  { keywords: /\b(frontier|llm|text.?gen|prompt|language model|chat|assistant|inference)\b/i, category: "Frontier inference", tier: GPU_TIERS.h100 },
   176→  { keywords: /\b(pretrain|pre-train|model training|training network|grpo|rlhf|post.?training)\b/i, category: "Model training", tier: GPU_TIERS.h200 },
   177→  { keywords: /\b(video|image|diffusion|vision|photo|generative media|3d|render)\b/i, category: "Media generation", tier: GPU_TIERS.consumer24 },
   178→  { keywords: /\b(audio|speech|voice|music|sound)\b/i, category: "Audio & speech", tier: GPU_TIERS.consumer24 },
   179→  { keywords: /\b(scrap|crawl|data collect|data index|social data|sentiment|feed|dataset)\b/i, category: "Data & scraping", tier: GPU_TIERS.cpu, infra: INFRA_SCRAPER_USD },
   180→  { keywords: /\b(storag|file|archiv|backup)\b/i, category: "Storage", tier: GPU_TIERS.cpu },
   181→  { keywords: /\b(prediction|market mak|trading|financ|quant|hedge|odds)\b/i, category: "Prediction & markets", tier: GPU_TIERS.cpu },
   182→  { keywords: /\b(agent|orchestrat|routing|logic|swarm|tool use|evm|smart contract|web3 infra)\b/i, category: "Agents & logic", tier: GPU_TIERS.cpu },
   183→  { keywords: /\b(protein|fold|bio|genom|drug|science|research|simulat)\b/i, category: "Science & simulation", tier: GPU_TIERS.a6000 },
   184→  { keywords: /\b(compute|gpu|depin|edge|latency|bandwidth|network shar)\b/i, category: "Compute sharing", tier: GPU_TIERS.consumer24 },
   185→  { keywords: /\b(moderat|detect|scan|verif|audit|secur|privacy|zero.?know)\b/i, category: "Verification & security", tier: GPU_TIERS.consumer24 },
   186→  { keywords: /\b(maps?|geo|weather|energy|robot|drone|iot)\b/i, category: "Real-world data", tier: GPU_TIERS.consumer24 },
   187→];
   188→
   189→export function classifySubnetHardware(
   190→  name: string | null | undefined,
   191→  description: string | null | undefined,
   192→  fallbacks?: {
   193→    /** Curated category text (e.g. "Inference") when the chain identity is absent. */
   194→    fallbackCategory?: string;
   195→    /** Curated/user GPU figures win when the classifier has no keyword match. */
   196→    fallbackVramGb?: number;
   197→    fallbackGpu?: string;
   198→    /** Per-earning-miner monthly USD — revenue-based fallback tier. */
   199→    fallbackMonthlyUsd?: number;
   200→  }
   201→): SubnetHardwareProfile {
   202→  const text = `${name ?? ""} ${description ?? ""} ${fallbacks?.fallbackCategory ?? ""}`;
   203→  for (const rule of CATEGORY_RULES) {
   204→    if (rule.keywords.test(text)) {
   205→      return {
   206→        category: rule.category,
   207→        tier: rule.tier,
   208→        minVramGb: rule.tier.minVramGb,
   209→        recommendedGpu: rule.tier.recommendedGpu,
   210→        monthlyCostUsd: rule.tier.monthlyRentUsd + (rule.infra ?? INFRA_BASE_USD),
   211→        isGpuWorkload: rule.tier.minVramGb > 0,
   212→        classified: true,
   213→      };
   214→    }
   215→  }
   216→  // No keyword match — fall back to curated VRAM data, else revenue tier.
   217→  if (fallbacks?.fallbackVramGb && fallbacks.fallbackVramGb > 0) {
   218→    const tier = Object.values(GPU_TIERS).find(
   219→      (t) => t.minVramGb === fallbacks.fallbackVramGb
   220→    ) ?? GPU_TIERS.consumer24;
   221→    return {
   222→      category: fallbacks.fallbackCategory || "Unclassified workload",
   223→      tier,
   224→      minVramGb: tier.minVramGb,
   225→      recommendedGpu: fallbacks.fallbackGpu ?? tier.recommendedGpu,
   226→      monthlyCostUsd: tier.monthlyRentUsd + INFRA_BASE_USD,
   227→      isGpuWorkload: true,
   228→      classified: false,
   229→    };
   230→  }
   231→  const tier = estimateGpuTierFromRevenue(fallbacks?.fallbackMonthlyUsd ?? 0);
   232→  return {
   233→    category: fallbacks?.fallbackCategory || "Unclassified workload",
   234→    tier,
   235→    minVramGb: tier.minVramGb,
   236→    recommendedGpu: tier.recommendedGpu,
   237→    monthlyCostUsd: tier.monthlyRentUsd + INFRA_BASE_USD,
   238→    isGpuWorkload: tier.minVramGb > 0,
   239→    classified: false,
   240→  };
   241→}

   296→/**
   297→ * Score one subnet the way a miner reads it. Deterministic per snapshot —
   298→ * the same inputs always produce the same score so ranks stay stable.
   299→ */
   300→export function scoreMinersLedger(inputs: {
   301→  live: LiveSubnetMetrics;
   302→  taoUsd: number;
   303→  hardware?: SubnetHardwareProfile | null;
   304→  taoChange24h?: number;
   305→  /** Blocks since the subnet was registered (for the maturity bonus). */
   306→  liveAgeBlocks?: number | null;
   307→  /** User-configurable cost lines (Profitability Engine settings). When
   308→   *  absent, market defaults apply — same engine, same shape. */
   309→  costs?: {
   310→    hardwareMode?: "rent" | "owned";
   311→    electricityUsdPerKwh?: number;
   312→    storageMonthlyUsd?: number;
   313→    /** 0/undefined = auto infra from the work-type classifier. */
   314→    infraMonthlyUsd?: number;
   315→    otherOpexMonthlyUsd?: number;
   316→    includeBurnAmortization?: boolean;
   317→    amortizeBurnMonths?: number;
   318→  };
   319→}): MinerLedger {
   320→  const { live, taoUsd } = inputs;
   321→  const usd = taoUsd || 0;
   322→
   323→  // --- Top line: newcomer-adjusted per-miner revenue ----------------------
   324→  // Miners get 41% of the subnet's emission (docs constant); the chain-
   325→  // measured minerEmissionTaoPerDay already reflects the realized split.
   326→  //
   327→  // The decision a miner makes is "if I register TODAY, what does a
   328→  // mid-pack seat pay?" — so the ledger starts from the per-REGISTERED
   329→  // slot (zero-earning seats included: on many subnets 250+ UIDs are
   330→  // registered and only a handful earn), then applies the median/mean
   331→  // spread share so a whale UID can't inflate the estimate.
   332→  // perEarningDailyTao (mean over rewarded UIDs) is kept as the upside
   333→  // reference — what current earners make.
   334→  const rewarded = Math.max(live.rewardedMiners ?? 0, 0);
   335→  const registered = Math.max(live.minersCount, 1);
   336→  const minerEmissionTaoPerDay = live.minerEmissionTaoPerDay ?? 0;
   337→  const perRegisteredDailyTao =
   338→    minerEmissionTaoPerDay > 0 ? minerEmissionTaoPerDay / registered : 0;
   339→  const perEarningDailyTao =
   340→    rewarded > 0 && minerEmissionTaoPerDay > 0
   341→      ? minerEmissionTaoPerDay / rewarded
   342→      : minerEmissionTaoPerDay > 0
   343→        ? perRegisteredDailyTao // vec unavailable — no zero-slot discount available
   344→        : 0;
   345→  const rewardMedianShare = live.incentiveMedianShare ?? 1;
   346→  // Newcomer-adjusted mid-pack revenue:
   347→  //  - reward vec available  → perRegistered × medianShare
   348→  //  - reward vec unavailable→ perRegistered × 0.6 (conservative)
   349→  const expectedDailyTao =
   350→    rewarded > 0 && minerEmissionTaoPerDay > 0
   351→      ? perRegisteredDailyTao * rewardMedianShare
   352→      : minerEmissionTaoPerDay > 0
   353→        ? perRegisteredDailyTao * 0.6
   354→        : 0;
   355→  const grossMonthlyUsd = Math.round(expectedDailyTao * 30 * usd);
   356→  const dailyAlpha = live.movingPrice > 0 ? expectedDailyTao / live.movingPrice : 0;
   357→
   358→  // --- Hardware profile (work type → GPU cost) ---------------------------
   359→  const fallbackMonthly = grossMonthlyUsd > 0 ? grossMonthlyUsd : undefined;
   360→  const hardware =
   361→    inputs.hardware ??
   362→    classifySubnetHardware(live.name, live.identityDescription, {
   363→      fallbackMonthlyUsd: fallbackMonthly,
   364→    });
   365→  // --- Cost stack (Profitability Engine lines) ----------------------------
   366→  const costs = inputs.costs ?? {};
   367→  const gpuCost =
   368→    costs.hardwareMode === "owned"
   369→      ? electricityMonthlyUsd(
   370→          hardware.tier.powerWatts,
   371→          costs.electricityUsdPerKwh ?? 0.12
   372→        )
   373→      : hardware.tier.monthlyRentUsd;
   374→  const infraCost =
   375→    costs.infraMonthlyUsd != null && costs.infraMonthlyUsd > 0
   376→      ? costs.infraMonthlyUsd
   377→      : Math.max(hardware.monthlyCostUsd - hardware.tier.monthlyRentUsd, INFRA_BASE_USD);
   378→  const storageCost = costs.storageMonthlyUsd ?? 0;
   379→  const otherOpex = costs.otherOpexMonthlyUsd ?? 0;
   380→  const burnUsd =
   381→    live.burnCostTao != null && costs.includeBurnAmortization
   382→      ? Math.max(live.burnCostTao, 0) * usd
   383→      : 0;
   384→  const amortizedBurn = burnUsd / Math.max(costs.amortizeBurnMonths ?? 3, 1);
   385→  const totalCosts = gpuCost + infraCost + storageCost + otherOpex + amortizedBurn;
   386→  // Zero-revenue subnets are honest losers: the cost stack still applies.
   387→  const netMonthlyUsd = grossMonthlyUsd > 0
   388→    ? Math.round(grossMonthlyUsd - totalCosts)
   389→    : -Math.round(totalCosts);
   390→
   391→  // --- Alpha economics ----------------------------------------------------
   392→  const alphaPriceTao = live.movingPrice;
   393→  const alphaPriceUsd = alphaPriceTao * usd;
   394→  const liquidityTao = live.subnetTao; // TAO side of the pool = exit liquidity
   395→  const slippagePct =
   396→    live.alphaOut > 0 && dailyAlpha > 0
   397→      ? Math.round((dailyAlpha / live.alphaOut) * 10000) / 100
   398→      : null;
   399→
   400→  // --- Seat safety inputs -------------------------------------------------
   401→  const maxUids = live.maxUids ?? 0;
   402→  const freeSlots = maxUids > 0 ? Math.max(0, maxUids - live.minersCount) : null;
   403→  const totalSlots = maxUids > 0 ? maxUids : null;
   404→  const rewardedRatio = live.rewardedMiners != null && live.rewardedMiners > 0
   405→    ? Math.min(1, live.rewardedMiners / registered)
   406→    : live.rewardedMiners === 0
   407→      ? 0
   408→      : null;
   409→  const top10 = live.top10IncentiveShare;
   410→
   411→  // Newcomer ramp: bonds build via EMA (α=0.1), and on concentrated subnets
   412→  // a fresh UID starts at zero. Research: 1-3 months of tuning is typical.
   413→  const rampWeeks =
   414→    3 +
   415→    (rewardedRatio != null ? (1 - rewardedRatio) * 6 : 3) +
   416→    (top10 != null && top10 > 0.7 ? 3 : 0) +
   417→    (freeSlots != null && freeSlots === 0 ? 2 : 0);
   418→
   419→  // --- Pillar 1: Net ROI --------------------------------------------------
   420→  // $0 → 10, +$100 → 54, +$500 → 69, +$1k → 76, +$5k → 92; losses slide to 2.
   421→  const netRoi =
   422→    netMonthlyUsd >= 0
   423→      ? clampScore(10 + 22 * Math.log10(netMonthlyUsd + 1), 2, 97)
   424→      : clampScore(10 + netMonthlyUsd / 100, 2, 10);
   425→
   426→  // --- Pillar 2: Seat safety ----------------------------------------------
   427→  const roomRatio = maxUids > 0 ? freeSlots! / maxUids : 0.5;
   428→  const scoreRoom = clampScore(10 + roomRatio * 45, 10, 55); // full subnet → 10
   429→  const scoreSpread =
   430→    top10 != null ? clampScore(100 - (top10 - 0.2) * 100, 10, 95) : 50;
   431→  const scoreBurn =
   432→    live.burnCostTao != null
   433→      ? clampScore(95 - Math.log10(live.burnCostTao + 1) * 25, 30, 95)
   434→      : 55;
   435→  const scoreImmunity =
   436→    live.immunityBlocks != null
   437→      ? clampScore(30 + (live.immunityBlocks / 7200) * 50, 30, 80)
   438→      : 45;
   439→  const seatSafety = !live.emissionEnabled
   440→    ? 12
   441→    : clampScore(
   442→        scoreRoom * 0.35 + scoreSpread * 0.35 + scoreBurn * 0.15 + scoreImmunity * 0.15,
   443→        5,
   444→        96
   445→      );
   446→
   447→  // --- Pillar 3: Alpha economics -------------------------------------------
   448→  const change = live.alphaPriceChange24h;
   449→  const scoreTrend = change != null ? clampScore(50 + change * 1.5, 10, 95) : 50;
   450→  const scoreLiq =
   451→    liquidityTao > 0
   452→      ? clampScore(20 + 19 * Math.log10(liquidityTao / 500 + 1), 10, 95)
   453→      : 10;
   454→  const scoreSlip =
   455→    slippagePct != null ? clampScore(95 - slippagePct * 1200, 15, 95) : 50;
   456→  const alphaEconomics = usd <= 0
   457→    ? 50
   458→    : clampScore(scoreTrend * 0.4 + scoreLiq * 0.3 + scoreSlip * 0.3, 5, 96);
   459→
   460→  // --- Pillar 4: Earning reality -------------------------------------------
   461→  const scoreRewarded =
   462→    rewardedRatio != null ? clampScore(15 + rewardedRatio * 70, 15, 85) : 50;
   463→  const scoreConc =
   464→    top10 != null ? clampScore(100 - (top10 - 0.2) * 90, 15, 92) : 50;
   465→  const scoreRamp = clampScore(90 - rampWeeks * 5, 20, 90);
   466→  const earningReality = !live.emissionEnabled
   467→    ? 8
   468→    : clampScore(scoreRewarded * 0.4 + scoreConc * 0.3 + scoreRamp * 0.3, 5, 96);
   469→
   470→  // --- Pillar 5: Fit & feasibility ------------------------------------------
   471→  let fitFeasibility = 25; // emission disabled base
   472→  if (live.emissionEnabled) {
   473→    fitFeasibility = 40;
   474→    const validators = live.validatorsCount ?? 0;
   475→    fitFeasibility += clampScore(validators * 2.5, 0, 20); // validator coverage
   476→    if (live.identityGithub) fitFeasibility += 12;
   477→    if (live.identityDescription) fitFeasibility += 6;
   478→    if (live.registeredAt != null && inputs.liveAgeBlocks != null) {
   479→      const ageDays = inputs.liveAgeBlocks / 7200;
   480→      fitFeasibility += clampScore(Math.min(ageDays / 30, 1) * 12, 0, 12); // maturity
   481→    }
   482→    if (live.immunityBlocks != null && live.immunityBlocks < 4096) fitFeasibility -= 5;
   483→  }
   484→  fitFeasibility = clampScore(fitFeasibility, 5, 92);
   485→
   486→  const components: ScoreComponents = {
   487→    net_roi: Math.round(netRoi * 10) / 10,
   488→    seat_safety: Math.round(seatSafety * 10) / 10,
   489→    alpha_economics: Math.round(alphaEconomics * 10) / 10,
   490→    earning_reality: Math.round(earningReality * 10) / 10,
   491→    fit_feasibility: Math.round(fitFeasibility * 10) / 10,
   492→  };
   493→
   494→  const diag: MinerLedgerDiagnostics = {
   495→    perEarningDailyTao: Math.round(perEarningDailyTao * 10000) / 10000,
   496→    expectedDailyTao: Math.round(expectedDailyTao * 10000) / 10000,
   497→    rewardMedianShare: live.incentiveMedianShare,
   498→    perRegisteredDailyTao: Math.round(perRegisteredDailyTao * 10000) / 10000,
   499→    grossMonthlyUsd,
   500→    alphaPriceTao,
   501→    alphaPriceUsd: Math.round(alphaPriceUsd * 1000) / 1000,
   502→    alphaChange24h: change ?? null,
   503→    liquidityTao: Math.round(liquidityTao),
   504→    slippagePct,
   505→    category: hardware.category,
   506→    gpuLabel: hardware.tier.label,
   507→    recommendedGpu: hardware.recommendedGpu,
   508→    minVramGb: hardware.minVramGb,
   509→    gpuCostMonthlyUsd: Math.round(gpuCost),
   510→    infraCostMonthlyUsd: Math.round(infraCost),
   511→    storageCostMonthlyUsd: Math.round(storageCost),
   512→    otherOpexMonthlyUsd: Math.round(otherOpex),
   513→    amortizedBurnUsd: Math.round(amortizedBurn),
   514→    totalCostMonthlyUsd: Math.round(totalCosts),
   515→    gpuPowerWatts: hardware.tier.powerWatts,
   516→    netMonthlyUsd,
   517→    netDailyTao:
   518→      usd > 0 ? Math.round((netMonthlyUsd / 30 / usd) * 10000) / 10000 : 0,
   519→    freeSlots,
   520→    totalSlots,
   521→    rewardedRatio: rewardedRatio != null ? Math.round(rewardedRatio * 1000) / 1000 : null,
   522→    top10IncentiveShare: top10,
   523→    burnCostTao: live.burnCostTao,
   524→    immunityBlocks: live.immunityBlocks,
   525→    rampWeeks: Math.round(rampWeeks * 10) / 10,
   526→    emissionEnabled: live.emissionEnabled,
   527→    hardwareClassified: hardware.classified,
   528→  };
   529→
   530→  return { components, factors: deriveFactors(components), diag };
   531→}

   533→// ---------------------------------------------------------------------------
   534→// Chance to earn · month 1 — a newcomer's honest odds of earning ANY reward
   535→// during their first month of continuous, good-uptime mining. Uptime alone
   536→// keeps the seat; the share of slots that actually earn decides whether the
   537→// wallet moves. Dominant factor: rewardedRatio (share of registered slots
   538→// that earned last epoch), discounted for whale dominance, full subnets
   539→// (must displace an incumbent) and long bond-EMA ramps.
   540→// ---------------------------------------------------------------------------
   541→
   542→export type EarnChanceLevel = "high" | "medium" | "low" | "none";
   543→
   544→export interface EarnChance {
   545→  level: EarnChanceLevel;
   546→  /** 0–100 modeled chance that a newcomer earns something in month 1. */
   547→  pct: number;
   548→  /** Human explanation of the dominant factors. */
   549→  note: string;
   550→}
   551→
   552→export function computeEarnChance(inputs: {
   553→  /** Share of registered slots that earned reward last epoch (0..1). */
   554→  rewardedRatio: number | null;
   555→  /** Free UID slots (null = unknown). 0 means you must displace an incumbent. */
   556→  freeSlots: number | null;
   557→  /** Top-10% share of epoch incentive (0..1) — whale dominance. */
   558→  top10IncentiveShare: number | null;
   559→  /** Newcomer bond-EMA ramp estimate in weeks. */
   560→  rampWeeks: number | null;
   561→}): EarnChance {
   562→  const { rewardedRatio, freeSlots, top10IncentiveShare, rampWeeks } = inputs;
   563→
   564→  // Base: the fraction of slots that actually earn — the single strongest
   565→  // predictor of whether a newcomer earns anything at all.
   566→  let pct = rewardedRatio != null ? rewardedRatio * 100 : 0;
   567→  const factors: string[] = [];
   568→
   569→  // Whale dominance: when the top 10% of UIDs take nearly everything, the
   570→  // rewarded tail is scraps — being "rewarded" still pays almost nothing.
   571→  if (top10IncentiveShare != null && top10IncentiveShare > 0.5) {
   572→    pct *= 1 - Math.min(1, top10IncentiveShare - 0.5) * 0.6;
   573→    factors.push(
   574→      `top-10% of UIDs take ${Math.round(top10IncentiveShare * 100)}% of rewards`
   575→    );
   576→  }
   577→
   578→  // Full subnet: no free seats — you must outperform an incumbent to get in,
   579→  // and you start from zero bonds against their established ones.
   580→  if (freeSlots === 0) {
   581→    pct *= 0.7;
   582→    factors.push("subnet full — must displace an incumbent");
   583→  }
   584→
   585→  // Long ramp: month 1 is mostly bond-building; long ramps leave little of it.
   586→  if (rampWeeks != null && rampWeeks > 8) {
   587→    pct *= 0.8;
   588→    factors.push(`~${Math.round(rampWeeks)} wk ramp to full bonds`);
   589→  }
   590→
   591→  pct = Math.round(Math.max(0, Math.min(100, pct)));
   592→  const level: EarnChanceLevel =
   593→    rewardedRatio != null && rewardedRatio <= 0
   594→      ? "none"
   595→      : pct >= 50
   596→        ? "high"
   597→        : pct >= 20
   598→          ? "medium"
   599→          : pct > 0
   600→            ? "low"
   601→            : "none";
   602→
   603→  const base = rewardedRatio != null
   604→    ? `${rewardedRatio < 0.1 && rewardedRatio > 0 ? (rewardedRatio * 100).toFixed(1) : Math.round(rewardedRatio * 100)}% of slots earned last epoch`
   605→    : "reward spread unknown";
   606→  const note =
   607→    factors.length > 0 ? `${base} · ${factors.join(" · ")}` : base;
   608→
   609→  return { level, pct, note };
   610→}

   612→// ---------------------------------------------------------------------------
   613→// Seat Chance — "can I actually GET IN?" (vs EarnChance's "will I earn?").
   614→//
   615→// Bittensor registration mechanics:
   616→//   • Subnets have a UID cap (maxUids). Free slots → register via burn or PoW.
   617→//   • FULL subnet → registration STILL works: a burn registration pays the
   618→//     dynamic recycle cost and the chain replaces the worst-performing
   619→//     NON-IMMUNE uid. PoW registrations compete for per-block slots the same
   620→//     way. You are only locked out if every uid is immune (rare — immunity
   621→//     covers ~14h by default) or registration is disabled on the subnet.
   622→//   • immunityBlocks = how long YOUR new seat cannot be deregistered — the
   623→//     runway to start earning before becoming replaceable yourself.
   624→// ---------------------------------------------------------------------------
   625→
   626→/**
   627→ * Burn-quote display precision — the chain's burn hyperparam can sit at the
   628→ * floor (e.g. Chutes ~0.0005 TAO); 3 decimals would show "0.001" — a 2x
   629→ * misquote shown to a user deciding whether to burn. <0.01 → 4 decimals,
   630→ * <1 → 3, else 2.
   631→ */
   632→export function formatBurnTao(n: number): string {
   633→  if (n < 0.01) return n.toFixed(4);
   634→  if (n < 1) return n.toFixed(3);
   635→  return n.toFixed(2);
   636→}

   638→export type SeatVerdict = "open" | "burn-entry" | "waitlist" | "unknown";
   639→
   640→export interface SeatChance {
   641→  verdict: SeatVerdict;
   642→  /** Registration is possible right now (burn path or free slot). */
   643→  canRegister: boolean;
   644→  slotsFree: number | null;
   645→  totalSlots: number | null;
   646→  /** 0–100 — how full the uid space is. */
   647→  fillPct: number | null;
   648→  burnCostTao: number | null;
   649→  /** Protection window for a NEW seat, in hours (blocks × 12s). */
   650→  immunityHours: number | null;
   651→  /** Share of uids that earned NOTHING last epoch — the replaceable bottom
   652→   *  a newcomer displaces from. High = churn churns in your favor. */
   653→  replaceableShare: number | null;
   654→  headline: string;
   655→  detail: string;
   656→}
   657→
   658→export function assessSeatChance(inputs: {
   659→  minersCount: number | null;
   660→  maxUids: number | null;
   661→  /** Already-computed free slots (skip when you only have counts). */
   662→  freeSlots?: number | null;
   663→  burnCostTao: number | null;
   664→  immunityBlocks: number | null;
   665→  /** Rewarded uids (either give the ratio directly or the absolute count). */
   666→  rewardedRatio?: number | null;
   667→  rewardedMiners?: number | null;
   668→}): SeatChance {
   669→  const { maxUids, burnCostTao, immunityBlocks } = inputs;
   670→  const minersCount = inputs.minersCount ?? 0;
   671→  const slotsFree =
   672→    inputs.freeSlots != null
   673→      ? Math.max(0, inputs.freeSlots)
   674→      : maxUids != null && maxUids > 0
   675→        ? Math.max(0, maxUids - minersCount)
   676→        : null;
   677→  const fillPct =
   678→    maxUids != null && maxUids > 0
   679→      ? Math.min(100, Math.round((minersCount / maxUids) * 100))
   680→      : null;
   681→  let replaceableShare: number | null = null;
   682→  if (inputs.rewardedRatio != null) {
   683→    replaceableShare = Math.max(0, Math.min(1, 1 - inputs.rewardedRatio));
   684→  } else if (inputs.rewardedMiners != null && minersCount > 0) {
   685→    replaceableShare = Math.max(0, Math.min(1, 1 - inputs.rewardedMiners / minersCount));
   686→  }
   687→  const immunityHours =
   688→    immunityBlocks != null && immunityBlocks > 0
   689→      ? Math.round(((immunityBlocks * 12) / 3600) * 10) / 10
   690→      : null;
   691→  const immunityNote = immunityHours != null
   692→    ? `a new seat is immune for ~${immunityHours}h — your runway to start earning before you're replaceable`
   693→    : null;
   694→  const deadNote = replaceableShare != null && replaceableShare > 0.05
   695→    ? `${Math.round(replaceableShare * 100)}% of uids earned nothing last epoch — a deep replaceable bottom`
   696→    : replaceableShare != null
   697→      ? "almost every seat earned last epoch — displacement targets are scarce and strong"
   698→      : null;
   699→
   700→  const unknown: SeatChance = {
   701→    verdict: "unknown",
   702→    canRegister: true,
   703→    slotsFree,
   704→    totalSlots: maxUids ?? null,
   705→    fillPct,
   706→    burnCostTao,
   707→    immunityHours,
   708→    replaceableShare,
   709→    headline: "slot data unavailable",
   710→    detail:
   711→      "UID capacity was not returned by the chain snapshot this pass — refresh the network data. " +
   712→      (burnCostTao != null ? `Burn registration quote: ~${formatBurnTao(burnCostTao)} TAO.` : ""),
   713→  };
   714→
   715→  if (slotsFree == null || maxUids == null || maxUids <= 0) return unknown;
   716→
   717→  // --- Open: free slots exist ---------------------------------------------
   718→  if (slotsFree > 0) {
   719→    return {
   720→      verdict: "open",
   721→      canRegister: true,
   722→      slotsFree,
   723→      totalSlots: maxUids,
   724→      fillPct,
   725→      burnCostTao,
   726→      immunityHours,
   727→      replaceableShare,
   728→      headline: `${slotsFree} of ${maxUids} slots free`,
   729→      detail:
   730→        `This subnet is NOT full — ${slotsFree} uid${slotsFree === 1 ? "" : "s"} open (${fillPct}% filled). ` +
   731→        `Register now via burn${burnCostTao != null ? ` (~${formatBurnTao(burnCostTao)} TAO)` : ""} or PoW. ` +
   732→        (immunityNote ? `${immunityNote}. ` : "") +
   733→        (deadNote ? `${deadNote}.` : ""),
   734→    };
   735→  }
   736→
   737→  // --- Full: burn-entry is the deterministic path ---------------------------
   738→  if (burnCostTao != null && burnCostTao > 0) {
   739→    return {
   740→      verdict: "burn-entry",
   741→      canRegister: true,
   742→      slotsFree: 0,
   743→      totalSlots: maxUids,
   744→      fillPct,
   745→      burnCostTao,
   746→      immunityHours,
   747→      replaceableShare,
   748→      headline: `full (${minersCount}/${maxUids}) — burn-entry ~${formatBurnTao(burnCostTao)} TAO`,
   749→      detail:
   750→        `Every uid is taken, but registration still works: paying the burn (~${formatBurnTao(burnCostTao)} TAO, floats with demand) ` +
   751→        `immediately replaces the WORST-performing non-immune uid. ` +
   752→        (deadNote ? `${deadNote}. ` : "The bottom of this cohort is well-defended — displacement may take several attempts. ") +
   753→        (immunityNote ? `${immunityNote}.` : ""),
   754→    };
   755→  }
   756→
   757→  // --- Full, no burn quote: PoW/competitive path ----------------------------
   758→  return {
   759→    verdict: "waitlist",
   760→    canRegister: true,
   761→    slotsFree: 0,
   762→    totalSlots: maxUids,
   763→    fillPct,
   764→    burnCostTao: null,
   765→    immunityHours,
   766→    replaceableShare,
   767→    headline: `full (${minersCount}/${maxUids}) — entry is competitive`,
   768→    detail:
   769→      `No free slots and no burn quote from this snapshot. Entry means winning a PoW registration ` +
   770→      `and replacing a zero-earning uid — expect per-block competition. ` +
   771→      (deadNote ? `${deadNote}. ` : "") +
   772→      (immunityNote ? `${immunityNote}.` : ""),
   773→  };
   774→}
