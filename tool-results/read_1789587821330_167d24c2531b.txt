     3→// picks a subnet. Research-backed (Bittensor docs, taostats miner guide,
     4→// miner-economics deep dives, dTAO analyst frameworks):
     5→//
     6→//   Q1 FIT        — can my hardware run this? (subnet type → GPU tier/cost)
     7→//   Q2 TOP LINE   — what does it pay? (emission × 41% miner share ÷ EARNING
     8→//                   miners, priced in alpha → TAO → USD)
     9→//   Q3 BOTTOM LINE— what do I keep? (minus GPU rental + infra → net monthly)
    10→//   Q4 SEAT SAFETY— can I keep the slot? (saturation, reward concentration,
    11→//                   burn cost, immunity runway)
    12→//   Q5 HOLD VALUE — will revenue persist? (alpha price trend, pool liquidity,
    13→//                   sell-pressure slippage, owner identity, maturity)
    14→//
    15→// Key constants from the chain: per-tempo split is 18% owner / 41% miners /
    16→// 41% validators+stakers. "Top 10-20% of miners take 60-80% of emissions;
    17→// median miners hover near breakeven" — so the ledger scores NET numbers,
    18→// not gross, and treats reward concentration as a first-class risk.

    35→export interface ScoreComponents {

    44→  /** Fit & feasibility — emission on, validator coverage, identity, maturity. */
    45→  fit_feasibility: number;

    48→export const SCORE_WEIGHTS: Record<keyof ScoreComponents, number> = {
    49→  net_roi: 0.3,
    50→  seat_safety: 0.2,
    51→  alpha_economics: 0.2,
    52→  earning_reality: 0.15,
    53→  fit_feasibility: 0.15,
    54→};

    56→const FACTOR_LABELS: Record<keyof ScoreComponents, string> = {
    57→  net_roi: "Net ROI (after GPU cost)",
    58→  seat_safety: "Seat Safety",

    60→  earning_reality: "Earning Reality",
    61→  fit_feasibility: "Fit & Feasibility",

    77→export function deriveFactors(c: ScoreComponents): OpportunityFactor[] {
    78→  return (Object.keys(c) as (keyof ScoreComponents)[]).map((k) => {
    79→    const raw = c[k];
    80→    const weight = SCORE_WEIGHTS[k];
    81→    return {
    82→      name: FACTOR_LABELS[k],
    83→      value: Math.round(raw * weight * 100) / 100,
    84→      weight,
    85→      raw,
    86→      impact: raw >= 60 ? "positive" : raw <= 40 ? "negative" : "neutral",
    87→      description: FACTOR_DESC[k],
    88→    };
    89→  });
    90→}
    91→
    92→export function totalScore(c: ScoreComponents): number {
    93→  return Math.round(
    94→    (Object.keys(c) as (keyof ScoreComponents)[]).reduce(
    95→      (acc, k) => acc + c[k] * SCORE_WEIGHTS[k],
    96→      0
    97→    ) * 10
    98→  ) / 10;
    99→}

   101→export function riskLevel(score: number): "low" | "medium" | "high" {
   102→  if (score >= 60) return "low";
   103→  if (score >= 40) return "medium";
   104→  return "high";
   105→}

   150→/**
   151→ * Resolve a GPU tier from a model string found in a repo README
   152→ * (github-scraper parseGpuRequirement output). Ground truth from the
   153→ * subnet's own docs beats keyword classification.
   154→ */
   155→export function resolveGpuTierFromModel(model: string): GpuTier | null {
   156→  const m = model.toLowerCase();
   157→  if (/\bb300\b/.test(m)) return GPU_TIERS.b200; // same Blackwell HGX class
   158→  if (/\bb200\b/.test(m)) return GPU_TIERS.b200;
   159→  if (/\bh200\b/.test(m)) return GPU_TIERS.h200;
   160→  if (/\bh100\b/.test(m)) return GPU_TIERS.h100;
   161→  if (/\b(?:rtx\s*)?pro\s*6000\b/.test(m) || /\brtx\s*6000\s*ada\b/.test(m)) return GPU_TIERS.pro6000;
   162→  if (/\ba100\b/.test(m)) return GPU_TIERS.a100;
   163→  if (/\bl40s?\b/.test(m)) return GPU_TIERS.a6000; // 48GB Ada-class
   164→  if (/\ba40\b/.test(m)) return GPU_TIERS.a6000;
   165→  if (/\ba6000\b/.test(m)) return GPU_TIERS.a6000;
   166→  if (/\ba5000\b/.test(m)) return GPU_TIERS.vram16; // 24GB but older — mid tier
   167→  if (/\ba10\b/.test(m)) return GPU_TIERS.vram16;
   168→  if (/\bt4\b/.test(m)) return GPU_TIERS.entry;
   169→  if (/\b(?:rtx\s*)?(?:4090|3090|5090)\b/.test(m)) return GPU_TIERS.consumer24;
   170→  if (/\bmi300x\b/.test(m)) return GPU_TIERS.h200; // 192GB HBM3 accelerator class
   171→  return null;
   172→}

   210→/** Deterministic GPU tier from the reward stream — fallback for subnets
   211→ *  with no registered identity. What per-miner revenue can plausibly fund. */
   212→export function estimateGpuTierFromRevenue(monthlyUsdPerMiner: number): GpuTier {
   213→  if (monthlyUsdPerMiner >= 1200) return GPU_TIERS.h200;
   214→  if (monthlyUsdPerMiner >= 600) return GPU_TIERS.h100;
   215→  if (monthlyUsdPerMiner >= 300) return GPU_TIERS.a100;
   216→  if (monthlyUsdPerMiner >= 140) return GPU_TIERS.a6000;
   217→  if (monthlyUsdPerMiner >= 60) return GPU_TIERS.consumer24;
   218→  if (monthlyUsdPerMiner >= 25) return GPU_TIERS.vram16;
   219→  if (monthlyUsdPerMiner >= 8) return GPU_TIERS.entry;
   220→  return GPU_TIERS.cpu;
   221→}

   247→export function classifySubnetHardware(
   248→  name: string | null | undefined,
   249→  description: string | null | undefined,
   250→  fallbacks?: {
   251→    /** Curated category text (e.g. "Inference") when the chain identity is absent. */
   252→    fallbackCategory?: string;
   253→    /** Curated/user GPU figures win when the classifier has no keyword match. */
   254→    fallbackVramGb?: number;
   255→    fallbackGpu?: string;
   256→    /** Per-earning-miner monthly USD — revenue-based fallback tier. */
   257→    fallbackMonthlyUsd?: number;
   258→    /** GitHub-scraped requirements — GROUND TRUTH, wins over everything. */
   259→    scraped?: {
   260→      recommendedGpu?: string | null;
   261→      gpuCount?: number | null;
   262→      minVramGb?: number | null;
   263→      hosting?: HostingRequirements | null;
   264→      requirementsSource?: string | null;
   265→    } | null;
   266→    /** Curated official mechanics (mechanics.ts) — reward window + ops costs. */
   267→    mechanics?: SubnetMechanics | null;
   268→  }
   269→): SubnetHardwareProfile {
   270→  // --- Base classification: work-type keywords → tier (existing logic) ----
   271→  const base = (() => {
   272→    const text = `${name ?? ""} ${description ?? ""} ${fallbacks?.fallbackCategory ?? ""}`;
   273→    for (const rule of CATEGORY_RULES) {
   274→      if (rule.keywords.test(text)) {
   275→        return {
   276→          profile: {
   277→            category: rule.category,
   278→            tier: rule.tier,
   279→            minVramGb: rule.tier.minVramGb,
   280→            recommendedGpu: rule.tier.recommendedGpu,
   281→            monthlyCostUsd: rule.tier.monthlyRentUsd + (rule.infra ?? INFRA_BASE_USD),
   282→            isGpuWorkload: rule.tier.minVramGb > 0,
   283→            classified: true,
   284→          } as SubnetHardwareProfile,
   285→          infraUsd: rule.infra ?? INFRA_BASE_USD,
   286→        };
   287→      }
   288→    }
   289→    // No keyword match — fall back to curated VRAM data, else revenue tier.
   290→    if (fallbacks?.fallbackVramGb && fallbacks.fallbackVramGb > 0) {
   291→      const tier = Object.values(GPU_TIERS).find(
   292→        (t) => t.minVramGb === fallbacks.fallbackVramGb
   293→      ) ?? GPU_TIERS.consumer24;
   294→      return {
   295→        profile: {
   296→          category: fallbacks.fallbackCategory || "Unclassified workload",
   297→          tier,
   298→          minVramGb: tier.minVramGb,
   299→          recommendedGpu: fallbacks.fallbackGpu ?? tier.recommendedGpu,
   300→          monthlyCostUsd: tier.monthlyRentUsd + INFRA_BASE_USD,
   301→          isGpuWorkload: true,
   302→          classified: false,
   303→        } as SubnetHardwareProfile,
   304→        infraUsd: INFRA_BASE_USD,
   305→      };
   306→    }
   307→    const tier = estimateGpuTierFromRevenue(fallbacks?.fallbackMonthlyUsd ?? 0);
   308→    return {
   309→      profile: {
   310→        category: fallbacks?.fallbackCategory || "Unclassified workload",
   311→        tier,
   312→        minVramGb: tier.minVramGb,
   313→        recommendedGpu: tier.recommendedGpu,
   314→        monthlyCostUsd: tier.monthlyRentUsd + INFRA_BASE_USD,
   315→        isGpuWorkload: tier.minVramGb > 0,
   316→        classified: false,
   317→      } as SubnetHardwareProfile,
   318→      infraUsd: INFRA_BASE_USD,
   319→    };
   320→  })();
   321→
   322→  // --- Scraped repo requirements layer on top: the subnet's own README ----
   323→  // (or its miner repo) is ground truth for GPU model + hosting rules.
   324→  // It overrides ONLY what it actually knows — a hosting-only README keeps
   325→  // the work-type GPU tier; a GPU-only README keeps the classifier category.
   326→  const scraped = fallbacks?.scraped;
   327→  const scrapedTier = scraped?.recommendedGpu
   328→    ? resolveGpuTierFromModel(scraped.recommendedGpu)
   329→    : null;
   330→  if (!scraped || (!scrapedTier && !scraped.hosting)) return base.profile;
   331→
   332→  const count = scraped.gpuCount && scraped.gpuCount > 1 ? scraped.gpuCount : 1;
   333→  const tier = scrapedTier ?? base.profile.tier;
   334→  // MECHANICS-1 — hosting-aware cost class: a subnet whose own README rejects
   335→  // container clouds (bareMetalOnly) can only be served from the dedicated
   336→  // market, so the per-GPU unit rent switches to the bare-metal rate.
   337→  const bareMetal = scraped.hosting?.bareMetalOnly === true;
   338→  const unitRent = bareMetal ? tier.bareMetalMonthlyUsd : tier.monthlyRentUsd;
   339→  // The scraper's raw string already carries the count prefix ("8x H200").
   340→  const gpuLabel = scraped.recommendedGpu?.trim() ?? base.profile.recommendedGpu;
   341→  // Official mechanics may demand extra always-on infra (Chutes: a separate
   342→  // non-GPU control-plane server for the miner API/gepetto/postgres).
   343→  const controlPlaneUsd =
   344→    bareMetal && fallbacks?.mechanics?.controlPlaneMonthlyUsd
   345→      ? fallbacks.mechanics.controlPlaneMonthlyUsd
   346→      : 0;
   347→  return {
   348→    ...base.profile,
   349→    tier,
   350→    minVramGb: tier.minVramGb,
   351→    recommendedGpu: gpuLabel,
   352→    monthlyCostUsd: unitRent * count + base.infraUsd + controlPlaneUsd,
   353→    isGpuWorkload: tier.minVramGb > 0,
   354→    gpuCount: count,
   355→    hosting: scraped.hosting ?? null,
   356→    requirementsSource: scraped.requirementsSource ?? null,
   357→    sourceScraped: true,
   358→    unitRentUsd: unitRent,
   359→    costClass: bareMetal ? "bare-metal" : "container",
   360→  };
   361→}

   432→/**
   433→ * Score one subnet the way a miner reads it. Deterministic per snapshot —
   434→ * the same inputs always produce the same score so ranks stay stable.
   435→ */
   436→export function scoreMinersLedger(inputs: {
   437→  live: LiveSubnetMetrics;
   438→  taoUsd: number;
   439→  hardware?: SubnetHardwareProfile | null;
   440→  taoChange24h?: number;
   441→  /** Blocks since the subnet was registered (for the maturity bonus). */
   442→  liveAgeBlocks?: number | null;
   443→  /** Curated official mechanics (mechanics.ts) — ramp window + infra. */
   444→  mechanics?: SubnetMechanics | null;
   445→  /** User-configurable cost lines (Profitability Engine settings). When
   446→   *  absent, market defaults apply — same engine, same shape. */
   447→  costs?: {
   448→    hardwareMode?: "rent" | "owned";
   449→    electricityUsdPerKwh?: number;
   450→    storageMonthlyUsd?: number;
   451→    /** 0/undefined = auto infra from the work-type classifier. */
   452→    infraMonthlyUsd?: number;
   453→    otherOpexMonthlyUsd?: number;
   454→    includeBurnAmortization?: boolean;
   455→    amortizeBurnMonths?: number;
   456→  };
   457→}): MinerLedger {
   458→  const { live, taoUsd } = inputs;
   459→  const usd = taoUsd || 0;
   460→
   461→  // --- Top line: newcomer-adjusted per-miner revenue ----------------------
   462→  // Miners get 41% of the subnet's emission (docs constant); the chain-
   463→  // measured minerEmissionTaoPerDay already reflects the realized split.
   464→  //
   465→  // The decision a miner makes is "if I register TODAY, what does a
   466→  // mid-pack seat pay?" — so the ledger starts from the per-REGISTERED
   467→  // slot (zero-earning seats included: on many subnets 250+ UIDs are
   468→  // registered and only a handful earn), then applies the median/mean
   469→  // spread share so a whale UID can't inflate the estimate.
   470→  // perEarningDailyTao (mean over rewarded UIDs) is kept as the upside
   471→  // reference — what current earners make.
   472→  const rewarded = Math.max(live.rewardedMiners ?? 0, 0);
   473→  const registered = Math.max(live.minersCount, 1);
   474→  const minerEmissionTaoPerDay = live.minerEmissionTaoPerDay ?? 0;
   475→  const perRegisteredDailyTao =
   476→    minerEmissionTaoPerDay > 0 ? minerEmissionTaoPerDay / registered : 0;
   477→  const perEarningDailyTao =
   478→    rewarded > 0 && minerEmissionTaoPerDay > 0
   479→      ? minerEmissionTaoPerDay / rewarded
   480→      : minerEmissionTaoPerDay > 0
   481→        ? perRegisteredDailyTao // vec unavailable — no zero-slot discount available
   482→        : 0;
   483→  const rewardMedianShare = live.incentiveMedianShare ?? 1;
   484→  // Newcomer-adjusted mid-pack revenue:
   485→  //  - reward vec available  → perRegistered × medianShare
   486→  //  - reward vec unavailable→ perRegistered × 0.6 (conservative)
   487→  const expectedDailyTao =
   488→    rewarded > 0 && minerEmissionTaoPerDay > 0
   489→      ? perRegisteredDailyTao * rewardMedianShare
   490→      : minerEmissionTaoPerDay > 0
   491→        ? perRegisteredDailyTao * 0.6
   492→        : 0;
   493→  const grossMonthlyUsd = Math.round(expectedDailyTao * 30 * usd);
   494→  const dailyAlpha = live.movingPrice > 0 ? expectedDailyTao / live.movingPrice : 0;
   495→
   496→  // --- Hardware profile (work type → GPU cost) ---------------------------
   497→  const fallbackMonthly = grossMonthlyUsd > 0 ? grossMonthlyUsd : undefined;
   498→  const hardware =
   499→    inputs.hardware ??
   500→    classifySubnetHardware(live.name, live.identityDescription, {
   501→      fallbackMonthlyUsd: fallbackMonthly,
   502→    });
   503→  // --- Cost stack (Profitability Engine lines) ----------------------------
   504→  // gpuCount > 1 (scraped, e.g. "8x H200") scales the GPU line: the requirement
   505→  // is a multi-GPU server, not one card. Owned mode scales the power draw.
   506→  // MECHANICS-1: unitRent switches to the dedicated rate when the subnet's
   507→  // hosting rules reject containers — the ONLY compliant hosting class.
   508→  const costs = inputs.costs ?? {};
   509→  const gpuCount = hardware.gpuCount && hardware.gpuCount > 1 ? hardware.gpuCount : 1;
   510→  const unitRent = hardware.unitRentUsd ??
   511→    (hardware.hosting?.bareMetalOnly
   512→      ? hardware.tier.bareMetalMonthlyUsd
   513→      : hardware.tier.monthlyRentUsd);
   514→  const gpuCost =
   515→    costs.hardwareMode === "owned"
   516→      ? electricityMonthlyUsd(
   517→          hardware.tier.powerWatts * gpuCount,
   518→          costs.electricityUsdPerKwh ?? 0.12
   519→        )
   520→      : unitRent * gpuCount;
   521→  const infraCost =
   522→    costs.infraMonthlyUsd != null && costs.infraMonthlyUsd > 0
   523→      ? costs.infraMonthlyUsd
   524→      : Math.max(hardware.monthlyCostUsd - unitRent * gpuCount, INFRA_BASE_USD);
   525→  const storageCost = costs.storageMonthlyUsd ?? 0;
   526→  const otherOpex = costs.otherOpexMonthlyUsd ?? 0;
   527→  const burnUsd =
   528→    live.burnCostTao != null && costs.includeBurnAmortization
   529→      ? Math.max(live.burnCostTao, 0) * usd
   530→      : 0;
   531→  const amortizedBurn = burnUsd / Math.max(costs.amortizeBurnMonths ?? 3, 1);
   532→  const totalCosts = gpuCost + infraCost + storageCost + otherOpex + amortizedBurn;
   533→  // Zero-revenue subnets are honest losers: the cost stack still applies.
   534→  const netMonthlyUsd = grossMonthlyUsd > 0
   535→    ? Math.round(grossMonthlyUsd - totalCosts)
   536→    : -Math.round(totalCosts);
   537→
   538→  // --- Alpha economics ----------------------------------------------------
   539→  const alphaPriceTao = live.movingPrice;
   540→  const alphaPriceUsd = alphaPriceTao * usd;
   541→  const liquidityTao = live.subnetTao; // TAO side of the pool = exit liquidity
   542→  const slippagePct =
   543→    live.alphaOut > 0 && dailyAlpha > 0
   544→      ? Math.round((dailyAlpha / live.alphaOut) * 10000) / 100
   545→      : null;
   546→
   547→  // --- Seat safety inputs -------------------------------------------------
   548→  const maxUids = live.maxUids ?? 0;
   549→  const freeSlots = maxUids > 0 ? Math.max(0, maxUids - live.minersCount) : null;
   550→  const totalSlots = maxUids > 0 ? maxUids : null;
   551→  const rewardedRatio = live.rewardedMiners != null && live.rewardedMiners > 0
   552→    ? Math.min(1, live.rewardedMiners / registered)
   553→    : live.rewardedMiners === 0
   554→      ? 0
   555→      : null;
   556→  const top10 = live.top10IncentiveShare;
   557→
   558→  // Newcomer ramp. Default: bonds build via EMA (α=0.1), and on concentrated
   559→  // subnets a fresh UID starts at zero. Research: 1-3 months of tuning typical.
   560→  // MECHANICS-1: when the subnet's OFFICIAL docs define the reward metric as a
   561→  // rolling N-day window (Chutes: "7 day sum of compute"), the official window
   562→  // replaces the heuristic — a stable newcomer reaches full weight within it.
   563→  const mechanics = inputs.mechanics ?? null;
   564→  const mechanicsRampWeeks =
   565→    mechanics?.rewardWindowDays != null
   566→      ? Math.max(0.5, mechanics.rewardWindowDays / 7)
   567→      : null;
   568→  const rampWeeks =
   569→    mechanicsRampWeeks ??
   570→    (3 +
   571→      (rewardedRatio != null ? (1 - rewardedRatio) * 6 : 3) +
   572→      (top10 != null && top10 > 0.7 ? 3 : 0) +
   573→      (freeSlots != null && freeSlots === 0 ? 2 : 0));
   574→  const rampWeeksSource: MinerLedgerDiagnostics["rampWeeksSource"] =
   575→    mechanicsRampWeeks != null
   576→      ? mechanics?.provenance === "derived"
   577→        ? "readme-derived"
   578→        : "official-reward-window"
   579→      : "bond-ema-heuristic";
   580→
   581→  // --- Pillar 1: Net ROI --------------------------------------------------
   582→  // $0 → 10, +$100 → 54, +$500 → 69, +$1k → 76, +$5k → 92; losses slide to 2.
   583→  const netRoi =
   584→    netMonthlyUsd >= 0
   585→      ? clampScore(10 + 22 * Math.log10(netMonthlyUsd + 1), 2, 97)
   586→      : clampScore(10 + netMonthlyUsd / 100, 2, 10);
   587→
   588→  // --- Pillar 2: Seat safety ----------------------------------------------
   589→  const roomRatio = maxUids > 0 ? freeSlots! / maxUids : 0.5;
   590→  const scoreRoom = clampScore(10 + roomRatio * 45, 10, 55); // full subnet → 10
   591→  const scoreSpread =
   592→    top10 != null ? clampScore(100 - (top10 - 0.2) * 100, 10, 95) : 50;
   593→  const scoreBurn =
   594→    live.burnCostTao != null
   595→      ? clampScore(95 - Math.log10(live.burnCostTao + 1) * 25, 30, 95)
   596→      : 55;
   597→  const scoreImmunity =
   598→    live.immunityBlocks != null
   599→      ? clampScore(30 + (live.immunityBlocks / 7200) * 50, 30, 80)
   600→      : 45;
   601→  const seatSafety = !live.emissionEnabled
   602→    ? 12
   603→    : clampScore(
   604→        scoreRoom * 0.35 + scoreSpread * 0.35 + scoreBurn * 0.15 + scoreImmunity * 0.15,
   605→        5,
   606→        96
   607→      );
   608→
   609→  // --- Pillar 3: Alpha economics -------------------------------------------
   610→  const change = live.alphaPriceChange24h;
   611→  const scoreTrend = change != null ? clampScore(50 + change * 1.5, 10, 95) : 50;
   612→  const scoreLiq =
   613→    liquidityTao > 0
   614→      ? clampScore(20 + 19 * Math.log10(liquidityTao / 500 + 1), 10, 95)
   615→      : 10;
   616→  const scoreSlip =
   617→    slippagePct != null ? clampScore(95 - slippagePct * 1200, 15, 95) : 50;
   618→  const alphaEconomics = usd <= 0
   619→    ? 50
   620→    : clampScore(scoreTrend * 0.4 + scoreLiq * 0.3 + scoreSlip * 0.3, 5, 96);
   621→
   622→  // --- Pillar 4: Earning reality -------------------------------------------
   623→  const scoreRewarded =
   624→    rewardedRatio != null ? clampScore(15 + rewardedRatio * 70, 15, 85) : 50;
   625→  const scoreConc =
   626→    top10 != null ? clampScore(100 - (top10 - 0.2) * 90, 15, 92) : 50;
   627→  const scoreRamp = clampScore(90 - rampWeeks * 5, 20, 90);
   628→  const earningReality = !live.emissionEnabled
   629→    ? 8
   630→    : clampScore(scoreRewarded * 0.4 + scoreConc * 0.3 + scoreRamp * 0.3, 5, 96);
   631→
   632→  // --- Pillar 5: Fit & feasibility ------------------------------------------
   633→  let fitFeasibility = 25; // emission disabled base
   634→  if (live.emissionEnabled) {
   635→    fitFeasibility = 40;
   636→    const validators = live.validatorsCount ?? 0;
   637→    fitFeasibility += clampScore(validators * 2.5, 0, 20); // validator coverage
   638→    if (live.identityGithub) fitFeasibility += 12;
   639→    if (live.identityDescription) fitFeasibility += 6;
   640→    if (live.registeredAt != null && inputs.liveAgeBlocks != null) {
   641→      const ageDays = inputs.liveAgeBlocks / 7200;
   642→      fitFeasibility += clampScore(Math.min(ageDays / 30, 1) * 12, 0, 12); // maturity
   643→    }
   644→    if (live.immunityBlocks != null && live.immunityBlocks < 4096) fitFeasibility -= 5;
   645→  }
   646→  fitFeasibility = clampScore(fitFeasibility, 5, 92);
   647→
   648→  const components: ScoreComponents = {
   649→    net_roi: Math.round(netRoi * 10) / 10,
   650→    seat_safety: Math.round(seatSafety * 10) / 10,
   651→    alpha_economics: Math.round(alphaEconomics * 10) / 10,
   652→    earning_reality: Math.round(earningReality * 10) / 10,
   653→    fit_feasibility: Math.round(fitFeasibility * 10) / 10,
   654→  };
   655→
   656→  const diag: MinerLedgerDiagnostics = {
   657→    perEarningDailyTao: Math.round(perEarningDailyTao * 10000) / 10000,
   658→    expectedDailyTao: Math.round(expectedDailyTao * 10000) / 10000,
   659→    rewardMedianShare: live.incentiveMedianShare,
   660→    perRegisteredDailyTao: Math.round(perRegisteredDailyTao * 10000) / 10000,
   661→    grossMonthlyUsd,
   662→    alphaPriceTao,
   663→    alphaPriceUsd: Math.round(alphaPriceUsd * 1000) / 1000,
   664→    alphaChange24h: change ?? null,
   665→    liquidityTao: Math.round(liquidityTao),
   666→    slippagePct,
   667→    category: hardware.category,
   668→    gpuLabel: hardware.tier.label,
   669→    recommendedGpu: hardware.recommendedGpu,
   670→    minVramGb: hardware.minVramGb,
   671→    gpuCostMonthlyUsd: Math.round(gpuCost),
   672→    gpuCount: hardware.gpuCount ?? null,
   673→    hosting: hardware.hosting ?? null,
   674→    requirementsSource: hardware.requirementsSource ?? null,
   675→    costClass: hardware.costClass ?? (hardware.hosting?.bareMetalOnly ? "bare-metal" : "container"),
   676→    unitRentUsd: Math.round(unitRent),
   677→    infraCostMonthlyUsd: Math.round(infraCost),
   678→    storageCostMonthlyUsd: Math.round(storageCost),
   679→    otherOpexMonthlyUsd: Math.round(otherOpex),
   680→    amortizedBurnUsd: Math.round(amortizedBurn),
   681→    totalCostMonthlyUsd: Math.round(totalCosts),
   682→    gpuPowerWatts: hardware.tier.powerWatts * gpuCount,
   683→    netMonthlyUsd,
   684→    netDailyTao:
   685→      usd > 0 ? Math.round((netMonthlyUsd / 30 / usd) * 10000) / 10000 : 0,
   686→    freeSlots,
   687→    totalSlots,
   688→    rewardedRatio: rewardedRatio != null ? Math.round(rewardedRatio * 1000) / 1000 : null,
   689→    top10IncentiveShare: top10,
   690→    burnCostTao: live.burnCostTao,
   691→    immunityBlocks: live.immunityBlocks,
   692→    rampWeeks: Math.round(rampWeeks * 10) / 10,
   693→    rampWeeksSource,
   694→    mechanicsApplied:
   695→      mechanics && mechanicsOverridesRamp(mechanics)
   696→        ? mechanics.sources.map((s) => s.label)
   697→        : null,
   698→    emissionEnabled: live.emissionEnabled,
   699→    hardwareClassified: hardware.classified,
   700→  };
   701→
   702→  return { components, factors: deriveFactors(components), diag };
   703→}
   704→
   705→// ---------------------------------------------------------------------------
   706→// Chance to earn · month 1 — a newcomer's honest odds of earning ANY reward
   707→// during their first month of continuous, good-uptime mining. Uptime alone
   708→// keeps the seat; the share of slots that actually earn decides whether the
   709→// wallet moves. Dominant factor: rewardedRatio (share of registered slots
   710→// that earned last epoch), discounted for whale dominance, full subnets
   711→// (must displace an incumbent) and long bond-EMA ramps.
   712→// ---------------------------------------------------------------------------
   713→
   714→export type EarnChanceLevel = "high" | "medium" | "low" | "none";
   715→
   716→export interface EarnChance {
   717→  level: EarnChanceLevel;
   718→  /** 0–100 modeled chance that a newcomer earns something in month 1. */
   719→  pct: number;
   720→  /** Human explanation of the dominant factors. */
   721→  note: string;
   722→}
   723→
   724→export function computeEarnChance(inputs: {
   725→  /** Share of registered slots that earned reward last epoch (0..1). */
   726→  rewardedRatio: number | null;
   727→  /** Free UID slots (null = unknown). 0 means you must displace an incumbent. */
   728→  freeSlots: number | null;
   729→  /** Top-10% share of epoch incentive (0..1) — whale dominance. */
   730→  top10IncentiveShare: number | null;
   731→  /** Newcomer bond-EMA ramp estimate in weeks. */
   732→  rampWeeks: number | null;
   733→}): EarnChance {
   734→  const { rewardedRatio, freeSlots, top10IncentiveShare, rampWeeks } = inputs;
   735→
   736→  // Base: the fraction of slots that actually earn — the single strongest
   737→  // predictor of whether a newcomer earns anything at all.
   738→  let pct = rewardedRatio != null ? rewardedRatio * 100 : 0;
   739→  const factors: string[] = [];
   740→
   741→  // Whale dominance: when the top 10% of UIDs take nearly everything, the
   742→  // rewarded tail is scraps — being "rewarded" still pays almost nothing.
   743→  if (top10IncentiveShare != null && top10IncentiveShare > 0.5) {
   744→    pct *= 1 - Math.min(1, top10IncentiveShare - 0.5) * 0.6;
   745→    factors.push(
   746→      `top-10% of UIDs take ${Math.round(top10IncentiveShare * 100)}% of rewards`
   747→    );
   748→  }
   749→
   750→  // Full subnet: no free seats — you must outperform an incumbent to get in,
   751→  // and you start from zero bonds against their established ones.
   752→  if (freeSlots === 0) {
   753→    pct *= 0.7;
   754→    factors.push("subnet full — must displace an incumbent");
   755→  }
   756→
   757→  // Long ramp: month 1 is mostly bond-building; long ramps leave little of it.
   758→  if (rampWeeks != null && rampWeeks > 8) {
   759→    pct *= 0.8;
   760→    factors.push(`~${Math.round(rampWeeks)} wk ramp to full bonds`);
   761→  }
   762→
   763→  pct = Math.round(Math.max(0, Math.min(100, pct)));
   764→  const level: EarnChanceLevel =
   765→    rewardedRatio != null && rewardedRatio <= 0
   766→      ? "none"
   767→      : pct >= 50
   768→        ? "high"
   769→        : pct >= 20
   770→          ? "medium"
   771→          : pct > 0
   772→            ? "low"
   773→            : "none";
   774→
   775→  const base = rewardedRatio != null
   776→    ? `${rewardedRatio < 0.1 && rewardedRatio > 0 ? (rewardedRatio * 100).toFixed(1) : Math.round(rewardedRatio * 100)}% of slots earned last epoch`
   777→    : "reward spread unknown";
   778→  const note =
   779→    factors.length > 0 ? `${base} · ${factors.join(" · ")}` : base;
   780→
   781→  return { level, pct, note };
   782→}

   784→// ---------------------------------------------------------------------------
   785→// Seat Chance — "can I actually GET IN?" (vs EarnChance's "will I earn?").
   786→//
   787→// Bittensor registration mechanics:
   788→//   • Subnets have a UID cap (maxUids). Free slots → register via burn or PoW.
   789→//   • FULL subnet → registration STILL works: a burn registration pays the
   790→//     dynamic recycle cost and the chain replaces the worst-performing
   791→//     NON-IMMUNE uid. PoW registrations compete for per-block slots the same
   792→//     way. You are only locked out if every uid is immune (rare — immunity
   793→//     covers ~14h by default) or registration is disabled on the subnet.
   794→//   • immunityBlocks = how long YOUR new seat cannot be deregistered — the
   795→//     runway to start earning before becoming replaceable yourself.
   796→// ---------------------------------------------------------------------------
   797→
   798→/**
   799→ * Burn-quote display precision — the chain's burn hyperparam can sit at the
   800→ * floor (e.g. Chutes ~0.0005 TAO); 3 decimals would show "0.001" — a 2x
   801→ * misquote shown to a user deciding whether to burn. <0.01 → 4 decimals,
   802→ * <1 → 3, else 2.
   803→ */
   804→export function formatBurnTao(n: number): string {
   805→  if (n < 0.01) return n.toFixed(4);
   806→  if (n < 1) return n.toFixed(3);
   807→  return n.toFixed(2);
   808→}
   809→
   810→export type SeatVerdict = "open" | "burn-entry" | "waitlist" | "unknown";
   811→
   812→export interface SeatChance {
   813→  verdict: SeatVerdict;
   814→  /** Registration is possible right now (burn path or free slot). */
   815→  canRegister: boolean;
   816→  slotsFree: number | null;
   817→  totalSlots: number | null;
   818→  /** 0–100 — how full the uid space is. */
   819→  fillPct: number | null;
   820→  burnCostTao: number | null;
   821→  /** Protection window for a NEW seat, in hours (blocks × 12s). */
   822→  immunityHours: number | null;
   823→  /** Share of uids that earned NOTHING last epoch — the replaceable bottom
   824→   *  a newcomer displaces from. High = churn churns in your favor. */
   825→  replaceableShare: number | null;
   826→  headline: string;
   827→  detail: string;
   828→}
   829→
   830→export function assessSeatChance(inputs: {
   831→  minersCount: number | null;
   832→  maxUids: number | null;
   833→  /** Already-computed free slots (skip when you only have counts). */
   834→  freeSlots?: number | null;
   835→  burnCostTao: number | null;
   836→  immunityBlocks: number | null;
   837→  /** Rewarded uids (either give the ratio directly or the absolute count). */
   838→  rewardedRatio?: number | null;
   839→  rewardedMiners?: number | null;
   840→}): SeatChance {
   841→  const { maxUids, burnCostTao, immunityBlocks } = inputs;
   842→  const minersCount = inputs.minersCount ?? 0;
   843→  const slotsFree =
   844→    inputs.freeSlots != null
   845→      ? Math.max(0, inputs.freeSlots)
   846→      : maxUids != null && maxUids > 0
   847→        ? Math.max(0, maxUids - minersCount)
   848→        : null;
   849→  const fillPct =
   850→    maxUids != null && maxUids > 0
   851→      ? Math.min(100, Math.round((minersCount / maxUids) * 100))
   852→      : null;
   853→  let replaceableShare: number | null = null;
   854→  if (inputs.rewardedRatio != null) {
   855→    replaceableShare = Math.max(0, Math.min(1, 1 - inputs.rewardedRatio));
   856→  } else if (inputs.rewardedMiners != null && minersCount > 0) {
   857→    replaceableShare = Math.max(0, Math.min(1, 1 - inputs.rewardedMiners / minersCount));
   858→  }
   859→  const immunityHours =
   860→    immunityBlocks != null && immunityBlocks > 0
   861→      ? Math.round(((immunityBlocks * 12) / 3600) * 10) / 10
   862→      : null;
   863→  const immunityNote = immunityHours != null
   864→    ? `a new seat is immune for ~${immunityHours}h — your runway to start earning before you're replaceable`
   865→    : null;
   866→  const deadNote = replaceableShare != null && replaceableShare > 0.05
   867→    ? `${Math.round(replaceableShare * 100)}% of uids earned nothing last epoch — a deep replaceable bottom`
   868→    : replaceableShare != null
   869→      ? "almost every seat earned last epoch — displacement targets are scarce and strong"
   870→      : null;
   871→
   872→  const unknown: SeatChance = {
   873→    verdict: "unknown",
   874→    canRegister: true,
   875→    slotsFree,
   876→    totalSlots: maxUids ?? null,
   877→    fillPct,
   878→    burnCostTao,
   879→    immunityHours,
   880→    replaceableShare,
   881→    headline: "slot data unavailable",
   882→    detail:
   883→      "UID capacity was not returned by the chain snapshot this pass — refresh the network data. " +
   884→      (burnCostTao != null ? `Burn registration quote: ~${formatBurnTao(burnCostTao)} TAO.` : ""),
   885→  };
   886→
   887→  if (slotsFree == null || maxUids == null || maxUids <= 0) return unknown;
   888→
   889→  // --- Open: free slots exist ---------------------------------------------
   890→  if (slotsFree > 0) {
   891→    return {
   892→      verdict: "open",
   893→      canRegister: true,
   894→      slotsFree,
   895→      totalSlots: maxUids,
   896→      fillPct,
   897→      burnCostTao,
   898→      immunityHours,
   899→      replaceableShare,
   900→      headline: `${slotsFree} of ${maxUids} slots free`,
   901→      detail:
