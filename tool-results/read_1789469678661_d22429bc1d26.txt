     6→  Dialog,
     7→  DialogContent,
     8→  DialogHeader,
     9→  DialogTitle,
    10→  DialogDescription,
    11→} from "@/components/ui/dialog";

    17→import {
    18→  Loader2,

    42→  type RegistrationWizardContext,
    43→} from "@/lib/infranex/use-deployments";
    44→import { WalletRegistrationDialog } from "@/components/devops/wallet-registration-dialog";
    45→import { assessSeatChance } from "@/lib/infranex/miner-score";
    46→import type { SubnetRequirementsProfile } from "@/lib/devops/subnet-requirements";

    48→
    49→/**
    50→ * Deploy Wizard — the WHOLE mining journey in one dialog, in the user's own
    51→ * order:
    52→ *   1. Choose the subnet        (live chain list + seat verdicts)
    53→ *   2. Check required GPU       (real requirements profiler)
    54→ *   3. Buy the GPU              (filtered provider offers)
    55→ *   4. Install requirements     (automatic — real 10-step runner, live log)
    56→ *   5. Add hotkey & run         (wallet wizard hand-off, register LAST)
    57→ *
    58→ * Replaces the old 3-step CreateDeploymentDialog: nothing is scattered
    59→ * across views anymore — rent → install → register flows end-to-end here.
    60→ */
    61→
    62→const STEP_LABELS = ["Subnet", "Requirements", "GPU", "Install", "Hotkey & run"];
    63→
    64→/** What an entry point can preselect when opening the wizard (FLOW-1). */
    65→export interface OpenDeployWizardOptions {
    66→  /** Subnet chosen upstream (opportunity card / mining journey). */
    67→  netuid?: number | null;
    68→  /** GPU offer chosen upstream (GPU catalog "Provision"). */
    69→  offerId?: string | null;
    70→}
    71→
    72→interface DeployWizardProps {
    73→  open: boolean;
    74→  onOpenChange: (open: boolean) => void;
    75→  /** Preselect a subnet when opening (e.g. "Start mining" from an opportunity). */
    76→  initialNetuid?: number | null;
    77→  /** Preselect a GPU offer when opening (e.g. "Provision" from the catalog). */
    78→  initialOfferId?: string | null;
    79→  onCreated?: (id: string) => void;
    80→}
    81→
    82→export function DeployWizard({ open, onOpenChange, initialNetuid, initialOfferId, onCreated }: DeployWizardProps) {
    83→  const [step, setStep] = useState<1 | 2 | 3 | 4 | 5>(1);
    84→  const [netuid, setNetuid] = useState<number | null>(null);
    85→  const [offerId, setOfferId] = useState<string | null>(null);
    86→  const [minerName, setMinerName] = useState("");
    87→  const [walletName, setWalletName] = useState("infranex");
    88→  // WALLET-ECON-1 — saved wallet profiles prefill the wallet name (default
    89→  // profile wins) until the operator types their own.
    90→  const { data: walletProfiles } = useWallets();
    91→  const [walletTouched, setWalletTouched] = useState(false);
    92→  useEffect(() => {
    93→    if (!walletTouched && walletProfiles && walletProfiles.length > 0) {
    94→      const def = walletProfiles.find((w) => w.isDefault) ?? walletProfiles[0];
    95→      setWalletName(def.walletName);
    96→    }
    97→  }, [walletProfiles, walletTouched]);
    98→  const [depId, setDepId] = useState<string | null>(null);
    99→  const [regCtx, setRegCtx] = useState<RegistrationWizardContext | null>(null);
   100→  const [regOpen, setRegOpen] = useState(false);
   101→  const [search, setSearch] = useState("");
   102→  const { toast } = useToast();
   103→  const qc = useQueryClient();
   104→
   105→  // Seed the preselection on every closed → open TRANSITION (not on mount,
   106→  // not while open): the wizard is a singleton at the app root, so each
   107→  // entry point (opportunity, GPU catalog, journey) re-seeds cleanly.
   108→  // A subnet preselect lands on step 2 — "check required GPU" — the step
   109→  // the user explicitly asked for after choosing the subnet.
   110→  const wasOpen = useRef(false);
   111→  /* WINDUP-1: seeding the wizard's state on the closed→open transition is
   112→   * deliberate (singleton dialog re-seeded per entry point). The
   113→   * react-hooks/set-state-in-effect rule forbids the pattern wholesale, so
   114→   * scope the disable to exactly this block. */
   115→  /* eslint-disable react-hooks/set-state-in-effect */
   116→  useEffect(() => {
   117→    if (open && !wasOpen.current) {
   118→      const hasSubnet = initialNetuid != null;
   119→      setNetuid(initialNetuid ?? null);
   120→      setOfferId(initialOfferId ?? null);
   121→      setStep(hasSubnet ? 2 : 1);
   122→      setMinerName("");
   123→      setWalletName("infranex");
   124→      setDepId(null);
   125→      setRegCtx(null);
   126→      setRegOpen(false);
   127→      setSearch("");
   128→    }
   129→    wasOpen.current = open;
   130→  }, [open, initialNetuid, initialOfferId]);
   131→  /* eslint-enable react-hooks/set-state-in-effect */
   132→
   133→  const reset = () => {
   134→    setStep(1);
   135→    setNetuid(null);
   136→    setOfferId(null);
   137→    setMinerName("");
   138→    setWalletName("infranex");
   139→    setDepId(null);
   140→    setRegCtx(null);
   141→    setRegOpen(false);
   142→    setSearch("");
   143→  };
   144→
   145→  const handleClose = (o: boolean) => {
   146→    onOpenChange(o);
   147→    if (!o) reset();
   148→  };
   149→
   150→  // --- Step 1 data: live subnets (chain + curated merged) ------------------
   151→  const { data: net, isLoading: netLoading } = useNetwork();
   152→  const subnets = useMemo(() => mergeSubnets(net), [net]);
   153→  const liveSubnet = useMemo(
   154→    () => subnets.find((s) => s.netuid === netuid) ?? null,
   155→    [subnets, netuid]
   156→  );
   157→  const filtered = useMemo(() => {
   158→    const q = search.trim().toLowerCase();
   159→    const list = q
   160→      ? subnets.filter(
   161→          (s) =>
   162→            s.name.toLowerCase().includes(q) ||
   163→            String(s.netuid) === q ||
   164→            s.category.toLowerCase().includes(q)
   165→        )
   166→      : subnets;
   167→    return list;
   168→  }, [subnets, search]);
   169→
   170→  // --- Step 2 data: real requirements profile ------------------------------
   171→  const reqQ = useQuery({
   172→    queryKey: ["wizard-requirements", netuid],
   173→    queryFn: async () => {
   174→      const res = await fetch(`/api/devops/subnet-requirements?netuid=${netuid}`, {
   175→        cache: "no-store",
   176→      });
   177→      const j = await res.json();
   178→      if (!res.ok) throw new Error(j?.error ?? `HTTP ${res.status}`);
   179→      return j.profile as SubnetRequirementsProfile;
   180→    },
   181→    enabled: open && netuid != null && step >= 2,
   182→    staleTime: 10 * 60_000,
   183→  });
   184→  const profile = reqQ.data ?? null;
   185→  const requiredVram = profile?.minVramGb ?? liveSubnet?.minVramGb ?? null;
   186→
   187→  // --- Step 3 data: offers --------------------------------------------------
   188→  const { offers, snap: offerSnap } = useMergedGpuOffers();
   189→  // Legacy snapshots (no providers array) default to "configured" so the hint
   190→  // never nags when the server can't tell us.
   191→  const runpodConfigured = offerSnap?.providers
   192→    ? Boolean(offerSnap.providers.find((p) => p.id === "runpod")?.configured)
   193→    : true;
   194→  // TIER4 — Vast.ai rental adapter needs its own key.
   195→  const vastConfigured = offerSnap?.providers
   196→    ? Boolean(offerSnap.providers.find((p) => p.id === "vast")?.configured)
   197→    : true;
   198→  const matchingOffers = useMemo(() => {
   199→    if (requiredVram == null) return offers;
   200→    return offers
   201→      .filter((o) => o.vramGb >= requiredVram)
   202→      .sort((a, b) => a.hourlyPrice - b.hourlyPrice);
   203→  }, [offers, requiredVram]);
   204→  const offer: MergedGpuOffer | null = offers.find((o) => o.id === offerId) ?? null;
   205→  // TIER4 — the real-rental button routes by the selected offer's provider:
   206→  // a Vast.ai offer provisions through the vast adapter, everything else RunPod.
   207→  const realMode: "runpod" | "vast" =
   208→    offer?.provider && offer.provider.toLowerCase().includes("vast") ? "vast" : "runpod";
   209→  const realConfigured = realMode === "vast" ? vastConfigured : runpodConfigured;
   210→
   211→  // --- Step 4/5 data: the deployment record --------------------------------
   212→  const detail = useDeploymentDetail(step >= 4 ? depId : null);
   213→  const dep = detail.data ?? null;
   214→  const tickMut = useTickDeployment();
   215→  const regAction = useRegistrationAction();
   216→
   217→  // Derived step: the install step AUTO-COMPLETES into the hotkey step the
   218→  // moment the miner is running — no extra click, no setState-in-effect.
   219→  const effStep: 1 | 2 | 3 | 4 | 5 =
   220→    step === 4 && dep?.status === "started" ? 5 : step;
   221→
   222→  // Auto-tick the lifecycle while the user watches step 4.
   223→  useEffect(() => {
   224→    if (step !== 4 || !depId || !open) return;
   225→    if (!dep) return;
   226→    if (dep.status === "started" || dep.status === "failed" || dep.status === "terminated") return;
   227→    if (tickMut.isPending) return;
   228→    const t = setTimeout(() => tickMut.mutate(depId), 2600);
   229→    return () => clearTimeout(t);
   230→  }, [step, depId, open, dep?.status, dep?.installStatus, tickMut.isPending, tickMut.mutate]);
   231→
   232→  // Step 5: fetch the wallet-wizard hand-off context once.
   233→  useEffect(() => {
   234→    if (effStep !== 5 || !depId || regCtx) return;
   235→    let cancelled = false;
   236→    (async () => {
   237→      try {
   238→        const res = await fetch(`/api/deployments/${depId}/registration`, { cache: "no-store" });
   239→        const j = await res.json();
   240→        if (!cancelled && res.ok && j.wizard) setRegCtx(j.wizard as RegistrationWizardContext);
   241→      } catch {
   242→        /* the retry button re-fetches */
   243→      }
   244→    })();
   245→    return () => {
   246→      cancelled = true;
   247→    };
   248→  }, [effStep, depId, regCtx]);
   249→
   250→  const seat = liveSubnet
   251→    ? assessSeatChance({
   252→        minersCount: liveSubnet.minersCount,
   253→        maxUids: liveSubnet.maxUids ?? null,
   254→        burnCostTao: liveSubnet.burnCostTao ?? null,
   255→        immunityBlocks: liveSubnet.immunityBlocks ?? null,
   256→        rewardedMiners: liveSubnet.rewardedMiners ?? null,
   257→      })
   258→    : null;
   259→
   260→  const createMut = useCreateDeployment();
   261→  const handleCreate = async () => {
   262→    if (!liveSubnet || !offer) return;
   263→    try {
   264→      const created = await createMut.mutateAsync({
   265→        netuid: liveSubnet.netuid,
   266→        offerId: offer.id,
   267→        minerName: minerName.trim(),
   268→        walletName: walletName.trim() || undefined,
   269→        mode: realMode,
   270→      });
   271→      toast({
   272→        title: "GPU rented — installing requirements",
   273→        description: `${offer.model} for ${liveSubnet.name} (α${liveSubnet.netuid})`,
   274→      });
   275→      setDepId(created.id);
   276→      onCreated?.(created.id);
   277→      setStep(4);
   278→    } catch (e) {
   279→      toast({
   280→        title: "Could not rent the GPU",
   281→        description: e instanceof Error ? e.message : "Unknown error",
   282→        variant: "destructive",
   283→      });
   284→    }
   285→  };
   286→
   287→  const registered = dep?.registrationState === "registered";
   288→  const restartPending = registered && dep != null && !dep.restartedAfterRegistration;
   289→  const fullyLive = registered && dep?.restartedAfterRegistration === true;
   290→
   291→  return (
   292→    <>
   293→      <Dialog open={open} onOpenChange={handleClose}>
   294→        <DialogContent className="max-h-[92vh] max-w-3xl overflow-y-auto custom-scroll">
   295→          <DialogHeader>
   296→            <DialogTitle className="flex items-center gap-2 text-display text-2xl">
   297→              <Rocket className="h-5 w-5 text-primary" />
   298→              Deploy a miner
   299→            </DialogTitle>
   300→            <DialogDescription>
   301→              Subnet → requirements → GPU → automatic install → hotkey &amp; run.
   302→              {dep ? (
   303→                <span className="ml-1 font-medium text-foreground">
   304→                  Mining {dep.subnetName} (α{dep.netuid}) on {dep.gpuModel}.
   305→                </span>
   306→              ) : null}
   307→            </DialogDescription>
   308→          </DialogHeader>
   309→
   310→          {/* Step indicator */}
   311→          <div className="grid grid-cols-5 gap-1.5" aria-label="Wizard progress">
   312→            {STEP_LABELS.map((label, i) => {
   313→              const n = (i + 1) as 1 | 2 | 3 | 4 | 5;
   314→              const reached = effStep >= n;
   315→              const current = effStep === n;
   316→              return (
   317→                <div key={label} className="space-y-1">
   318→                  <div
   319→                    className={cn(
   320→                      "h-1.5 rounded-full transition-colors",
   321→                      reached ? "bg-primary" : "bg-muted",
   322→                      current && "ring-2 ring-primary/30"
   323→                    )}
   324→                  />
   325→                  <p
   326→                    className={cn(
   327→                      "text-[10px] leading-tight",
   328→                      current ? "font-semibold text-foreground" : "text-muted-foreground"
   329→                    )}
   330→                  >
   331→                    {i + 1}. {label}
   332→                  </p>
   333→                </div>
   334→              );
   335→            })}
   336→          </div>
   337→
   338→          {/* ------------------------- STEP 1 ------------------------- */}
   339→          {effStep === 1 && (
   340→            <div className="space-y-3">
   341→              <div className="relative">
   342→                <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   343→                <Input
   344→                  placeholder="Search 120+ live subnets — name, netuid or category…"
   345→                  value={search}
   346→                  onChange={(e) => setSearch(e.target.value)}
   347→                  className="pl-8"
   348→                />
   349→              </div>
   350→              {netLoading ? (
   351→                <div className="flex items-center justify-center gap-2 py-12 text-muted-foreground">
   352→                  <Loader2 className="h-4 w-4 animate-spin" />
   353→                  <span className="text-sm">Loading the live chain…</span>
   354→                </div>
   355→              ) : (
   356→                <div className="max-h-80 space-y-1.5 overflow-y-auto custom-scroll pr-1">
   357→                  {filtered.map((s) => {
   358→                    const seatHere = assessSeatChance({
   359→                      minersCount: s.minersCount,
   360→                      maxUids: s.maxUids ?? null,
   361→                      burnCostTao: s.burnCostTao ?? null,
   362→                      immunityBlocks: s.immunityBlocks ?? null,
   363→                      rewardedMiners: s.rewardedMiners ?? null,
   364→                    });
   365→                    const selected = netuid === s.netuid;
   366→                    return (
   367→                      <button
   368→                        key={s.netuid}
   369→                        type="button"
   370→                        onClick={() => {
   371→                          setNetuid(s.netuid);
   372→                          setMinerName(
   373→                            minerName || `${s.name.toLowerCase().replace(/[^a-z0-9]+/g, "-").slice(0, 24)}-01`
   374→                          );
   375→                        }}
   376→                        className={cn(
   377→                          "w-full rounded-lg border p-3 text-left transition-colors",
   378→                          selected
   379→                            ? "border-primary/50 bg-primary/[0.06]"
   380→                            : "border-border/60 bg-card/30 hover:border-border"
   381→                        )}
   382→                      >
   383→                        <div className="flex items-center justify-between gap-2">
   384→                          <div className="flex min-w-0 items-center gap-2">
   385→                            <span className="mono text-xs text-muted-foreground">α{s.netuid}</span>
   386→                            <span className="truncate text-sm font-semibold">{s.name}</span>
   387→                            <Badge variant="outline" className="hidden text-[9px] sm:inline-flex">
   388→                              {s.category}
   389→                            </Badge>
   390→                          </div>
   391→                          <SeatChip verdict={seatHere.verdict} headline={seatHere.headline} />
   392→                        </div>
   393→                        <p className="mt-1 truncate text-xs text-muted-foreground">
   394→                          {s.minersCount}/{s.maxUids ?? "?"} seats · min {s.minVramGb}GB VRAM ·{" "}
   395→                          {s.rewardedMiners ?? "?"} earned last epoch
   396→                        </p>
   397→                      </button>
   398→                    );
   399→                  })}
   400→                  {filtered.length === 0 && (
   401→                    <p className="py-8 text-center text-sm text-muted-foreground">
   402→                      No subnet matches “{search}”.
   403→                    </p>
   404→                  )}
   405→                </div>
   406→              )}
   407→              <div className="flex justify-end">
   408→                <Button disabled={netuid == null} onClick={() => setStep(2)} className="gap-1.5">
   409→                  Check requirements
   410→                </Button>
   411→              </div>
   412→            </div>
   413→          )}
   414→
   415→          {/* ------------------------- STEP 2 ------------------------- */}
   416→          {effStep === 2 && liveSubnet && (
   417→            <div className="space-y-3">
   418→              {reqQ.isLoading ? (
   419→                <div className="flex items-center justify-center gap-2 py-12 text-muted-foreground">
   420→                  <Loader2 className="h-4 w-4 animate-spin" />
   421→                  <span className="text-sm">
   422→                    Profiling α{liveSubnet.netuid} — chain identity + GitHub repo…
   423→                  </span>
   424→                </div>
   425→              ) : reqQ.isError || !profile ? (
   426→                <div className="flex items-start gap-2 rounded-lg border border-warning/30 bg-warning/[0.04] p-4">
   427→                  <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-warning" />
   428→                  <div className="text-xs">
   429→                    <p className="font-medium text-warning">Profiler unavailable</p>
   430→                    <p className="mt-1 text-muted-foreground">
   431→                      Falling back to the catalog requirement: min {liveSubnet.minVramGb}GB ·{" "}
   432→                      {liveSubnet.recommendedGpu}.
   433→                    </p>
   434→                  </div>
   435→                </div>
   436→              ) : (
   437→                <div className="space-y-3">
   438→                  <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   439→                    <ReqStat label="Min VRAM" value={`${profile.minVramGb} GB`} />
   440→                    <ReqStat label="Recommended GPU" value={profile.recommendedGpu} small />
   441→                    <ReqStat label="Runtime" value={profile.dockerfileFound ? "Docker" : "venv"} />
   442→                    <ReqStat
   443→                      label="Python / CUDA"
   444→                      value={`${profile.pythonVersion ?? "distro"} · ${profile.cudaMinVersion ?? "—"}`}
   445→                      small
   446→                    />
   447→                  </div>
   448→                  <div className="rounded-lg border border-border/60 bg-card/40 p-3 text-xs">
   449→                    <p className="text-eyebrow mb-1.5 text-muted-foreground">
   450→                      What gets installed automatically
   451→                    </p>
   452→                    <p className="leading-relaxed text-muted-foreground">
   453→                      {profile.osPackages.length} OS packages · {profile.pipPackageCount} pip deps
   454→                      {profile.bittensorStack.length > 0
   455→                        ? ` · ${profile.bittensorStack.join(", ")}`
   456→                        : ""}
   457→                      {profile.entrypoint ? ` · entrypoint ${profile.entrypoint}` : ""} ·{" "}
   458→                      {profile.repoUrl ? "repo cloned from GitHub" : "curated plan"}
   459→                    </p>
   460→                  </div>
   461→                  {seat && seat.verdict === "burn-entry" && (
   462→                    <div className="flex items-start gap-2 rounded-lg border border-warning/30 bg-warning/[0.04] p-3 text-xs">
   463→                      <Flame className="mt-0.5 h-4 w-4 shrink-0 text-warning" />
   464→                      <p className="text-muted-foreground">
   465→                        <span className="font-medium text-warning">Full subnet — burn entry.</span>{" "}
   466→                        {seat.headline.replace(`full (${seat.totalSlots}/${seat.totalSlots}) — `, "")}{" "}
   467→                        · immunity ~{seat.immunityHours}h for the new seat. Registration happens LAST
   468→                        (step 5) so the clock starts only when you are ready.
   469→                      </p>
   470→                    </div>
   471→                  )}
   472→                </div>
   473→              )}
   474→              <div className="flex justify-between">
   475→                <Button variant="outline" onClick={() => setStep(1)}>
   476→                  Back
   477→                </Button>
   478→                <Button onClick={() => setStep(3)} className="gap-1.5">
   479→                  Find a GPU
   480→                </Button>
   481→              </div>
   482→            </div>
   483→          )}
   484→
   485→          {/* ------------------------- STEP 3 ------------------------- */}
   486→          {effStep === 3 && liveSubnet && (
   487→            <div className="space-y-3">
   488→              <p className="text-xs text-muted-foreground">
   489→                Offers meeting the{" "}
   490→                <span className="font-medium text-foreground">{requiredVram ?? "?"}GB</span>{" "}
   491→                requirement, cheapest first.
   492→              </p>
   493→              {offer && !matchingOffers.some((o) => o.id === offer.id) && (
   494→                <p className="rounded-lg border border-warning/30 bg-warning/[0.04] p-2.5 text-xs text-warning">
   495→                  The preselected {offer.model} falls short of {requiredVram}GB for this subnet —
   496→                  pick one from the list below.
   497→                </p>
   498→              )}
   499→              <div className="max-h-56 space-y-1.5 overflow-y-auto custom-scroll pr-1">
   500→                {matchingOffers.map((o) => {
   501→                  const selected = offerId === o.id;
   502→                  return (
   503→                    <button
   504→                      key={o.id}
   505→                      type="button"
   506→                      onClick={() => setOfferId(o.id)}
   507→                      className={cn(
   508→                        "flex w-full items-center justify-between gap-2 rounded-lg border p-3 text-left transition-colors",
