     1→import { ApiPromise, WsProvider, HttpProvider } from "@polkadot/api";
     2→import { readFileSync, writeFileSync, existsSync } from "node:fs";
     3→import path from "node:path";

     5→/**
     6→ * Singleton Bittensor Finney chain client.
     7→ *
     8→ * Connects over WebSocket to the public entrypoint node (HTTP JSON-RPC
     9→ * fallback), and reuses the connection across API route invocations.
    10→ *
    11→ * Bittensor's chain metadata is huge (~700KB), so it is persisted to disk
    12→ * after the first successful connect and passed back to polkadot-js on
    13→ * subsequent boots — cutting cold start from ~10-60s down to ~1-2s.
    14→ */
    15→
    16→const WS_URL = "wss://entrypoint-finney.opentensor.ai:443";
    17→const RPC_URL = "https://entrypoint-finney.opentensor.ai/rpc";
    18→// Standalone server chdirs into .next/standalone — anchor caches on the
    19→// repo root (INFRANEX_REPO_ROOT pinned in the start script) so rebuilds
    20→// don't reset the chain metadata + alpha price trend windows.
    21→const REPO_ROOT = process.env.INFRANEX_REPO_ROOT ?? process.cwd();
    22→const METADATA_FILE = path.join(REPO_ROOT, ".chain-metadata.json");
    23→const PRICE_HISTORY_FILE = path.join(REPO_ROOT, ".alpha-price-history.json");
    24→
    25→// ---------------------------------------------------------------------------
    26→// Alpha price history — server-side ring buffer that powers the 24h alpha
    27→// price trend per subnet (the dTAO "rising vs dying subnet" signal).
    28→// Samples every 10 minutes, keeps 48h, persisted to disk so restarts don't
    29→// reset the trend window. Alpha price is tracked in TAO (the pool ratio) so
    30→// the trend reflects subnet-specific flow, not TAO's own USD moves.
    31→// ---------------------------------------------------------------------------
    32→
    33→const PRICE_SAMPLE_INTERVAL_MS = 10 * 60_000;
    34→const PRICE_HISTORY_TTL_MS = 48 * 3_600_000;
    35→
    36→interface PriceSample {
    37→  t: number; // epoch ms
    38→  p: Record<string, number>; // netuid → alpha price in TAO
    39→}
    40→
    41→let priceHistoryCache: PriceSample[] | null = null;
    42→
    43→function loadPriceHistory(): PriceSample[] {
    44→  if (priceHistoryCache) return priceHistoryCache;
    45→  try {
    46→    if (existsSync(PRICE_HISTORY_FILE)) {
    47→      const parsed = JSON.parse(readFileSync(PRICE_HISTORY_FILE, "utf8")) as unknown;
    48→      if (Array.isArray(parsed)) {
    49→        priceHistoryCache = parsed.filter(
    50→          (s): s is PriceSample =>
    51→            !!s && typeof (s as PriceSample).t === "number" && !!(s as PriceSample).p
    52→        );
    53→      }
    54→    }
    55→  } catch {
    56→    // corrupt file — start fresh
    57→  }
    58→  return priceHistoryCache ?? [];
    59→}
    60→
    61→function recordAlphaPrices(prices: Map<number, number>): void {
    62→  const hist = loadPriceHistory();
    63→  const now = Date.now();
    64→  const last = hist[hist.length - 1];
    65→  if (last && now - last.t < PRICE_SAMPLE_INTERVAL_MS) return;
    66→  const p: Record<string, number> = {};
    67→  for (const [netuid, price] of prices) {
    68→    if (Number.isFinite(price) && price > 0) p[String(netuid)] = price;
    69→  }
    70→  hist.push({ t: now, p });
    71→  const cutoff = now - PRICE_HISTORY_TTL_MS;
    72→  const pruned = hist.filter((s) => s.t >= cutoff).slice(-400);
    73→  priceHistoryCache = pruned;
    74→  try {
    75→    writeFileSync(PRICE_HISTORY_FILE, JSON.stringify(pruned));
    76→  } catch {
    77→    // best-effort persistence
    78→  }
    79→}
    80→
    81→/** 24h change in alpha/TAO price for a subnet, in % (1 decimal). */
    82→function alphaChange24h(netuid: number, current: number): number | null {
    83→  if (!Number.isFinite(current) || current <= 0) return null;
    84→  const hist = loadPriceHistory();
    85→  if (hist.length < 2) return null;
    86→  const cutoff = Date.now() - 24 * 3_600_000;
    87→  let ref: PriceSample | null = null;
    88→  for (const s of hist) {
    89→    if (s.t >= cutoff) {
    90→      ref = s;
    91→      break;
    92→    }
    93→  }
    94→  if (!ref) ref = hist[0];
    95→  const old = ref.p[String(netuid)];
    96→  if (!old || old <= 0) return null;
    97→  return Math.round(((current - old) / old) * 1000) / 10;
    98→}

   100→let _api: ApiPromise | null = null;
   101→let _connecting: Promise<ApiPromise> | null = null;
   102→
   103→function loadCachedMetadata(): Record<string, string> | null {
   104→  try {
   105→    if (existsSync(METADATA_FILE)) {
   106→      return JSON.parse(readFileSync(METADATA_FILE, "utf8")) as Record<string, string>;
   107→    }
   108→  } catch {
   109→    // corrupt file — ignore and fetch fresh
   110→  }
   111→  return null;
   112→}
   113→
   114→function saveMetadata(api: ApiPromise): void {
   115→  try {
   116→    const key = `${api.genesisHash.toHex()}-${api.runtimeVersion.specVersion}`;
   117→    const existing = loadCachedMetadata() ?? {};
   118→    if (existing[key]) return; // already cached for this runtime version
   119→    existing[key] = api.runtimeMetadata.toHex();
   120→    writeFileSync(METADATA_FILE, JSON.stringify(existing));
   121→  } catch {
   122→    // best-effort persistence
   123→  }
   124→}
   125→
   126→async function createApi(): Promise<ApiPromise> {
   127→  const metadata = loadCachedMetadata() as unknown as Record<string, `0x${string}`> | null;
   128→  const createOpts = { noInitWarn: true, throwOnConnect: true, metadata: metadata ?? undefined };
   129→
   130→  // WebSocket first — persistent connection, better for repeated reads.
   131→  try {
   132→    const provider = new WsProvider(WS_URL, 4000, undefined, 60_000);
   133→    const api = await ApiPromise.create({ provider, ...createOpts });
   134→    saveMetadata(api);
   135→    return api;
   136→  } catch {
   137→    // fall through to HTTP
   138→  }
   139→
   140→  // HTTP JSON-RPC fallback.
   141→  const provider = new HttpProvider(RPC_URL);
   142→  const api = await ApiPromise.create({ provider, ...createOpts });
   143→  saveMetadata(api);
   144→  return api;
   145→}
   146→
   147→export async function getChainApi(): Promise<ApiPromise> {
   148→  if (_api && _api.isConnected) return _api;
   149→  if (_connecting) return _connecting;
   150→  _connecting = (async () => {
   151→    // Disconnect any stale connection before creating a fresh one.
   152→    if (_api) {
   153→      try { await _api.disconnect(); } catch { /* ignore */ }
   154→      _api = null;
   155→    }
   156→    _api = await createApi();
   157→    return _api;
   158→  })();
   159→  try {
   160→    return await _connecting;
   161→  } finally {
   162→    _connecting = null;
   163→  }
   164→}

   248→function toTao(raw: unknown): number {
   249→  const s = String(raw ?? "0").replace(/[^0-9]/g, "");
   250→  const n = Number(s) / 10 ** TAO_DECIMALS;
   251→  return Number.isFinite(n) ? n : 0;

   256→  const n = Number(s);
   257→  return Number.isFinite(n) ? n : 0;

   260→async function fetchTaoPrice(): Promise<{
   261→  usd: number;
   262→  marketCap: number;
   263→  change24h: number;
   264→}> {
   265→  // Try CoinGecko first (most reliable), then fall back to alternative sources.
   266→  // CoinGecko rate-limits to ~10-30 req/min; if we get 429, we cache the last
   267→  // good price and try a fallback API.
   268→
   269→  // 1. CoinGecko
   270→  try {
   271→    const res = await fetch(
   272→      "https://api.coingecko.com/api/v3/simple/price?ids=bittensor&vs_currencies=usd&include_24hr_change=true&include_market_cap=true",
   273→      {
   274→        headers: {
   275→          "User-Agent": "infranex-bt/1.0",
   276→          Accept: "application/json",
   277→        },
   278→        cache: "no-store",
   279→        signal: AbortSignal.timeout(5000),
   280→      }
   281→    );
   282→    if (res.ok) {
   283→      const j = (await res.json()) as {
   284→        bittensor?: { usd?: number; usd_market_cap?: number; usd_24h_change?: number };
   285→      };
   286→      if (j.bittensor?.usd) {
   287→        return {
   288→          usd: j.bittensor.usd,
   289→          marketCap: j.bittensor.usd_market_cap ?? 0,
   290→          change24h: j.bittensor.usd_24h_change ?? 0,
   291→        };
   292→      }
   293→    }
   294→    // 429 or other error — fall through to fallback
   295→  } catch {
   296→    // timeout or network error — fall through to fallback
   297→  }
   298→
   299→  // 2. Fallback: Coinbase (no rate limit, different data path)
   300→  try {
   301→    const res = await fetch(
   302→      "https://api.coinbase.com/v2/prices/TAO-USD/spot",
   303→      {
   304→        headers: { Accept: "application/json" },
   305→        cache: "no-store",
   306→        signal: AbortSignal.timeout(5000),
   307→      }
   308→    );
   309→    if (res.ok) {
   310→      const j = (await res.json()) as {
   311→        data?: { amount?: string };
   312→      };
   313→      const usd = parseFloat(j.data?.amount ?? "0");
   314→      if (usd > 0) {
   315→        return { usd, marketCap: 0, change24h: 0 };
   316→      }
   317→    }
   318→  } catch {
   319→    // fall through
   320→  }
   321→
   322→  // 3. Last resort: cached price (last known good value)
   323→  if (lastKnownTaoPrice > 0) {
   324→    return { usd: lastKnownTaoPrice, marketCap: 0, change24h: 0 };
   325→  }
   326→
   327→  return { usd: 0, marketCap: 0, change24h: 0 };
   328→}

   330→// Cache the last known good TAO price as a fallback when APIs are rate-limited.
   331→let lastKnownTaoPrice = 0;

   333→/**
   334→ * Fetch neurons (individual miners) for a subnet via the metagraph.
   335→ * Returns per-miner incentive, trust, rank, emission, stake, etc.
   336→ * Limited to `limit` neurons to avoid heavy queries.
   337→ */
   338→async function fetchNeuronsForSubnet(
   339→  api: Awaited<ReturnType<typeof getChainApi>>,
   340→  netuid: number,
   341→  limit: number
   342→): Promise<NeuronMetrics[]> {
   343→  const neurons: NeuronMetrics[] = [];
   344→  try {
   345→    const mods = api.query.subtensorModule as unknown as Record<
   346→      string,
   347→      (n: number) => Promise<{ toString(): string; isEmpty: boolean; toJSON?: () => unknown }>
   348→    >;
   349→    // Per-neuron data lives in per-SUBNET vector maps: each storage entry is
   350→    // keyed by netuid and returns a Vec indexed by uid (length = max uids).
   351→    // 7 single RPC calls per subnet instead of 7 × N per-neuron round-trips.
   352→    const vec = async (
   353→      map: ((n: number) => Promise<unknown>) | undefined,
   354→      fallback: unknown[],
   355→    ): Promise<unknown[]> => {
   356→      if (!map) return fallback;
   357→      try {
   358→        const res = (await map(netuid)) as { toJSON?: () => unknown };
   359→        return (res.toJSON?.() ?? fallback) as unknown[];
   360→      } catch {
   361→        return fallback;
   362→      }
   363→    };
   364→
   365→    const [
   366→      actives,
   367→      incentives,
   368→      consensuses,
   369→      dividendsArr,
   370→      emissions,
   371→      lastUpdates,
   372→      validatorTrusts,
   373→    ] = await Promise.all([
   374→      vec(mods.active, []),
   375→      vec(mods.incentive, []),
   376→      vec(mods.consensus, []),
   377→      vec(mods.dividends, []),
   378→      vec(mods.emission, []),
   379→      vec(mods.lastUpdate, []),
   380→      vec(mods.validatorTrust, []),
   381→    ]);
   382→
   383→    if (actives.length === 0) return neurons;
   384→    const count = Math.min(actives.length, limit, 256);
   385→
   386→    for (let i = 0; i < count; i++) {
   387→      const isActive = actives[i] === true;
   388→      const incentive = Number(incentives[i] ?? 0) || 0;
   389→      const emission = Number(emissions[i] ?? 0) || 0;
   390→      // Skip dormant uids that have never been active and hold no value.
   391→      if (!isActive && incentive === 0 && emission === 0) continue;
   392→      neurons.push({
   393→        uid: i,
   394→        netuid,
   395→        hotkey: "",
   396→        coldkey: null,
   397→        stake: 0,
   398→        rank: 0,
   399→        emission,
   400→        incentive,
   401→        trust: 0,
   402→        consensus: Number(consensuses[i] ?? 0) || 0,
   403→        validatorTrust: Number(validatorTrusts[i] ?? 0) || 0,
   404→        dividends: Number(dividendsArr[i] ?? 0) || 0,
   405→        lastUpdate: Number(lastUpdates[i] ?? 0) || 0,
   406→        isActive,
   407→      } as NeuronMetrics & { netuid: number });
   408→    }
   409→  } catch {
   410→    // best-effort
   411→  }
   412→  return neurons;
   413→}

   415→export async function fetchLiveSnapshot(): Promise<LiveNetworkSnapshot> {
   416→  return snapshotCache.get();
   417→}
   418→
   419→// ---------------------------------------------------------------------------
   420→// In-memory snapshot cache with stale-while-revalidate. The chain read batches
   421→// all storage queries via .multi() (~40 RPC calls instead of ~2000), so a cold
   422→// refresh takes ~2-5s. Results are cached for 60s; if the cache is expired but
   423→// a stale snapshot exists, it is served IMMEDIATELY while a background refresh
   424→// runs — so /api/network never hangs waiting on the chain.
   425→// ---------------------------------------------------------------------------
   426→
   427→const CACHE_TTL_MS = 60_000;
   428→// Cold fetch now includes per-neuron emission vectors + identity registry
   429→// (~15 batched RPC groups, ~20s measured) — give it headroom.
   430→const CHAIN_FETCH_TIMEOUT_MS = 45_000;
   431→
   432→class SnapshotCache {
   433→  private cached: LiveNetworkSnapshot | null = null;
   434→  private expiresAt = 0;
   435→  private refreshing = false;
   436→  private refreshPromise: Promise<LiveNetworkSnapshot> | null = null;
   437→
   438→  async get(): Promise<LiveNetworkSnapshot> {
   439→    const now = Date.now();
   440→    if (this.cached && now < this.expiresAt) {
   441→      return this.cached;
   442→    }
   443→    // Stale-but-present: return it now, refresh in the background.
   444→    if (this.cached) {
   445→      if (!this.refreshing) {
   446→        this.refreshing = true;
   447→        this.refreshPromise = this.refresh().finally(() => {
   448→          this.refreshing = false;
   449→          this.refreshPromise = null;
   450→        });
   451→      }
   452→      return this.cached;
   453→    }
   454→    // No snapshot at all — single-flight the first fetch so concurrent
   455→    // requests share one chain read instead of each spawning their own.
   456→    if (!this.refreshPromise) {
   457→      this.refreshing = true;
   458→      this.refreshPromise = this.refresh().finally(() => {
   459→        this.refreshing = false;
   460→        this.refreshPromise = null;
   461→      });
   462→    }
   463→    return this.refreshPromise;
   464→  }
   465→
   466→  private async refresh(): Promise<LiveNetworkSnapshot> {
   467→    try {
   468→      const snap = await Promise.race([
   469→        this.fetchFromChain(),
   470→        new Promise<LiveNetworkSnapshot>((_, reject) =>
   471→          setTimeout(() => reject(new Error("chain fetch timed out")), CHAIN_FETCH_TIMEOUT_MS)
   472→        ),
   473→      ]);
   474→      this.cached = snap;
   475→      this.expiresAt = Date.now() + CACHE_TTL_MS;
   476→      return snap;
   477→    } catch (e) {
   478→      // Timed out or failed: keep serving any existing stale snapshot and
   479→      // retry on the next poll instead of caching an error result.
   480→      if (this.cached) return this.cached;
   481→      return {
   482→        blockNumber: 0,
   483→        totalSubnets: 0,
   484→        specVersion: 0,
   485→        fetchedAt: new Date().toISOString(),
   486→        taoPriceUsd: lastKnownTaoPrice,
   487→        taoMarketCapUsd: 0,
   488→        taoChange24h: 0,
   489→        subnets: [],
   490→        neurons: [],
   491→        source: "error",
   492→        error: e instanceof Error ? e.message : String(e),
   493→      };
   494→    }
   495→  }
   496→
   497→  private async fetchFromChain(): Promise<LiveNetworkSnapshot> {
   498→    const price = await fetchTaoPrice();
   499→    if (price.usd > 0) {
   500→      lastKnownTaoPrice = price.usd;
   501→    }
   502→    try {
   503→      const api = await getChainApi();
   504→      const [header, totalNetworksRaw] = await Promise.all([
   505→        api.rpc.chain.getHeader(),
   506→        api.query.subtensorModule.totalNetworks(),
   507→      ]);
   508→      const totalSubnets = Number(totalNetworksRaw.toString()) || 0;
   509→      const mods = api.query.subtensorModule as unknown as Record<
   510→        string,
   511→        (n: number) => Promise<{ toString(): string; isEmpty: boolean }>
   512→      >;
   513→
   514→      // Scan ALL subnets on chain (not just the 16 tracked ones).
   515→      const allNetuids = Array.from({ length: Math.min(totalSubnets, 200) }, (_, i) => i);
   516→      const subnets: LiveSubnetMetrics[] = [];
   517→      const alphaPriceByNetuid = new Map<number, number>();
   518→
   519→      // Batch queries: 7 .multi() RPC calls per batch of netuids instead of
   520→      // 7 individual round-trips per subnet. Batches of 50 keep each request
   521→      // payload reasonable while cutting total RPC calls from ~900 to ~20.
   522→      const batchOf = 50;
   523→      for (let i = 0; i < allNetuids.length; i += batchOf) {
   524→        const batch = allNetuids.slice(i, i + batchOf);
   525→        const multi = async (
   526→          map: unknown,
   527→        ): Promise<unknown[]> => {
   528→          const m = map as { multi?: (keys: unknown[]) => Promise<unknown[]> } | undefined;
   529→          if (!m?.multi) return batch.map(() => null);
   530→          try {
   531→            return (await m.multi(batch)) as unknown[];
   532→          } catch {
   533→            return batch.map(() => null);
   534→          }
   535→        };
   536→
   537→        const [
   538→          minersArr,
   539→          taoArr,
   540→          alphaInArr,
   541→          alphaOutArr,
   542→          tempoArr,
   543→          emEnabledArr,
   544→          priceArr,
   545→        ] = await Promise.all([
   546→          multi(mods.subnetworkN as never),
   547→          multi(mods.subnetTAO as never),
   548→          multi(mods.subnetAlphaIn as never),
   549→          multi(mods.subnetAlphaOut as never),
   550→          multi(mods.tempo as never),
   551→          multi(mods.subnetEmissionEnabled as never),
   552→          multi(mods.subnetMovingPrice as never),
   553→        ]);
   554→
   555→        // Real per-neuron emission data + subnet registry info for this
   556→        // batch — 7 more batched .multi() calls instead of per-subnet
   557→        // round-trips. The Emission vec holds per-uid alpha totals (rao)
   558→        // for the last epoch; incentive/dividends split miners vs
   559→        // validators. (The Active vec only flags validator weight
   560→        // updates — verified on chain — so it is intentionally skipped.)
   561→        const [
   562→          emissionVecs,
   563→          incentiveVecs,
   564→          dividendsVecs,
   565→          identitiesArr,
   566→          ownerArr,
   567→          registeredAtArr,
   568→          maxUidsArr,
   569→          burnArr,
   570→          immunityArr,
   571→        ] = await Promise.all([
   572→          multi(mods.emission as never),
   573→          multi(mods.incentive as never),
   574→          multi(mods.dividends as never),
   575→          multi(mods.subnetIdentitiesV3 as never),
   576→          multi(mods.subnetOwner as never),
   577→          multi(mods.networkRegisteredAt as never),
   578→          multi(mods.maxAllowedUids as never),
   579→          multi(mods.burn as never),
   580→          multi(mods.immunityPeriod as never),
   581→        ]);
   582→
   583→        const numVec = (raw: unknown): number[] => {
   584→          const v = (raw as { toJSON?: () => unknown } | null)?.toJSON?.();
   585→          return Array.isArray(v) ? v.map((x) => Number(x) || 0) : [];
   586→        };
   587→
   588→        for (let j = 0; j < batch.length; j++) {
   589→          const n = batch[j];
   590→          try {
   591→            const miners = minersArr[j] as { toString(): string } | null;
   592→            const minerCount = Number(miners?.toString() ?? "0") || 0;
   593→            const subnetTao = toTao((taoArr[j] as { toString(): string } | null)?.toString());
   594→            if (minerCount === 0 && subnetTao === 0) continue;
   595→            const emEnabled = emEnabledArr[j] as { isEmpty: boolean } | null;
   596→
   597→            // --- Real emission accounting from per-neuron vectors ---
   598→            const emVec = numVec(emissionVecs[j]);
   599→            const incVec = numVec(incentiveVecs[j]);
   600→            const divVec = numVec(dividendsVecs[j]);
   601→            const tempoSafe = Math.max(
   602→              Number((tempoArr[j] as { toString(): string } | null)?.toString() ?? "0") || 0,
   603→              1
   604→            );
   605→            let epochAlphaRao = 0;
   606→            let minerAlphaRao = 0;
   607→            let rewardedMiners = 0;
   608→            let validators = 0;
   609→            const uidCount = Math.max(emVec.length, incVec.length);
   610→            for (let u = 0; u < uidCount; u++) {
   611→              const e = emVec[u] ?? 0;
   612→              const inc = incVec[u] ?? 0;
   613→              const div = divVec[u] ?? 0;
   614→              if (e > 0) epochAlphaRao += e;
   615→              if (inc > 0) { rewardedMiners++; minerAlphaRao += Math.max(e, 0); }
   616→              if (div > 0) validators++;
   617→            }
   618→            // Reward concentration: share of last-epoch incentive captured by
   619→            // the top 10% of UIDs. A wide spread (low share) means a decent
   620→            // miner holds its seat safely; a narrow spread (share near 1) is
   621→            // the taostats "knife fight" signal — any disruption → deregistered.
   622→            let top10IncentiveShare: number | null = null;
   623→            const incTotal = incVec.reduce((a, b) => a + b, 0);
   624→            if (incTotal > 0) {
   625→              const sorted = [...incVec].sort((a, b) => b - a);
   626→              const topN = Math.max(1, Math.ceil(sorted.length * 0.1));
   627→              top10IncentiveShare =
   628→                Math.round(
   629→                  (sorted.slice(0, topN).reduce((a, b) => a + b, 0) / incTotal) * 1000
   630→                ) / 1000;
   631→            }
   632→            // Median-vs-mean share across EARNING uids — the taostats
   633→            // "emission spread" signal. When a whale UID takes the bulk,
   634→            // the per-earning MEAN wildly overstates what a mid-pack miner
   635→            // actually receives; the ledger discounts revenue by this.
   636→            let incentiveMedianShare: number | null = null;
   637→            if (incTotal > 0 && rewardedMiners > 0) {
   638→              const nonzero = incVec.filter((v) => v > 0).sort((a, b) => a - b);
   639→              if (nonzero.length > 0) {
   640→                const mid = nonzero[Math.floor(nonzero.length / 2)];
   641→                const mean = incTotal / rewardedMiners;
   642→                if (mean > 0) {
   643→                  incentiveMedianShare =
   644→                    Math.round(Math.min(mid / mean, 1.2) * 1000) / 1000;
   645→                }
   646→              }
   647→            }
   648→            // Per-uid values are last-epoch alpha totals → per-block = /tempo.
   649→            const alphaPerBlock = epochAlphaRao / tempoSafe / 1e9; // α/block
   650→            const movingPrice = toTao((priceArr[j] as { toString(): string } | null)?.toString());
   651→            const alphaInTao = toTao((alphaInArr[j] as { toString(): string } | null)?.toString());
   652→            // Alpha price: moving price when present, else pool ratio fallback.
   653→            const priceTao =
   654→              movingPrice > 0
   655→                ? movingPrice
   656→                : alphaInTao > 0 && subnetTao > 0
   657→                  ? subnetTao / alphaInTao
   658→                  : 0;
   659→            // TAO-value emission for the whole subnet.
   660→            const emissionPerBlockTao = alphaPerBlock * priceTao;
   661→            const emissionPerDayTao = emissionPerBlockTao * 720;
   662→            // Miner share of emission (miners vs validators split); 50/50 fallback.
   663→            const validatorAlphaRao = Math.max(epochAlphaRao - minerAlphaRao, 0);
   664→            const minerShare =
   665→              minerAlphaRao + validatorAlphaRao > 0
   666→                ? minerAlphaRao / (minerAlphaRao + validatorAlphaRao)
   667→                : 0.5;
   668→
   669→            // --- On-chain identity / owner / registration age ---
   670→            const identityOpt = identitiesArr[j] as {
   671→              isEmpty: boolean;
   672→              unwrap?: () => {
   673→                subnetName?: unknown;
   674→                githubRepo?: unknown;
   675→                description?: unknown;
   676→              };
   677→            } | null;
   678→            // Identity fields are Vec<u8> — decode via toUtf8, hex fallback.
   679→            const idText = (v: unknown): string => {
   680→              if (v == null) return "";
   681→              const codec = v as { toUtf8?: () => string };
   682→              if (typeof codec.toUtf8 === "function") {
   683→                try {
   684→                  const s = codec.toUtf8().trim();
   685→                  if (s && !s.startsWith("0x")) return s;
   686→                } catch { /* fall through */ }
   687→              }
   688→              const s = typeof v === "string" ? v : String(v);
   689→              if (s.startsWith("0x") && s.length > 2) {
   690→                try {
   691→                  const hex = s.slice(2);
   692→                  const bytes = new Uint8Array(hex.length / 2);
   693→                  for (let i = 0; i < bytes.length; i++) {
   694→                    bytes[i] = parseInt(hex.substr(i * 2, 2), 16);
   695→                  }
   696→                  return new TextDecoder().decode(bytes).trim();
   697→                } catch { return ""; }
   698→              }
   699→              return s.trim();
   700→            };
   701→            let idName = "";
   702→            let idGithub = "";
   703→            let idDesc = "";
   704→            if (identityOpt && !identityOpt.isEmpty && identityOpt.unwrap) {
   705→              try {
   706→                const v = identityOpt.unwrap();
   707→                idName = idText(v.subnetName);
   708→                idGithub = idText(v.githubRepo);
   709→                idDesc = idText(v.description);
   710→              } catch {
   711→                // identity unreadable — leave empty
   712→              }
   713→            }
   714→            const ownerRaw = ownerArr[j] as
   715→              | { toString(): string; isEmpty?: boolean }
   716→              | null;
   717→            const regBlock = Number(
   718→              (registeredAtArr[j] as { toString(): string } | null)?.toString() ?? "0"
   719→            ) || 0;
   720→            const maxUids = Number(
   721→              (maxUidsArr[j] as { toString(): string } | null)?.toString() ?? "0"
   722→            ) || 0;
   723→            const burnCostTao = toTao(
   724→              (burnArr[j] as { toString(): string } | null)?.toString()
   725→            );
   726→            const immunityBlocks = Number(
   727→              (immunityArr[j] as { toString(): string } | null)?.toString() ?? "0"
   728→            ) || 0;
   729→
   730→            subnets.push({
   731→              netuid: n,
   732→              name: idName || null,
   733→              minersCount: minerCount,
   734→              validatorsCount: validators || null,
   735→              subnetTao,
   736→              alphaIn: alphaInTao,
   737→              alphaOut: toTao((alphaOutArr[j] as { toString(): string } | null)?.toString()),
   738→              tempo: tempoSafe,
   739→              emissionEnabled: !emEnabled?.isEmpty,
   740→              movingPrice,
   741→              emission: emissionPerBlockTao > 0 ? emissionPerBlockTao : null,
   742→              emissionTaoPerDay: emissionPerDayTao > 0 ? emissionPerDayTao : null,
   743→              minerEmissionTaoPerDay:
   744→                emissionPerDayTao > 0 ? emissionPerDayTao * minerShare : null,
   745→              rewardedMiners,
   746→              top10IncentiveShare,
   747→              incentiveMedianShare,
   748→              burnCostTao: burnCostTao > 0 ? burnCostTao : null,
   749→              immunityBlocks: immunityBlocks > 0 ? immunityBlocks : null,
   750→              alphaPriceChange24h: alphaChange24h(n, priceTao),
   751→              maxUids: maxUids || null,
   752→              owner: ownerRaw && !ownerRaw.isEmpty ? ownerRaw.toString() : null,
   753→              registeredAt: regBlock || null,
   754→              identityGithub: idGithub || null,
   755→              identityDescription: idDesc || null,
   756→            });
   757→
   758→            // Collect this sample window's alpha prices for the ring buffer.
   759→            if (priceTao > 0) alphaPriceByNetuid.set(n, priceTao);
   760→          } catch {
   761→            // skip malformed entry
   762→          }
   763→        }
   764→      }
   765→
   766→      // Fetch neurons (metagraph data) for top 3 subnets by miner count.
   767→      const topSubnets = [...subnets].sort((a, b) => b.minersCount - a.minersCount).slice(0, 3);
   768→      const neurons: NeuronMetrics[] = [];
   769→      for (const sub of topSubnets) {
   770→        try {
   771→          const subNeurons = await fetchNeuronsForSubnet(api, sub.netuid, Math.min(sub.minersCount, 50));
   772→          neurons.push(...subNeurons);
   773→        } catch {
   774→          // skip
   775→        }
   776→      }
   777→
   778→      // One runtime-API call for all 129 DynamicInfo records — gives the
   779→      // REAL TAO pool per subnet (taoIn). The SubnetTAO storage value is
   780→      // not the full staked pool, and requiredStake/APY math needs the
   781→      // true figure.
   782→      try {
   783→        const dyn = (await api.call.subnetInfoRuntimeApi.getAllDynamicInfo()) as unknown as Array<{
   784→          netuid: { toString(): string };
   785→          taoIn: { toString(): string };
   786→        }>;
   787→        const taoInByNetuid = new Map<number, number>();
   788→        for (const d of dyn) {
   789→          try {
   790→            const netuid = Number(d.netuid.toString());
   791→            const taoIn = toTao(d.taoIn.toString());
   792→            if (Number.isFinite(netuid) && taoIn > 0) taoInByNetuid.set(netuid, taoIn);
   793→          } catch {
   794→            // skip malformed entry
   795→          }
   796→        }
   797→        for (const sub of subnets) {
   798→          const taoIn = taoInByNetuid.get(sub.netuid);
   799→          if (taoIn && taoIn > sub.subnetTao) sub.subnetTao = taoIn;
   800→        }
   801→      } catch {
   802→        // DynamicInfo unavailable — keep storage values
   803→      }
   804→
   805→      // Persist this refresh's alpha prices into the 48h ring buffer (no-op
   806→      // inside the 10-min sample interval) so the 24h trend survives restarts.
   807→      recordAlphaPrices(alphaPriceByNetuid);
   808→
   809→      // Keep the singleton connection alive — recycling it would force the
   810→      // next refresh to re-fetch chain metadata (~3s cold start).
   811→
   812→      return {
   813→        blockNumber: header.number.toNumber(),
   814→        totalSubnets,
   815→        specVersion: api.runtimeVersion.specVersion.toNumber(),
   816→        fetchedAt: new Date().toISOString(),
   817→        taoPriceUsd: price.usd,
   818→        taoMarketCapUsd: price.marketCap,
   819→        taoChange24h: price.change24h,
   820→        subnets,
   821→        neurons,
   822→        source: subnets.length > 0 ? "live" : "partial",
   823→      };
   824→    } catch (e) {
   825→      return {
   826→        blockNumber: 0,
   827→        totalSubnets: 0,
   828→        specVersion: 0,
   829→        fetchedAt: new Date().toISOString(),
   830→        taoPriceUsd: price.usd,
   831→        taoMarketCapUsd: price.marketCap,
   832→        taoChange24h: price.change24h,
   833→        subnets: [],
   834→        neurons: [],
   835→        source: "error",
   836→        error: e instanceof Error ? e.message : String(e),
   837→      };
   838→    }
   839→  }
   840→}
