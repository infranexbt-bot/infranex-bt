   191→   191→function DeploymentCard({
   192→   192→  deployment: d,
   193→   193→  onSelect,
   194→   194→  isSelected,
   195→   195→  onOpenWizard,
   196→   196→}: {
   197→   197→  deployment: DeploymentRecord;
   198→   198→  onSelect: () => void;
   199→   199→  isSelected: boolean;
   200→   200→  onOpenWizard: (ctx: RegistrationWizardContext) => void;
   201→   201→}) {
   202→   202→  const tickMut = useTickDeployment();
   203→   203→  const termMut = useTerminateDeployment();
   204→   204→  const delMut = useDeleteDeployment();
   205→   205→  const regAction = useRegistrationAction();
   206→   206→  const qc = useQueryClient();
   207→   207→  const [devopsNote, setDevopsNote] = useState<string | null>(null);
   208→   208→  const [devopsBusy, setDevopsBusy] = useState(false);
   209→   209→  const [wizardBusy, setWizardBusy] = useState(false);
   210→   210→  const [revOpen, setRevOpen] = useState(false);
   211→   211→  const [migOpen, setMigOpen] = useState(false);
   212→   212→  const [daemonOpen, setDaemonOpen] = useState(false);
   213→   213→
   214→   214→  const isStarted = d.status === "started";
   215→   215→  const hasHotkey = SS58_RE.test(d.hotkey ?? "");
   216→   216→
   217→   217→  // Phase 2 background watcher: for a STARTED deployment with a plausible
   218→   218→  // hotkey, poll the registration check every 45 s (server-throttled).
   219→   219→  const regWatch = useRegistrationWatcher(d.id, isStarted && hasHotkey);
   220→   220→  useEffect(() => {
   221→   221→    if (regWatch.data?.changed) {
   222→   222→      qc.invalidateQueries({ queryKey: ["deployments"] });
   223→   223→    }
   224→   224→  }, [regWatch.data?.changed, regWatch.data?.checkedAt, qc]);
   225→   225→
   226→   226→  // Open the wizard pre-bound to THIS deployment (fetch its hand-off context).
   227→   227→  const openWizard = async () => {
   228→   228→    setWizardBusy(true);
   229→   229→    try {
   230→   230→      const res = await fetch(`/api/deployments/${d.id}/registration`, { cache: "no-store" });
   231→   231→      const j = await res.json();
   232→   232→      if (res.ok && j.wizard) onOpenWizard(j.wizard as RegistrationWizardContext);
   233→   233→      else setDevopsNote(j?.error ?? "Could not load the wizard context");
   234→   234→    } catch {
   235→   235→      setDevopsNote("Could not reach the registration API");
   236→   236→    } finally {
   237→   237→      setWizardBusy(false);
   238→   238→    }
   239→   239→  };
   240→   240→
   241→   241→  const restartAfterReg = () => regAction.mutate({ id: d.id, action: "restart" });
   242→   242→
   243→   243→  const registerToDevOps = async () => {
   244→   244→    setDevopsBusy(true);
   245→   245→    setDevopsNote(null);
   246→   246→    try {
   247→   247→      const res = await fetch(`/api/deployments/${d.id}/to-devops`, { method: "POST" });
   248→   248→      const j = await res.json();
   249→   249→      if (!res.ok) throw new Error(j?.error ?? `HTTP ${res.status}`);
   250→   250→      setDevopsNote(`Registered as host ${String(j.hostName ?? j.hostId)} — run validation in DevOps.`);
   251→   251→    } catch (e) {
   252→   252→      setDevopsNote(e instanceof Error ? e.message : "Register failed");
   253→   253→    } finally {
   254→   254→      setDevopsBusy(false);
   255→   255→    }
   256→   256→  };
   257→   257→
   258→   258→  const isTerminal = d.status === "terminated" || d.status === "failed";
   259→   259→  const isInProgress = !isTerminal && d.status !== "started";
   260→   260→
   261→   261→  const statusColor =
   262→   262→    d.status === "started"
   263→   263→      ? "bg-success/10 text-success"
   264→   264→      : isTerminal
   265→   265→        ? "bg-muted/50 text-muted-foreground"
   266→   266→        : "bg-primary/10 text-primary";
   267→   267→
   268→   268→  return (
   269→   269→    <Card
   270→   270→      className={cn(
   271→   271→        "border-border/60 bg-card/40 backdrop-blur-sm transition-all",
   272→   272→        isSelected && "border-primary/40",
   273→   273→        !isTerminal && "hover:border-primary/30"
   274→   274→      )}
   275→   275→    >
   276→   276→      <CardContent className="p-4">
   277→   277→        <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
   278→   278→          <div className="flex min-w-0 items-start gap-3">
   279→   279→            <div className="flex flex-col items-center gap-1 pt-1">
   280→   280→              <span className={cn("pulse-dot", isStarted ? "text-success" : isInProgress ? "text-primary" : "text-muted-foreground")} />
   281→   281→            </div>
   282→   282→            <div className="min-w-0">
   283→   283→              <div className="flex flex-wrap items-center gap-2">
   284→   284→                <p className="font-medium">{d.minerName}</p>
   285→   285→                <Badge variant="outline" className="mono text-[10px]">α{d.netuid}</Badge>
   286→   286→                <Badge variant="outline" className="text-[10px]">{d.subnetName}</Badge>
   287→   287→                <span className={cn("badge-status capitalize", statusColor)}>{d.status}</span>
   288→   288→                <Badge variant="outline" className={cn("text-[9px]", d.mode === "runpod" ? "border-primary/30 text-primary" : d.mode === "vast" ? "border-warning/40 text-warning" : "text-muted-foreground")}>
   289→   289→                  {d.mode}
   290→   290→                </Badge>
   291→   291→                {/* TIER4 — creator attribution */}
   292→   292→                {d.createdByLabel && (
   293→   293→                  <Badge variant="outline" className="text-[9px] text-muted-foreground">
   294→   294→                    by {d.createdByLabel}
   295→   295→                  </Badge>
   296→   296→                )}
   297→   297→                {/* Phase 2: registration lifecycle chip */}
   298→   298→                {isStarted && d.registrationState === "registered" && (
   299→   299→                  <Badge variant="outline" className="gap-1 border-success/40 text-[10px] text-success">
   300→   300→                    <ShieldCheck className="h-3 w-3" />
   301→   301→                    UID {d.registeredUid ?? "?"}
   302→   302→                  </Badge>
   303→   303→                )}
   304→   304→                {isStarted && d.registrationState === "unregistered" && (
   305→   305→                  <Badge variant="outline" className="gap-1 border-amber-500/40 text-[10px] text-amber-600 dark:text-amber-400">
   306→   306→                    <KeyRound className="h-3 w-3" />
   307→   307→                    unregistered
   308→   308→                  </Badge>
   309→   309→                )}
   310→   310→                {isStarted && d.registrationState === null && (
   311→   311→                  <Badge variant="outline" className={cn("gap-1 text-[10px]", hasHotkey ? "border-amber-500/40 text-amber-600 dark:text-amber-400" : "text-muted-foreground")}>
   312→   312→                    <KeyRound className="h-3 w-3" />
   313→   313→                    {hasHotkey ? "registration unverified" : "no hotkey"}
   314→   314→                  </Badge>
   315→   315→                )}
   316→   316→              </div>
   317→   317→              <p className="mt-1 text-xs text-muted-foreground">
   318→   318→                {d.gpuModel} · {d.provider}
   319→   319→                {d.providerPodId && <span className="mono"> · pod {d.providerPodId.slice(0, 12)}</span>}
   320→   320→                {" · "}created {formatRelativeTime(new Date(d.createdAt))}
   321→   321→              </p>
   322→   322→            </div>
   323→   323→          </div>
   324→   324→
   325→   325→          <div className="flex items-center gap-4">
   326→   326→            <div className="text-right">
   327→   327→              <p className="text-[10px] text-muted-foreground">Cost/mo</p>
   328→   328→              <p className="tabular text-sm font-medium">{formatCurrency(d.monthlyCost)}</p>
   329→   329→            </div>
   330→   330→            <div className="text-right">
   331→   331→              <p className="text-[10px] text-muted-foreground">Rev/mo</p>
   332→   332→              <p className="tabular text-sm font-medium text-success">{formatCurrency(d.estimatedRevenue)}</p>
   333→   333→            </div>
   334→   334→          </div>
   335→   335→        </div>
   336→   336→
   337→   337→        {/* Progress bar */}
   338→   338→        <div className="mt-3">
   339→   339→          <div className="flex items-center justify-between text-[10px] text-muted-foreground">
   340→   340→            <span>Progress</span>

   342→   342→          </div>
   343→   343→          <Progress
   344→   344→            value={d.progress}
   345→   345→            className="mt-1 h-1"
   346→   346→            indicatorClassName={cn(
   347→   347→              isStarted ? "bg-success" : isTerminal ? "bg-muted-foreground" : "bg-primary"
   348→   348→            )}
   349→   349→          />
   350→   350→        </div>

   352→   352→        {/* Step indicators */}
   353→   353→        <div className="mt-3 flex flex-wrap items-center gap-x-3 gap-y-1">
   354→   354→          {d.steps.map((s) => {
   355→   355→            const Icon =
   356→   356→              s.status === "done" ? CheckCircle2
   357→   357→              : s.status === "running" ? Loader2
   358→   358→              : s.status === "failed" ? XCircle

   360→   360→            return (

   372→   372→                  {s.label}
   373→   373→                </span>

   377→   377→        </div>
   378→   378→
   379→   379→        {/* Actions */}

   401→   401→              size="sm"
   402→   402→              className="gap-1.5"
   403→   403→              onClick={() => setDaemonOpen(true)}
   404→   404→            >

   407→   407→            </Button>
   408→   408→          )}
   409→   409→          {/* TIER3 — migrate to a different GPU offer (spec §20) */}

   412→   412→              variant="outline"
   413→   413→              size="sm"
   414→   414→              className="gap-1.5"

   418→   418→              Migrate
   419→   419→            </Button>
   420→   420→          )}

   437→   437→            </Button>

   439→   439→          {isStarted && (
