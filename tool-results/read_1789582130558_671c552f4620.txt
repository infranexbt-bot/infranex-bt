     2→// Miner's Ledger scoring v2 — models how an experienced GPU miner actually
     3→// picks a subnet. Research-backed (Bittensor docs, taostats miner guide,
     4→// miner-economics deep dives, dTAO analyst frameworks):
     5→//
     6→//   Q1 FIT        — can my hardware run this? (subnet type → GPU tier/cost)
     7→//   Q2 TOP LINE   — what does it pay? (emission × 41% miner share ÷ EARNING
     8→//                   miners, priced in alpha → TAO → USD)
     9→//   Q3 BOTTOM LINE— what do I keep? (minus GPU rental + infra → net monthly)
    10→//   Q4 SEAT SAFETY— can I keep the slot? (saturation, reward concentration,
    11→//                   burn cost, immunity runway)
    12→//   Q5 HOLD VALUE — will revenue persist? (alpha price trend, pool liquidity,
    13→//                   sell-pressure slippage, owner identity, maturity)
    14→//
    15→// Key constants from the chain: per-tempo split is 18% owner / 41% miners /
    16→// 41% validators+stakers. "Top 10-20% of miners take 60-80% of emissions;
    17→// median miners hover near breakeven" — so the ledger scores NET numbers,
    18→// not gross, and treats reward concentration as a first-class risk.

    35→export interface ScoreComponents {

    48→export const SCORE_WEIGHTS: Record<keyof ScoreComponents, number> = {
    49→  net_roi: 0.3,
    50→  seat_safety: 0.2,
    51→  alpha_economics: 0.2,
    52→  earning_reality: 0.15,
    53→  fit_feasibility: 0.15,
    54→};

    77→export function deriveFactors(c: ScoreComponents): OpportunityFactor[] {
    78→  return (Object.keys(c) as (keyof ScoreComponents)[]).map((k) => {
    79→    const raw = c[k];
    80→    const weight = SCORE_WEIGHTS[k];
    81→    return {
    82→      name: FACTOR_LABELS[k],
    83→      value: Math.round(raw * weight * 100) / 100,
    84→      weight,
    85→      raw,
    86→      impact: raw >= 60 ? "positive" : raw <= 40 ? "negative" : "neutral",
    87→      description: FACTOR_DESC[k],
    88→    };
    89→  });
    90→}
    91→
    92→export function totalScore(c: ScoreComponents): number {
    93→  return Math.round(
    94→    (Object.keys(c) as (keyof ScoreComponents)[]).reduce(
    95→      (acc, k) => acc + c[k] * SCORE_WEIGHTS[k],
    96→      0
    97→    ) * 10
    98→  ) / 10;
    99→}

   101→export function riskLevel(score: number): "low" | "medium" | "high" {
   102→  if (score >= 60) return "low";
   103→  if (score >= 40) return "medium";
   104→  return "high";
   105→}

   145→/**
   146→ * Resolve a GPU tier from a model string found in a repo README
   147→ * (github-scraper parseGpuRequirement output). Ground truth from the
   148→ * subnet's own docs beats keyword classification.
   149→ */
   150→export function resolveGpuTierFromModel(model: string): GpuTier | null {
   151→  const m = model.toLowerCase();
   152→  if (/\bb300\b/.test(m)) return GPU_TIERS.b200; // same Blackwell HGX class
   153→  if (/\bb200\b/.test(m)) return GPU_TIERS.b200;
   154→  if (/\bh200\b/.test(m)) return GPU_TIERS.h200;
   155→  if (/\bh100\b/.test(m)) return GPU_TIERS.h100;
   156→  if (/\b(?:rtx\s*)?pro\s*6000\b/.test(m) || /\brtx\s*6000\s*ada\b/.test(m)) return GPU_TIERS.pro6000;
   157→  if (/\ba100\b/.test(m)) return GPU_TIERS.a100;
   158→  if (/\bl40s?\b/.test(m)) return GPU_TIERS.a6000; // 48GB Ada-class
   159→  if (/\ba40\b/.test(m)) return GPU_TIERS.a6000;
   160→  if (/\ba6000\b/.test(m)) return GPU_TIERS.a6000;
   161→  if (/\ba5000\b/.test(m)) return GPU_TIERS.vram16; // 24GB but older — mid tier
   162→  if (/\ba10\b/.test(m)) return GPU_TIERS.vram16;
   163→  if (/\bt4\b/.test(m)) return GPU_TIERS.entry;
   164→  if (/\b(?:rtx\s*)?(?:4090|3090|5090)\b/.test(m)) return GPU_TIERS.consumer24;
   165→  if (/\bmi300x\b/.test(m)) return GPU_TIERS.h200; // 192GB HBM3 accelerator class
   166→  return null;
   167→}

   205→/** Deterministic GPU tier from the reward stream — fallback for subnets
   206→ *  with no registered identity. What per-miner revenue can plausibly fund. */
   207→export function estimateGpuTierFromRevenue(monthlyUsdPerMiner: number): GpuTier {
   208→  if (monthlyUsdPerMiner >= 1200) return GPU_TIERS.h200;
   209→  if (monthlyUsdPerMiner >= 600) return GPU_TIERS.h100;
   210→  if (monthlyUsdPerMiner >= 300) return GPU_TIERS.a100;
   211→  if (monthlyUsdPerMiner >= 140) return GPU_TIERS.a6000;
   212→  if (monthlyUsdPerMiner >= 60) return GPU_TIERS.consumer24;
   213→  if (monthlyUsdPerMiner >= 25) return GPU_TIERS.vram16;
   214→  if (monthlyUsdPerMiner >= 8) return GPU_TIERS.entry;
   215→  return GPU_TIERS.cpu;
   216→}

   242→export function classifySubnetHardware(
   243→  name: string | null | undefined,
   244→  description: string | null | undefined,
   245→  fallbacks?: {
   246→    /** Curated category text (e.g. "Inference") when the chain identity is absent. */
   247→    fallbackCategory?: string;
   248→    /** Curated/user GPU figures win when the classifier has no keyword match. */
   249→    fallbackVramGb?: number;
   250→    fallbackGpu?: string;
   251→    /** Per-earning-miner monthly USD — revenue-based fallback tier. */
   252→    fallbackMonthlyUsd?: number;
   253→    /** GitHub-scraped requirements — GROUND TRUTH, wins over everything. */
   254→    scraped?: {
   255→      recommendedGpu?: string | null;
   256→      gpuCount?: number | null;
   257→      minVramGb?: number | null;
   258→      hosting?: HostingRequirements | null;
   259→      requirementsSource?: string | null;
   260→    } | null;
   261→    /** Curated official mechanics (mechanics.ts) — reward window + ops costs. */
   262→    mechanics?: SubnetMechanics | null;
   263→  }
   264→): SubnetHardwareProfile {
   265→  // --- Base classification: work-type keywords → tier (existing logic) ----
   266→  const base = (() => {
   267→    const text = `${name ?? ""} ${description ?? ""} ${fallbacks?.fallbackCategory ?? ""}`;
   268→    for (const rule of CATEGORY_RULES) {
   269→      if (rule.keywords.test(text)) {
   270→        return {
   271→          profile: {
   272→            category: rule.category,
   273→            tier: rule.tier,
   274→            minVramGb: rule.tier.minVramGb,
   275→            recommendedGpu: rule.tier.recommendedGpu,
   276→            monthlyCostUsd: rule.tier.monthlyRentUsd + (rule.infra ?? INFRA_BASE_USD),
   277→            isGpuWorkload: rule.tier.minVramGb > 0,
   278→            classified: true,
   279→          } as SubnetHardwareProfile,
   280→          infraUsd: rule.infra ?? INFRA_BASE_USD,
   281→        };
   282→      }
   283→    }
   284→    // No keyword match — fall back to curated VRAM data, else revenue tier.
   285→    if (fallbacks?.fallbackVramGb && fallbacks.fallbackVramGb > 0) {
   286→      const tier = Object.values(GPU_TIERS).find(
   287→        (t) => t.minVramGb === fallbacks.fallbackVramGb
   288→      ) ?? GPU_TIERS.consumer24;
   289→      return {
   290→        profile: {
   291→          category: fallbacks.fallbackCategory || "Unclassified workload",
   292→          tier,
   293→          minVramGb: tier.minVramGb,
   294→          recommendedGpu: fallbacks.fallbackGpu ?? tier.recommendedGpu,
   295→          monthlyCostUsd: tier.monthlyRentUsd + INFRA_BASE_USD,
   296→          isGpuWorkload: true,
   297→          classified: false,
   298→        } as SubnetHardwareProfile,
   299→        infraUsd: INFRA_BASE_USD,
   300→      };
   301→    }
   302→    const tier = estimateGpuTierFromRevenue(fallbacks?.fallbackMonthlyUsd ?? 0);
   303→    return {
   304→      profile: {
   305→        category: fallbacks?.fallbackCategory || "Unclassified workload",
   306→        tier,
   307→        minVramGb: tier.minVramGb,
   308→        recommendedGpu: tier.recommendedGpu,
   309→        monthlyCostUsd: tier.monthlyRentUsd + INFRA_BASE_USD,
   310→        isGpuWorkload: tier.minVramGb > 0,
   311→        classified: false,
   312→      } as SubnetHardwareProfile,
   313→      infraUsd: INFRA_BASE_USD,
   314→    };
   315→  })();
   316→
   317→  // --- Scraped repo requirements layer on top: the subnet's own README ----
   318→  // (or its miner repo) is ground truth for GPU model + hosting rules.
   319→  // It overrides ONLY what it actually knows — a hosting-only README keeps
   320→  // the work-type GPU tier; a GPU-only README keeps the classifier category.
   321→  const scraped = fallbacks?.scraped;
   322→  const scrapedTier = scraped?.recommendedGpu
   323→    ? resolveGpuTierFromModel(scraped.recommendedGpu)
   324→    : null;
   325→  if (!scraped || (!scrapedTier && !scraped.hosting)) return base.profile;
   326→
   327→  const count = scraped.gpuCount && scraped.gpuCount > 1 ? scraped.gpuCount : 1;
   328→  const tier = scrapedTier ?? base.profile.tier;
   329→  // MECHANICS-1 — hosting-aware cost class: a subnet whose own README rejects
   330→  // container clouds (bareMetalOnly) can only be served from the dedicated
   331→  // market, so the per-GPU unit rent switches to the bare-metal rate.
   332→  const bareMetal = scraped.hosting?.bareMetalOnly === true;
   333→  const unitRent = bareMetal ? tier.bareMetalMonthlyUsd : tier.monthlyRentUsd;
   334→  // The scraper's raw string already carries the count prefix ("8x H200").
   335→  const gpuLabel = scraped.recommendedGpu?.trim() ?? base.profile.recommendedGpu;
   336→  // Official mechanics may demand extra always-on infra (Chutes: a separate
   337→  // non-GPU control-plane server for the miner API/gepetto/postgres).
   338→  const controlPlaneUsd =
   339→    bareMetal && fallbacks?.mechanics?.controlPlaneMonthlyUsd
   340→      ? fallbacks.mechanics.controlPlaneMonthlyUsd
   341→      : 0;
   342→  return {
   343→    ...base.profile,
   344→    tier,
   345→    minVramGb: tier.minVramGb,
   346→    recommendedGpu: gpuLabel,
   347→    monthlyCostUsd: unitRent * count + base.infraUsd + controlPlaneUsd,
   348→    isGpuWorkload: tier.minVramGb > 0,
   349→    gpuCount: count,
   350→    hosting: scraped.hosting ?? null,
   351→    requirementsSource: scraped.requirementsSource ?? null,
   352→    sourceScraped: true,
   353→    unitRentUsd: unitRent,
   354→    costClass: bareMetal ? "bare-metal" : "container",
   355→  };
   356→}

   434→  hardware?: SubnetHardwareProfile | null;
   435→  taoChange24h?: number;
   436→  /** Blocks since the subnet was registered (for the maturity bonus). */
   437→  liveAgeBlocks?: number | null;
   438→  /** Curated official mechanics (mechanics.ts) — ramp window + infra. */
   439→  mechanics?: SubnetMechanics | null;
   440→  /** User-configurable cost lines (Profitability Engine settings). When
   441→   *  absent, market defaults apply — same engine, same shape. */
   442→  costs?: {
   443→    hardwareMode?: "rent" | "owned";
   444→    electricityUsdPerKwh?: number;
   445→    storageMonthlyUsd?: number;
   446→    /** 0/undefined = auto infra from the work-type classifier. */
   447→    infraMonthlyUsd?: number;
   448→    otherOpexMonthlyUsd?: number;
   449→    includeBurnAmortization?: boolean;
   450→    amortizeBurnMonths?: number;
   451→  };
   452→}): MinerLedger {
   453→  const { live, taoUsd } = inputs;
   454→  const usd = taoUsd || 0;
   455→
   456→  // --- Top line: newcomer-adjusted per-miner revenue ----------------------
   457→  // Miners get 41% of the subnet's emission (docs constant); the chain-
   458→  // measured minerEmissionTaoPerDay already reflects the realized split.
   459→  //
   460→  // The decision a miner makes is "if I register TODAY, what does a
   461→  // mid-pack seat pay?" — so the ledger starts from the per-REGISTERED
   462→  // slot (zero-earning seats included: on many subnets 250+ UIDs are
   463→  // registered and only a handful earn), then applies the median/mean
   464→  // spread share so a whale UID can't inflate the estimate.
   465→  // perEarningDailyTao (mean over rewarded UIDs) is kept as the upside
   466→  // reference — what current earners make.
   467→  const rewarded = Math.max(live.rewardedMiners ?? 0, 0);
   468→  const registered = Math.max(live.minersCount, 1);
   469→  const minerEmissionTaoPerDay = live.minerEmissionTaoPerDay ?? 0;
   470→  const perRegisteredDailyTao =
   471→    minerEmissionTaoPerDay > 0 ? minerEmissionTaoPerDay / registered : 0;
   472→  const perEarningDailyTao =
   473→    rewarded > 0 && minerEmissionTaoPerDay > 0
   474→      ? minerEmissionTaoPerDay / rewarded
   475→      : minerEmissionTaoPerDay > 0
   476→        ? perRegisteredDailyTao // vec unavailable — no zero-slot discount available
   477→        : 0;
   478→  const rewardMedianShare = live.incentiveMedianShare ?? 1;
   479→  // Newcomer-adjusted mid-pack revenue:
   480→  //  - reward vec available  → perRegistered × medianShare
   481→  //  - reward vec unavailable→ perRegistered × 0.6 (conservative)
   482→  const expectedDailyTao =
   483→    rewarded > 0 && minerEmissionTaoPerDay > 0
   484→      ? perRegisteredDailyTao * rewardMedianShare
   485→      : minerEmissionTaoPerDay > 0
   486→        ? perRegisteredDailyTao * 0.6
   487→        : 0;
   488→  const grossMonthlyUsd = Math.round(expectedDailyTao * 30 * usd);
   489→  const dailyAlpha = live.movingPrice > 0 ? expectedDailyTao / live.movingPrice : 0;
   490→
   491→  // --- Hardware profile (work type → GPU cost) ---------------------------
   492→  const fallbackMonthly = grossMonthlyUsd > 0 ? grossMonthlyUsd : undefined;
   493→  const hardware =
   494→    inputs.hardware ??
   495→    classifySubnetHardware(live.name, live.identityDescription, {
   496→      fallbackMonthlyUsd: fallbackMonthly,
   497→    });
   498→  // --- Cost stack (Profitability Engine lines) ----------------------------
   499→  // gpuCount > 1 (scraped, e.g. "8x H200") scales the GPU line: the requirement
   500→  // is a multi-GPU server, not one card. Owned mode scales the power draw.
   501→  // MECHANICS-1: unitRent switches to the dedicated rate when the subnet's
   502→  // hosting rules reject containers — the ONLY compliant hosting class.
   503→  const costs = inputs.costs ?? {};
   504→  const gpuCount = hardware.gpuCount && hardware.gpuCount > 1 ? hardware.gpuCount : 1;
   505→  const unitRent = hardware.unitRentUsd ??
   506→    (hardware.hosting?.bareMetalOnly
   507→      ? hardware.tier.bareMetalMonthlyUsd
   508→      : hardware.tier.monthlyRentUsd);
   509→  const gpuCost =
   510→    costs.hardwareMode === "owned"
   511→      ? electricityMonthlyUsd(
   512→          hardware.tier.powerWatts * gpuCount,
   513→          costs.electricityUsdPerKwh ?? 0.12
   514→        )
   515→      : unitRent * gpuCount;
   516→  const infraCost =
   517→    costs.infraMonthlyUsd != null && costs.infraMonthlyUsd > 0
   518→      ? costs.infraMonthlyUsd
   519→      : Math.max(hardware.monthlyCostUsd - unitRent * gpuCount, INFRA_BASE_USD);
   520→  const storageCost = costs.storageMonthlyUsd ?? 0;
   521→  const otherOpex = costs.otherOpexMonthlyUsd ?? 0;
   522→  const burnUsd =
   523→    live.burnCostTao != null && costs.includeBurnAmortization
   524→      ? Math.max(live.burnCostTao, 0) * usd
   525→      : 0;
   526→  const amortizedBurn = burnUsd / Math.max(costs.amortizeBurnMonths ?? 3, 1);
   527→  const totalCosts = gpuCost + infraCost + storageCost + otherOpex + amortizedBurn;
   528→  // Zero-revenue subnets are honest losers: the cost stack still applies.
   529→  const netMonthlyUsd = grossMonthlyUsd > 0
   530→    ? Math.round(grossMonthlyUsd - totalCosts)
   531→    : -Math.round(totalCosts);
   532→
   533→  // --- Alpha economics ----------------------------------------------------
   534→  const alphaPriceTao = live.movingPrice;
   535→  const alphaPriceUsd = alphaPriceTao * usd;
   536→  const liquidityTao = live.subnetTao; // TAO side of the pool = exit liquidity
   537→  const slippagePct =
   538→    live.alphaOut > 0 && dailyAlpha > 0
   539→      ? Math.round((dailyAlpha / live.alphaOut) * 10000) / 100
   540→      : null;
   541→
   542→  // --- Seat safety inputs -------------------------------------------------
   543→  const maxUids = live.maxUids ?? 0;
   544→  const freeSlots = maxUids > 0 ? Math.max(0, maxUids - live.minersCount) : null;
   545→  const totalSlots = maxUids > 0 ? maxUids : null;
   546→  const rewardedRatio = live.rewardedMiners != null && live.rewardedMiners > 0
   547→    ? Math.min(1, live.rewardedMiners / registered)
   548→    : live.rewardedMiners === 0
   549→      ? 0
   550→      : null;
   551→  const top10 = live.top10IncentiveShare;
   552→
   553→  // Newcomer ramp. Default: bonds build via EMA (α=0.1), and on concentrated
   554→  // subnets a fresh UID starts at zero. Research: 1-3 months of tuning typical.
   555→  // MECHANICS-1: when the subnet's OFFICIAL docs define the reward metric as a
   556→  // rolling N-day window (Chutes: "7 day sum of compute"), the official window
   557→  // replaces the heuristic — a stable newcomer reaches full weight within it.
   558→  const mechanics = inputs.mechanics ?? null;
   559→  const mechanicsRampWeeks =
   560→    mechanics?.rewardWindowDays != null
   561→      ? Math.max(0.5, mechanics.rewardWindowDays / 7)
   562→      : null;
   563→  const rampWeeks =
   564→    mechanicsRampWeeks ??
   565→    (3 +
   566→      (rewardedRatio != null ? (1 - rewardedRatio) * 6 : 3) +
   567→      (top10 != null && top10 > 0.7 ? 3 : 0) +
   568→      (freeSlots != null && freeSlots === 0 ? 2 : 0));
   569→  const rampWeeksSource: MinerLedgerDiagnostics["rampWeeksSource"] =
   570→    mechanicsRampWeeks != null
   571→      ? mechanics?.provenance === "derived"
   572→        ? "readme-derived"
   573→        : "official-reward-window"
   574→      : "bond-ema-heuristic";
   575→
   576→  // --- Pillar 1: Net ROI --------------------------------------------------
   577→  // $0 → 10, +$100 → 54, +$500 → 69, +$1k → 76, +$5k → 92; losses slide to 2.
   578→  const netRoi =
   579→    netMonthlyUsd >= 0
   580→      ? clampScore(10 + 22 * Math.log10(netMonthlyUsd + 1), 2, 97)
   581→      : clampScore(10 + netMonthlyUsd / 100, 2, 10);
   582→
   583→  // --- Pillar 2: Seat safety ----------------------------------------------
   584→  const roomRatio = maxUids > 0 ? freeSlots! / maxUids : 0.5;
   585→  const scoreRoom = clampScore(10 + roomRatio * 45, 10, 55); // full subnet → 10
   586→  const scoreSpread =
   587→    top10 != null ? clampScore(100 - (top10 - 0.2) * 100, 10, 95) : 50;
   588→  const scoreBurn =
   589→    live.burnCostTao != null
   590→      ? clampScore(95 - Math.log10(live.burnCostTao + 1) * 25, 30, 95)
   591→      : 55;
   592→  const scoreImmunity =
   593→    live.immunityBlocks != null
   594→      ? clampScore(30 + (live.immunityBlocks / 7200) * 50, 30, 80)
   595→      : 45;
   596→  const seatSafety = !live.emissionEnabled
   597→    ? 12
   598→    : clampScore(
   599→        scoreRoom * 0.35 + scoreSpread * 0.35 + scoreBurn * 0.15 + scoreImmunity * 0.15,
   600→        5,
   601→        96
   602→      );
   603→
   604→  // --- Pillar 3: Alpha economics -------------------------------------------
   605→  const change = live.alphaPriceChange24h;
   606→  const scoreTrend = change != null ? clampScore(50 + change * 1.5, 10, 95) : 50;
   607→  const scoreLiq =
   608→    liquidityTao > 0
   609→      ? clampScore(20 + 19 * Math.log10(liquidityTao / 500 + 1), 10, 95)
   610→      : 10;
   611→  const scoreSlip =
   612→    slippagePct != null ? clampScore(95 - slippagePct * 1200, 15, 95) : 50;
   613→  const alphaEconomics = usd <= 0
   614→    ? 50
   615→    : clampScore(scoreTrend * 0.4 + scoreLiq * 0.3 + scoreSlip * 0.3, 5, 96);
   616→
   617→  // --- Pillar 4: Earning reality -------------------------------------------
   618→  const scoreRewarded =
   619→    rewardedRatio != null ? clampScore(15 + rewardedRatio * 70, 15, 85) : 50;
   620→  const scoreConc =
   621→    top10 != null ? clampScore(100 - (top10 - 0.2) * 90, 15, 92) : 50;
   622→  const scoreRamp = clampScore(90 - rampWeeks * 5, 20, 90);
   623→  const earningReality = !live.emissionEnabled
   624→    ? 8
   625→    : clampScore(scoreRewarded * 0.4 + scoreConc * 0.3 + scoreRamp * 0.3, 5, 96);
   626→
   627→  // --- Pillar 5: Fit & feasibility ------------------------------------------
   628→  let fitFeasibility = 25; // emission disabled base
   629→  if (live.emissionEnabled) {
   630→    fitFeasibility = 40;
   631→    const validators = live.validatorsCount ?? 0;
   632→    fitFeasibility += clampScore(validators * 2.5, 0, 20); // validator coverage
   633→    if (live.identityGithub) fitFeasibility += 12;
   634→    if (live.identityDescription) fitFeasibility += 6;
   635→    if (live.registeredAt != null && inputs.liveAgeBlocks != null) {
   636→      const ageDays = inputs.liveAgeBlocks / 7200;
   637→      fitFeasibility += clampScore(Math.min(ageDays / 30, 1) * 12, 0, 12); // maturity
   638→    }
   639→    if (live.immunityBlocks != null && live.immunityBlocks < 4096) fitFeasibility -= 5;
   640→  }
   641→  fitFeasibility = clampScore(fitFeasibility, 5, 92);
   642→
   643→  const components: ScoreComponents = {
   644→    net_roi: Math.round(netRoi * 10) / 10,
   645→    seat_safety: Math.round(seatSafety * 10) / 10,
   646→    alpha_economics: Math.round(alphaEconomics * 10) / 10,
   647→    earning_reality: Math.round(earningReality * 10) / 10,
   648→    fit_feasibility: Math.round(fitFeasibility * 10) / 10,
   649→  };
   650→
   651→  const diag: MinerLedgerDiagnostics = {
   652→    perEarningDailyTao: Math.round(perEarningDailyTao * 10000) / 10000,
   653→    expectedDailyTao: Math.round(expectedDailyTao * 10000) / 10000,
   654→    rewardMedianShare: live.incentiveMedianShare,
   655→    perRegisteredDailyTao: Math.round(perRegisteredDailyTao * 10000) / 10000,
   656→    grossMonthlyUsd,
   657→    alphaPriceTao,
   658→    alphaPriceUsd: Math.round(alphaPriceUsd * 1000) / 1000,
   659→    alphaChange24h: change ?? null,
   660→    liquidityTao: Math.round(liquidityTao),
   661→    slippagePct,
   662→    category: hardware.category,
   663→    gpuLabel: hardware.tier.label,
   664→    recommendedGpu: hardware.recommendedGpu,
   665→    minVramGb: hardware.minVramGb,
   666→    gpuCostMonthlyUsd: Math.round(gpuCost),
   667→    gpuCount: hardware.gpuCount ?? null,
   668→    hosting: hardware.hosting ?? null,
   669→    requirementsSource: hardware.requirementsSource ?? null,
   670→    costClass: hardware.costClass ?? (hardware.hosting?.bareMetalOnly ? "bare-metal" : "container"),
   671→    unitRentUsd: Math.round(unitRent),
   672→    infraCostMonthlyUsd: Math.round(infraCost),
   673→    storageCostMonthlyUsd: Math.round(storageCost),
   674→    otherOpexMonthlyUsd: Math.round(otherOpex),
   675→    amortizedBurnUsd: Math.round(amortizedBurn),
   676→    totalCostMonthlyUsd: Math.round(totalCosts),
   677→    gpuPowerWatts: hardware.tier.powerWatts * gpuCount,
   678→    netMonthlyUsd,
   679→    netDailyTao:
   680→      usd > 0 ? Math.round((netMonthlyUsd / 30 / usd) * 10000) / 10000 : 0,
   681→    freeSlots,
   682→    totalSlots,
   683→    rewardedRatio: rewardedRatio != null ? Math.round(rewardedRatio * 1000) / 1000 : null,
   684→    top10IncentiveShare: top10,
   685→    burnCostTao: live.burnCostTao,
   686→    immunityBlocks: live.immunityBlocks,
   687→    rampWeeks: Math.round(rampWeeks * 10) / 10,
   688→    rampWeeksSource,
   689→    mechanicsApplied:
   690→      mechanics && mechanicsOverridesRamp(mechanics)
   691→        ? mechanics.sources.map((s) => s.label)
   692→        : null,
   693→    emissionEnabled: live.emissionEnabled,
   694→    hardwareClassified: hardware.classified,
   695→  };
   696→
   697→  return { components, factors: deriveFactors(components), diag };
   698→}
   699→
   700→// ---------------------------------------------------------------------------
   701→// Chance to earn · month 1 — a newcomer's honest odds of earning ANY reward
   702→// during their first month of continuous, good-uptime mining. Uptime alone
   703→// keeps the seat; the share of slots that actually earn decides whether the
   704→// wallet moves. Dominant factor: rewardedRatio (share of registered slots
   705→// that earned last epoch), discounted for whale dominance, full subnets
   706→// (must displace an incumbent) and long bond-EMA ramps.
   707→// ---------------------------------------------------------------------------
   708→
   709→export type EarnChanceLevel = "high" | "medium" | "low" | "none";
   710→
   711→export interface EarnChance {
   712→  level: EarnChanceLevel;
   713→  /** 0–100 modeled chance that a newcomer earns something in month 1. */
   714→  pct: number;
   715→  /** Human explanation of the dominant factors. */
   716→  note: string;
   717→}
   718→
   719→export function computeEarnChance(inputs: {
   720→  /** Share of registered slots that earned reward last epoch (0..1). */
   721→  rewardedRatio: number | null;
   722→  /** Free UID slots (null = unknown). 0 means you must displace an incumbent. */
   723→  freeSlots: number | null;
   724→  /** Top-10% share of epoch incentive (0..1) — whale dominance. */
   725→  top10IncentiveShare: number | null;
   726→  /** Newcomer bond-EMA ramp estimate in weeks. */
   727→  rampWeeks: number | null;
   728→}): EarnChance {
   729→  const { rewardedRatio, freeSlots, top10IncentiveShare, rampWeeks } = inputs;
   730→
   731→  // Base: the fraction of slots that actually earn — the single strongest
   732→  // predictor of whether a newcomer earns anything at all.
   733→  let pct = rewardedRatio != null ? rewardedRatio * 100 : 0;
   734→  const factors: string[] = [];
   735→
   736→  // Whale dominance: when the top 10% of UIDs take nearly everything, the
   737→  // rewarded tail is scraps — being "rewarded" still pays almost nothing.
   738→  if (top10IncentiveShare != null && top10IncentiveShare > 0.5) {
   739→    pct *= 1 - Math.min(1, top10IncentiveShare - 0.5) * 0.6;
   740→    factors.push(
   741→      `top-10% of UIDs take ${Math.round(top10IncentiveShare * 100)}% of rewards`
   742→    );
   743→  }
   744→
   745→  // Full subnet: no free seats — you must outperform an incumbent to get in,
   746→  // and you start from zero bonds against their established ones.
   747→  if (freeSlots === 0) {
   748→    pct *= 0.7;
   749→    factors.push("subnet full — must displace an incumbent");
   750→  }
   751→
   752→  // Long ramp: month 1 is mostly bond-building; long ramps leave little of it.
   753→  if (rampWeeks != null && rampWeeks > 8) {
   754→    pct *= 0.8;
   755→    factors.push(`~${Math.round(rampWeeks)} wk ramp to full bonds`);
   756→  }
   757→
   758→  pct = Math.round(Math.max(0, Math.min(100, pct)));
   759→  const level: EarnChanceLevel =
   760→    rewardedRatio != null && rewardedRatio <= 0
   761→      ? "none"
   762→      : pct >= 50
   763→        ? "high"
   764→        : pct >= 20
   765→          ? "medium"
   766→          : pct > 0
   767→            ? "low"
   768→            : "none";
   769→
   770→  const base = rewardedRatio != null
   771→    ? `${rewardedRatio < 0.1 && rewardedRatio > 0 ? (rewardedRatio * 100).toFixed(1) : Math.round(rewardedRatio * 100)}% of slots earned last epoch`
   772→    : "reward spread unknown";
   773→  const note =
   774→    factors.length > 0 ? `${base} · ${factors.join(" · ")}` : base;
   775→
   776→  return { level, pct, note };
   777→}

   779→// ---------------------------------------------------------------------------
   780→// Seat Chance — "can I actually GET IN?" (vs EarnChance's "will I earn?").
   781→//
   782→// Bittensor registration mechanics:
   783→//   • Subnets have a UID cap (maxUids). Free slots → register via burn or PoW.
   784→//   • FULL subnet → registration STILL works: a burn registration pays the
   785→//     dynamic recycle cost and the chain replaces the worst-performing
   786→//     NON-IMMUNE uid. PoW registrations compete for per-block slots the same
   787→//     way. You are only locked out if every uid is immune (rare — immunity
   788→//     covers ~14h by default) or registration is disabled on the subnet.
   789→//   • immunityBlocks = how long YOUR new seat cannot be deregistered — the
   790→//     runway to start earning before becoming replaceable yourself.
   791→// ---------------------------------------------------------------------------
   792→
   793→/**
   794→ * Burn-quote display precision — the chain's burn hyperparam can sit at the
   795→ * floor (e.g. Chutes ~0.0005 TAO); 3 decimals would show "0.001" — a 2x
   796→ * misquote shown to a user deciding whether to burn. <0.01 → 4 decimals,
   797→ * <1 → 3, else 2.
   798→ */
   799→export function formatBurnTao(n: number): string {
   800→  if (n < 0.01) return n.toFixed(4);
   801→  if (n < 1) return n.toFixed(3);
   802→  return n.toFixed(2);
   803→}
   804→
   805→export type SeatVerdict = "open" | "burn-entry" | "waitlist" | "unknown";
   806→
   807→export interface SeatChance {
   808→  verdict: SeatVerdict;
   809→  /** Registration is possible right now (burn path or free slot). */
   810→  canRegister: boolean;
   811→  slotsFree: number | null;
   812→  totalSlots: number | null;
   813→  /** 0–100 — how full the uid space is. */
   814→  fillPct: number | null;
   815→  burnCostTao: number | null;
   816→  /** Protection window for a NEW seat, in hours (blocks × 12s). */
   817→  immunityHours: number | null;
   818→  /** Share of uids that earned NOTHING last epoch — the replaceable bottom
   819→   *  a newcomer displaces from. High = churn churns in your favor. */
   820→  replaceableShare: number | null;
   821→  headline: string;
   822→  detail: string;
   823→}
   824→
   825→export function assessSeatChance(inputs: {
   826→  minersCount: number | null;
   827→  maxUids: number | null;
   828→  /** Already-computed free slots (skip when you only have counts). */
   829→  freeSlots?: number | null;
   830→  burnCostTao: number | null;
   831→  immunityBlocks: number | null;
   832→  /** Rewarded uids (either give the ratio directly or the absolute count). */
   833→  rewardedRatio?: number | null;
   834→  rewardedMiners?: number | null;
   835→}): SeatChance {
   836→  const { maxUids, burnCostTao, immunityBlocks } = inputs;
   837→  const minersCount = inputs.minersCount ?? 0;
   838→  const slotsFree =
   839→    inputs.freeSlots != null
   840→      ? Math.max(0, inputs.freeSlots)
   841→      : maxUids != null && maxUids > 0
   842→        ? Math.max(0, maxUids - minersCount)
   843→        : null;
   844→  const fillPct =
   845→    maxUids != null && maxUids > 0
   846→      ? Math.min(100, Math.round((minersCount / maxUids) * 100))
   847→      : null;
   848→  let replaceableShare: number | null = null;
   849→  if (inputs.rewardedRatio != null) {
   850→    replaceableShare = Math.max(0, Math.min(1, 1 - inputs.rewardedRatio));
   851→  } else if (inputs.rewardedMiners != null && minersCount > 0) {
   852→    replaceableShare = Math.max(0, Math.min(1, 1 - inputs.rewardedMiners / minersCount));
   853→  }
   854→  const immunityHours =
   855→    immunityBlocks != null && immunityBlocks > 0
   856→      ? Math.round(((immunityBlocks * 12) / 3600) * 10) / 10
   857→      : null;
   858→  const immunityNote = immunityHours != null
   859→    ? `a new seat is immune for ~${immunityHours}h — your runway to start earning before you're replaceable`
   860→    : null;
   861→  const deadNote = replaceableShare != null && replaceableShare > 0.05
   862→    ? `${Math.round(replaceableShare * 100)}% of uids earned nothing last epoch — a deep replaceable bottom`
   863→    : replaceableShare != null
   864→      ? "almost every seat earned last epoch — displacement targets are scarce and strong"
   865→      : null;
   866→
   867→  const unknown: SeatChance = {
   868→    verdict: "unknown",
   869→    canRegister: true,
   870→    slotsFree,
   871→    totalSlots: maxUids ?? null,
   872→    fillPct,
   873→    burnCostTao,
   874→    immunityHours,
   875→    replaceableShare,
   876→    headline: "slot data unavailable",
   877→    detail:
   878→      "UID capacity was not returned by the chain snapshot this pass — refresh the network data. " +
   879→      (burnCostTao != null ? `Burn registration quote: ~${formatBurnTao(burnCostTao)} TAO.` : ""),
   880→  };
   881→
   882→  if (slotsFree == null || maxUids == null || maxUids <= 0) return unknown;
   883→
   884→  // --- Open: free slots exist ---------------------------------------------
   885→  if (slotsFree > 0) {
   886→    return {
   887→      verdict: "open",
   888→      canRegister: true,
   889→      slotsFree,
   890→      totalSlots: maxUids,
   891→      fillPct,
   892→      burnCostTao,
   893→      immunityHours,
   894→      replaceableShare,
   895→      headline: `${slotsFree} of ${maxUids} slots free`,
   896→      detail:
   897→        `This subnet is NOT full — ${slotsFree} uid${slotsFree === 1 ? "" : "s"} open (${fillPct}% filled). ` +
   898→        `Register now via burn${burnCostTao != null ? ` (~${formatBurnTao(burnCostTao)} TAO)` : ""} or PoW. ` +
   899→        (immunityNote ? `${immunityNote}. ` : "") +
   900→        (deadNote ? `${deadNote}.` : ""),
   901→    };
   902→  }
