     2→// Miner's Ledger scoring v2 — models how an experienced GPU miner actually
     3→// picks a subnet. Research-backed (Bittensor docs, taostats miner guide,
     4→// miner-economics deep dives, dTAO analyst frameworks):
     5→//
     6→//   Q1 FIT        — can my hardware run this? (subnet type → GPU tier/cost)
     7→//   Q2 TOP LINE   — what does it pay? (emission × 41% miner share ÷ EARNING
     8→//                   miners, priced in alpha → TAO → USD)
     9→//   Q3 BOTTOM LINE— what do I keep? (minus GPU rental + infra → net monthly)
    10→//   Q4 SEAT SAFETY— can I keep the slot? (saturation, reward concentration,
    11→//                   burn cost, immunity runway)
    12→//   Q5 HOLD VALUE — will revenue persist? (alpha price trend, pool liquidity,
    13→//                   sell-pressure slippage, owner identity, maturity)
    14→//
    15→// Key constants from the chain: per-tempo split is 18% owner / 41% miners /
    16→// 41% validators+stakers. "Top 10-20% of miners take 60-80% of emissions;
    17→// median miners hover near breakeven" — so the ledger scores NET numbers,
    18→// not gross, and treats reward concentration as a first-class risk.
    19→// ---------------------------------------------------------------------------

    33→export interface ScoreComponents {
    34→  /** Net ROI after GPU rental + infra costs (the bottom line). */
    35→  net_roi: number;
    36→  /** Seat safety — free slots, reward spread, burn cost, immunity runway. */
    37→  seat_safety: number;
    38→  /** Alpha economics — 24h price trend, pool liquidity, sell slippage. */
    39→  alpha_economics: number;
    40→  /** Earning reality — rewarded ratio, concentration, newcomer ramp penalty. */
    41→  earning_reality: number;
    42→  /** Fit & feasibility — emission on, validator coverage, identity, maturity. */
    43→  fit_feasibility: number;
    44→}
    45→
    46→export const SCORE_WEIGHTS: Record<keyof ScoreComponents, number> = {
    47→  net_roi: 0.3,
    48→  seat_safety: 0.2,
    49→  alpha_economics: 0.2,
    50→  earning_reality: 0.15,
    51→  fit_feasibility: 0.15,
    52→};
    53→
    54→const FACTOR_LABELS: Record<keyof ScoreComponents, string> = {
    55→  net_roi: "Net ROI (after GPU cost)",
    56→  seat_safety: "Seat Safety",
    57→  alpha_economics: "Alpha Economics",
    58→  earning_reality: "Earning Reality",
    59→  fit_feasibility: "Fit & Feasibility",
    60→};
    61→
    62→const FACTOR_DESC: Record<keyof ScoreComponents, string> = {
    63→  net_roi:
    64→    "Newcomer-adjusted mid-pack monthly USD (per-registered slot × reward spread) minus GPU rental + infra at market rates — the miner's bottom line, not gross emission.",
    65→  seat_safety:
    66→    "Free UID slots, reward spread (top-10% concentration), registration burn level and immunity runway — can you keep the seat?",
    67→  alpha_economics:
    68→    "dTAO hold value: alpha price 24h trend, TAO-side pool depth (exit liquidity) and slippage from converting daily earnings.",
    69→  earning_reality:
    70→    "Share of registered miners actually rewarded, top-10% emission concentration and the bond-EMA ramp penalty for newcomers.",
    71→  fit_feasibility:
    72→    "Emission enabled, validator coverage, on-chain identity/GitHub and subnet maturity — operational basics before committing hardware.",
    73→};
    74→
    75→export function deriveFactors(c: ScoreComponents): OpportunityFactor[] {
    76→  return (Object.keys(c) as (keyof ScoreComponents)[]).map((k) => {
    77→    const raw = c[k];
    78→    const weight = SCORE_WEIGHTS[k];
    79→    return {
    80→      name: FACTOR_LABELS[k],
    81→      value: Math.round(raw * weight * 100) / 100,
    82→      weight,
    83→      raw,
    84→      impact: raw >= 60 ? "positive" : raw <= 40 ? "negative" : "neutral",
    85→      description: FACTOR_DESC[k],
    86→    };
    87→  });
    88→}
    89→
    90→export function totalScore(c: ScoreComponents): number {
    91→  return Math.round(
    92→    (Object.keys(c) as (keyof ScoreComponents)[]).reduce(
    93→      (acc, k) => acc + c[k] * SCORE_WEIGHTS[k],
    94→      0
    95→    ) * 10
    96→  ) / 10;
    97→}
    98→
    99→export function riskLevel(score: number): "low" | "medium" | "high" {
   100→  if (score >= 60) return "low";
   101→  if (score >= 40) return "medium";
   102→  return "high";
   103→}

   134→/**
   135→ * Resolve a GPU tier from a model string found in a repo README
   136→ * (github-scraper parseGpuRequirement output). Ground truth from the
   137→ * subnet's own docs beats keyword classification.
   138→ */
   139→export function resolveGpuTierFromModel(model: string): GpuTier | null {
   140→  const m = model.toLowerCase();
   141→  if (/\bb300\b/.test(m)) return GPU_TIERS.b200; // same Blackwell HGX class
   142→  if (/\bb200\b/.test(m)) return GPU_TIERS.b200;
   143→  if (/\bh200\b/.test(m)) return GPU_TIERS.h200;
   144→  if (/\bh100\b/.test(m)) return GPU_TIERS.h100;
   145→  if (/\b(?:rtx\s*)?pro\s*6000\b/.test(m) || /\brtx\s*6000\s*ada\b/.test(m)) return GPU_TIERS.pro6000;
   146→  if (/\ba100\b/.test(m)) return GPU_TIERS.a100;
   147→  if (/\bl40s?\b/.test(m)) return GPU_TIERS.a6000; // 48GB Ada-class
   148→  if (/\ba40\b/.test(m)) return GPU_TIERS.a6000;
   149→  if (/\ba6000\b/.test(m)) return GPU_TIERS.a6000;
   150→  if (/\ba5000\b/.test(m)) return GPU_TIERS.vram16; // 24GB but older — mid tier
   151→  if (/\ba10\b/.test(m)) return GPU_TIERS.vram16;
   152→  if (/\bt4\b/.test(m)) return GPU_TIERS.entry;
   153→  if (/\b(?:rtx\s*)?(?:4090|3090|5090)\b/.test(m)) return GPU_TIERS.consumer24;
   154→  if (/\bmi300x\b/.test(m)) return GPU_TIERS.h200; // 192GB HBM3 accelerator class
   155→  return null;
   156→}

   187→/** Deterministic GPU tier from the reward stream — fallback for subnets
   188→ *  with no registered identity. What per-miner revenue can plausibly fund. */
   189→export function estimateGpuTierFromRevenue(monthlyUsdPerMiner: number): GpuTier {
   190→  if (monthlyUsdPerMiner >= 1200) return GPU_TIERS.h200;
   191→  if (monthlyUsdPerMiner >= 600) return GPU_TIERS.h100;
   192→  if (monthlyUsdPerMiner >= 300) return GPU_TIERS.a100;
   193→  if (monthlyUsdPerMiner >= 140) return GPU_TIERS.a6000;
   194→  if (monthlyUsdPerMiner >= 60) return GPU_TIERS.consumer24;
   195→  if (monthlyUsdPerMiner >= 25) return GPU_TIERS.vram16;
   196→  if (monthlyUsdPerMiner >= 8) return GPU_TIERS.entry;
   197→  return GPU_TIERS.cpu;
   198→}

   224→export function classifySubnetHardware(
   225→  name: string | null | undefined,
   226→  description: string | null | undefined,
   227→  fallbacks?: {
   228→    /** Curated category text (e.g. "Inference") when the chain identity is absent. */
   229→    fallbackCategory?: string;
   230→    /** Curated/user GPU figures win when the classifier has no keyword match. */
   231→    fallbackVramGb?: number;
   232→    fallbackGpu?: string;
   233→    /** Per-earning-miner monthly USD — revenue-based fallback tier. */
   234→    fallbackMonthlyUsd?: number;
   235→    /** GitHub-scraped requirements — GROUND TRUTH, wins over everything. */
   236→    scraped?: {
   237→      recommendedGpu?: string | null;
   238→      gpuCount?: number | null;
   239→      minVramGb?: number | null;
   240→      hosting?: HostingRequirements | null;
   241→      requirementsSource?: string | null;
   242→    } | null;
   243→  }
   244→): SubnetHardwareProfile {
   245→  // --- Base classification: work-type keywords → tier (existing logic) ----
   246→  const base = (() => {
   247→    const text = `${name ?? ""} ${description ?? ""} ${fallbacks?.fallbackCategory ?? ""}`;
   248→    for (const rule of CATEGORY_RULES) {
   249→      if (rule.keywords.test(text)) {
   250→        return {
   251→          profile: {
   252→            category: rule.category,
   253→            tier: rule.tier,
   254→            minVramGb: rule.tier.minVramGb,
   255→            recommendedGpu: rule.tier.recommendedGpu,
   256→            monthlyCostUsd: rule.tier.monthlyRentUsd + (rule.infra ?? INFRA_BASE_USD),
   257→            isGpuWorkload: rule.tier.minVramGb > 0,
   258→            classified: true,
   259→          } as SubnetHardwareProfile,
   260→          infraUsd: rule.infra ?? INFRA_BASE_USD,
   261→        };
   262→      }
   263→    }
   264→    // No keyword match — fall back to curated VRAM data, else revenue tier.
   265→    if (fallbacks?.fallbackVramGb && fallbacks.fallbackVramGb > 0) {
   266→      const tier = Object.values(GPU_TIERS).find(
   267→        (t) => t.minVramGb === fallbacks.fallbackVramGb
   268→      ) ?? GPU_TIERS.consumer24;
   269→      return {
   270→        profile: {
   271→          category: fallbacks.fallbackCategory || "Unclassified workload",
   272→          tier,
   273→          minVramGb: tier.minVramGb,
   274→          recommendedGpu: fallbacks.fallbackGpu ?? tier.recommendedGpu,
   275→          monthlyCostUsd: tier.monthlyRentUsd + INFRA_BASE_USD,
   276→          isGpuWorkload: true,
   277→          classified: false,
   278→        } as SubnetHardwareProfile,
   279→        infraUsd: INFRA_BASE_USD,
   280→      };
   281→    }
   282→    const tier = estimateGpuTierFromRevenue(fallbacks?.fallbackMonthlyUsd ?? 0);
   283→    return {
   284→      profile: {
   285→        category: fallbacks?.fallbackCategory || "Unclassified workload",
   286→        tier,
   287→        minVramGb: tier.minVramGb,
   288→        recommendedGpu: tier.recommendedGpu,
   289→        monthlyCostUsd: tier.monthlyRentUsd + INFRA_BASE_USD,
   290→        isGpuWorkload: tier.minVramGb > 0,
   291→        classified: false,
   292→      } as SubnetHardwareProfile,
   293→      infraUsd: INFRA_BASE_USD,
   294→    };
   295→  })();
   296→
   297→  // --- Scraped repo requirements layer on top: the subnet's own README ----
   298→  // (or its miner repo) is ground truth for GPU model + hosting rules.
   299→  // It overrides ONLY what it actually knows — a hosting-only README keeps
   300→  // the work-type GPU tier; a GPU-only README keeps the classifier category.
   301→  const scraped = fallbacks?.scraped;
   302→  const scrapedTier = scraped?.recommendedGpu
   303→    ? resolveGpuTierFromModel(scraped.recommendedGpu)
   304→    : null;
   305→  if (!scraped || (!scrapedTier && !scraped.hosting)) return base.profile;
   306→
   307→  const count = scraped.gpuCount && scraped.gpuCount > 1 ? scraped.gpuCount : 1;
   308→  const tier = scrapedTier ?? base.profile.tier;
   309→  // The scraper's raw string already carries the count prefix ("8x H200").
   310→  const gpuLabel = scraped.recommendedGpu?.trim() ?? base.profile.recommendedGpu;
   311→  return {
   312→    ...base.profile,
   313→    tier,
   314→    minVramGb: tier.minVramGb,
   315→    recommendedGpu: gpuLabel,
   316→    monthlyCostUsd: tier.monthlyRentUsd * count + base.infraUsd,
   317→    isGpuWorkload: tier.minVramGb > 0,
   318→    gpuCount: count,
   319→    hosting: scraped.hosting ?? null,
   320→    requirementsSource: scraped.requirementsSource ?? null,
   321→    sourceScraped: true,
   322→  };
   323→}

   384→/**
   385→ * Score one subnet the way a miner reads it. Deterministic per snapshot —
   386→ * the same inputs always produce the same score so ranks stay stable.
   387→ */
   388→export function scoreMinersLedger(inputs: {
   389→  live: LiveSubnetMetrics;
   390→  taoUsd: number;
   391→  hardware?: SubnetHardwareProfile | null;
   392→  taoChange24h?: number;
   393→  /** Blocks since the subnet was registered (for the maturity bonus). */
   394→  liveAgeBlocks?: number | null;
   395→  /** User-configurable cost lines (Profitability Engine settings). When
   396→   *  absent, market defaults apply — same engine, same shape. */
   397→  costs?: {
   398→    hardwareMode?: "rent" | "owned";
   399→    electricityUsdPerKwh?: number;
   400→    storageMonthlyUsd?: number;
   401→    /** 0/undefined = auto infra from the work-type classifier. */
   402→    infraMonthlyUsd?: number;
   403→    otherOpexMonthlyUsd?: number;
   404→    includeBurnAmortization?: boolean;
   405→    amortizeBurnMonths?: number;
   406→  };
   407→}): MinerLedger {
   408→  const { live, taoUsd } = inputs;
   409→  const usd = taoUsd || 0;
   410→
   411→  // --- Top line: newcomer-adjusted per-miner revenue ----------------------
   412→  // Miners get 41% of the subnet's emission (docs constant); the chain-
   413→  // measured minerEmissionTaoPerDay already reflects the realized split.
   414→  //
   415→  // The decision a miner makes is "if I register TODAY, what does a
   416→  // mid-pack seat pay?" — so the ledger starts from the per-REGISTERED
   417→  // slot (zero-earning seats included: on many subnets 250+ UIDs are
   418→  // registered and only a handful earn), then applies the median/mean
   419→  // spread share so a whale UID can't inflate the estimate.
   420→  // perEarningDailyTao (mean over rewarded UIDs) is kept as the upside
   421→  // reference — what current earners make.
   422→  const rewarded = Math.max(live.rewardedMiners ?? 0, 0);
   423→  const registered = Math.max(live.minersCount, 1);
   424→  const minerEmissionTaoPerDay = live.minerEmissionTaoPerDay ?? 0;
   425→  const perRegisteredDailyTao =
   426→    minerEmissionTaoPerDay > 0 ? minerEmissionTaoPerDay / registered : 0;
   427→  const perEarningDailyTao =
   428→    rewarded > 0 && minerEmissionTaoPerDay > 0
   429→      ? minerEmissionTaoPerDay / rewarded
   430→      : minerEmissionTaoPerDay > 0
   431→        ? perRegisteredDailyTao // vec unavailable — no zero-slot discount available
   432→        : 0;
   433→  const rewardMedianShare = live.incentiveMedianShare ?? 1;
   434→  // Newcomer-adjusted mid-pack revenue:
   435→  //  - reward vec available  → perRegistered × medianShare
   436→  //  - reward vec unavailable→ perRegistered × 0.6 (conservative)
   437→  const expectedDailyTao =
   438→    rewarded > 0 && minerEmissionTaoPerDay > 0
   439→      ? perRegisteredDailyTao * rewardMedianShare
   440→      : minerEmissionTaoPerDay > 0
   441→        ? perRegisteredDailyTao * 0.6
   442→        : 0;
   443→  const grossMonthlyUsd = Math.round(expectedDailyTao * 30 * usd);
   444→  const dailyAlpha = live.movingPrice > 0 ? expectedDailyTao / live.movingPrice : 0;
   445→
   446→  // --- Hardware profile (work type → GPU cost) ---------------------------
   447→  const fallbackMonthly = grossMonthlyUsd > 0 ? grossMonthlyUsd : undefined;
   448→  const hardware =
   449→    inputs.hardware ??
   450→    classifySubnetHardware(live.name, live.identityDescription, {
   451→      fallbackMonthlyUsd: fallbackMonthly,
   452→    });
   453→  // --- Cost stack (Profitability Engine lines) ----------------------------
   454→  // gpuCount > 1 (scraped, e.g. "8x H200") scales the GPU line: the requirement
   455→  // is a multi-GPU server, not one card. Owned mode scales the power draw.
   456→  const costs = inputs.costs ?? {};
   457→  const gpuCount = hardware.gpuCount && hardware.gpuCount > 1 ? hardware.gpuCount : 1;
   458→  const gpuCost =
   459→    costs.hardwareMode === "owned"
   460→      ? electricityMonthlyUsd(
   461→          hardware.tier.powerWatts * gpuCount,
   462→          costs.electricityUsdPerKwh ?? 0.12
   463→        )
   464→      : hardware.tier.monthlyRentUsd * gpuCount;
   465→  const infraCost =
   466→    costs.infraMonthlyUsd != null && costs.infraMonthlyUsd > 0
   467→      ? costs.infraMonthlyUsd
   468→      : Math.max(hardware.monthlyCostUsd - hardware.tier.monthlyRentUsd * gpuCount, INFRA_BASE_USD);
   469→  const storageCost = costs.storageMonthlyUsd ?? 0;
   470→  const otherOpex = costs.otherOpexMonthlyUsd ?? 0;
   471→  const burnUsd =
   472→    live.burnCostTao != null && costs.includeBurnAmortization
   473→      ? Math.max(live.burnCostTao, 0) * usd
   474→      : 0;
   475→  const amortizedBurn = burnUsd / Math.max(costs.amortizeBurnMonths ?? 3, 1);
   476→  const totalCosts = gpuCost + infraCost + storageCost + otherOpex + amortizedBurn;
   477→  // Zero-revenue subnets are honest losers: the cost stack still applies.
   478→  const netMonthlyUsd = grossMonthlyUsd > 0
   479→    ? Math.round(grossMonthlyUsd - totalCosts)
   480→    : -Math.round(totalCosts);
   481→
   482→  // --- Alpha economics ----------------------------------------------------
   483→  const alphaPriceTao = live.movingPrice;
   484→  const alphaPriceUsd = alphaPriceTao * usd;
   485→  const liquidityTao = live.subnetTao; // TAO side of the pool = exit liquidity
   486→  const slippagePct =
   487→    live.alphaOut > 0 && dailyAlpha > 0
   488→      ? Math.round((dailyAlpha / live.alphaOut) * 10000) / 100
   489→      : null;
   490→
   491→  // --- Seat safety inputs -------------------------------------------------
   492→  const maxUids = live.maxUids ?? 0;
   493→  const freeSlots = maxUids > 0 ? Math.max(0, maxUids - live.minersCount) : null;
   494→  const totalSlots = maxUids > 0 ? maxUids : null;
   495→  const rewardedRatio = live.rewardedMiners != null && live.rewardedMiners > 0
   496→    ? Math.min(1, live.rewardedMiners / registered)
   497→    : live.rewardedMiners === 0
   498→      ? 0
   499→      : null;
   500→  const top10 = live.top10IncentiveShare;
   501→
   502→  // Newcomer ramp: bonds build via EMA (α=0.1), and on concentrated subnets
   503→  // a fresh UID starts at zero. Research: 1-3 months of tuning is typical.
   504→  const rampWeeks =
   505→    3 +
   506→    (rewardedRatio != null ? (1 - rewardedRatio) * 6 : 3) +
   507→    (top10 != null && top10 > 0.7 ? 3 : 0) +
   508→    (freeSlots != null && freeSlots === 0 ? 2 : 0);
   509→
   510→  // --- Pillar 1: Net ROI --------------------------------------------------
   511→  // $0 → 10, +$100 → 54, +$500 → 69, +$1k → 76, +$5k → 92; losses slide to 2.
   512→  const netRoi =
   513→    netMonthlyUsd >= 0
   514→      ? clampScore(10 + 22 * Math.log10(netMonthlyUsd + 1), 2, 97)
   515→      : clampScore(10 + netMonthlyUsd / 100, 2, 10);
   516→
   517→  // --- Pillar 2: Seat safety ----------------------------------------------
   518→  const roomRatio = maxUids > 0 ? freeSlots! / maxUids : 0.5;
   519→  const scoreRoom = clampScore(10 + roomRatio * 45, 10, 55); // full subnet → 10
   520→  const scoreSpread =
   521→    top10 != null ? clampScore(100 - (top10 - 0.2) * 100, 10, 95) : 50;
   522→  const scoreBurn =
   523→    live.burnCostTao != null
   524→      ? clampScore(95 - Math.log10(live.burnCostTao + 1) * 25, 30, 95)
   525→      : 55;
   526→  const scoreImmunity =
   527→    live.immunityBlocks != null
   528→      ? clampScore(30 + (live.immunityBlocks / 7200) * 50, 30, 80)
   529→      : 45;
   530→  const seatSafety = !live.emissionEnabled
   531→    ? 12
   532→    : clampScore(
   533→        scoreRoom * 0.35 + scoreSpread * 0.35 + scoreBurn * 0.15 + scoreImmunity * 0.15,
   534→        5,
   535→        96
   536→      );
   537→
   538→  // --- Pillar 3: Alpha economics -------------------------------------------
   539→  const change = live.alphaPriceChange24h;
   540→  const scoreTrend = change != null ? clampScore(50 + change * 1.5, 10, 95) : 50;
   541→  const scoreLiq =
   542→    liquidityTao > 0
   543→      ? clampScore(20 + 19 * Math.log10(liquidityTao / 500 + 1), 10, 95)
   544→      : 10;
   545→  const scoreSlip =
   546→    slippagePct != null ? clampScore(95 - slippagePct * 1200, 15, 95) : 50;
   547→  const alphaEconomics = usd <= 0
   548→    ? 50
   549→    : clampScore(scoreTrend * 0.4 + scoreLiq * 0.3 + scoreSlip * 0.3, 5, 96);
   550→
   551→  // --- Pillar 4: Earning reality -------------------------------------------
   552→  const scoreRewarded =
   553→    rewardedRatio != null ? clampScore(15 + rewardedRatio * 70, 15, 85) : 50;
   554→  const scoreConc =
   555→    top10 != null ? clampScore(100 - (top10 - 0.2) * 90, 15, 92) : 50;
   556→  const scoreRamp = clampScore(90 - rampWeeks * 5, 20, 90);
   557→  const earningReality = !live.emissionEnabled
   558→    ? 8
   559→    : clampScore(scoreRewarded * 0.4 + scoreConc * 0.3 + scoreRamp * 0.3, 5, 96);
   560→
   561→  // --- Pillar 5: Fit & feasibility ------------------------------------------
   562→  let fitFeasibility = 25; // emission disabled base
   563→  if (live.emissionEnabled) {
   564→    fitFeasibility = 40;
   565→    const validators = live.validatorsCount ?? 0;
   566→    fitFeasibility += clampScore(validators * 2.5, 0, 20); // validator coverage
   567→    if (live.identityGithub) fitFeasibility += 12;
   568→    if (live.identityDescription) fitFeasibility += 6;
   569→    if (live.registeredAt != null && inputs.liveAgeBlocks != null) {
   570→      const ageDays = inputs.liveAgeBlocks / 7200;
   571→      fitFeasibility += clampScore(Math.min(ageDays / 30, 1) * 12, 0, 12); // maturity
   572→    }
   573→    if (live.immunityBlocks != null && live.immunityBlocks < 4096) fitFeasibility -= 5;
   574→  }
   575→  fitFeasibility = clampScore(fitFeasibility, 5, 92);
   576→
   577→  const components: ScoreComponents = {
   578→    net_roi: Math.round(netRoi * 10) / 10,
   579→    seat_safety: Math.round(seatSafety * 10) / 10,
   580→    alpha_economics: Math.round(alphaEconomics * 10) / 10,
   581→    earning_reality: Math.round(earningReality * 10) / 10,
   582→    fit_feasibility: Math.round(fitFeasibility * 10) / 10,
   583→  };
   584→
   585→  const diag: MinerLedgerDiagnostics = {
   586→    perEarningDailyTao: Math.round(perEarningDailyTao * 10000) / 10000,
   587→    expectedDailyTao: Math.round(expectedDailyTao * 10000) / 10000,
   588→    rewardMedianShare: live.incentiveMedianShare,
   589→    perRegisteredDailyTao: Math.round(perRegisteredDailyTao * 10000) / 10000,
   590→    grossMonthlyUsd,
   591→    alphaPriceTao,
   592→    alphaPriceUsd: Math.round(alphaPriceUsd * 1000) / 1000,
   593→    alphaChange24h: change ?? null,
   594→    liquidityTao: Math.round(liquidityTao),
   595→    slippagePct,
   596→    category: hardware.category,
   597→    gpuLabel: hardware.tier.label,
   598→    recommendedGpu: hardware.recommendedGpu,
   599→    minVramGb: hardware.minVramGb,
   600→    gpuCostMonthlyUsd: Math.round(gpuCost),
   601→    gpuCount: hardware.gpuCount ?? null,
   602→    hosting: hardware.hosting ?? null,
   603→    requirementsSource: hardware.requirementsSource ?? null,
   604→    infraCostMonthlyUsd: Math.round(infraCost),
   605→    storageCostMonthlyUsd: Math.round(storageCost),
   606→    otherOpexMonthlyUsd: Math.round(otherOpex),
   607→    amortizedBurnUsd: Math.round(amortizedBurn),
   608→    totalCostMonthlyUsd: Math.round(totalCosts),
   609→    gpuPowerWatts: hardware.tier.powerWatts * gpuCount,
   610→    netMonthlyUsd,
   611→    netDailyTao:
   612→      usd > 0 ? Math.round((netMonthlyUsd / 30 / usd) * 10000) / 10000 : 0,
   613→    freeSlots,
   614→    totalSlots,
   615→    rewardedRatio: rewardedRatio != null ? Math.round(rewardedRatio * 1000) / 1000 : null,
   616→    top10IncentiveShare: top10,
   617→    burnCostTao: live.burnCostTao,
   618→    immunityBlocks: live.immunityBlocks,
   619→    rampWeeks: Math.round(rampWeeks * 10) / 10,
   620→    emissionEnabled: live.emissionEnabled,
   621→    hardwareClassified: hardware.classified,
   622→  };
   623→
   624→  return { components, factors: deriveFactors(components), diag };
   625→}

   627→// ---------------------------------------------------------------------------
   628→// Chance to earn · month 1 — a newcomer's honest odds of earning ANY reward
   629→// during their first month of continuous, good-uptime mining. Uptime alone
   630→// keeps the seat; the share of slots that actually earn decides whether the
   631→// wallet moves. Dominant factor: rewardedRatio (share of registered slots
   632→// that earned last epoch), discounted for whale dominance, full subnets
   633→// (must displace an incumbent) and long bond-EMA ramps.
   634→// ---------------------------------------------------------------------------
   635→
   636→export type EarnChanceLevel = "high" | "medium" | "low" | "none";
   637→
   638→export interface EarnChance {
   639→  level: EarnChanceLevel;
   640→  /** 0–100 modeled chance that a newcomer earns something in month 1. */
   641→  pct: number;
   642→  /** Human explanation of the dominant factors. */
   643→  note: string;
   644→}
   645→
   646→export function computeEarnChance(inputs: {
   647→  /** Share of registered slots that earned reward last epoch (0..1). */
   648→  rewardedRatio: number | null;
   649→  /** Free UID slots (null = unknown). 0 means you must displace an incumbent. */
   650→  freeSlots: number | null;
   651→  /** Top-10% share of epoch incentive (0..1) — whale dominance. */
   652→  top10IncentiveShare: number | null;
   653→  /** Newcomer bond-EMA ramp estimate in weeks. */
   654→  rampWeeks: number | null;
   655→}): EarnChance {
   656→  const { rewardedRatio, freeSlots, top10IncentiveShare, rampWeeks } = inputs;
   657→
   658→  // Base: the fraction of slots that actually earn — the single strongest
   659→  // predictor of whether a newcomer earns anything at all.
   660→  let pct = rewardedRatio != null ? rewardedRatio * 100 : 0;
   661→  const factors: string[] = [];
   662→
   663→  // Whale dominance: when the top 10% of UIDs take nearly everything, the
   664→  // rewarded tail is scraps — being "rewarded" still pays almost nothing.
   665→  if (top10IncentiveShare != null && top10IncentiveShare > 0.5) {
   666→    pct *= 1 - Math.min(1, top10IncentiveShare - 0.5) * 0.6;
   667→    factors.push(
   668→      `top-10% of UIDs take ${Math.round(top10IncentiveShare * 100)}% of rewards`
   669→    );
   670→  }
   671→
   672→  // Full subnet: no free seats — you must outperform an incumbent to get in,
   673→  // and you start from zero bonds against their established ones.
   674→  if (freeSlots === 0) {
   675→    pct *= 0.7;
   676→    factors.push("subnet full — must displace an incumbent");
   677→  }
   678→
   679→  // Long ramp: month 1 is mostly bond-building; long ramps leave little of it.
   680→  if (rampWeeks != null && rampWeeks > 8) {
   681→    pct *= 0.8;
   682→    factors.push(`~${Math.round(rampWeeks)} wk ramp to full bonds`);
   683→  }
   684→
   685→  pct = Math.round(Math.max(0, Math.min(100, pct)));
   686→  const level: EarnChanceLevel =
   687→    rewardedRatio != null && rewardedRatio <= 0
   688→      ? "none"
   689→      : pct >= 50
   690→        ? "high"
   691→        : pct >= 20
   692→          ? "medium"
   693→          : pct > 0
   694→            ? "low"
   695→            : "none";
   696→
   697→  const base = rewardedRatio != null
   698→    ? `${rewardedRatio < 0.1 && rewardedRatio > 0 ? (rewardedRatio * 100).toFixed(1) : Math.round(rewardedRatio * 100)}% of slots earned last epoch`
   699→    : "reward spread unknown";
   700→  const note =
   701→    factors.length > 0 ? `${base} · ${factors.join(" · ")}` : base;
   702→
   703→  return { level, pct, note };
   704→}

   706→// ---------------------------------------------------------------------------
   707→// Seat Chance — "can I actually GET IN?" (vs EarnChance's "will I earn?").
   708→//
   709→// Bittensor registration mechanics:
   710→//   • Subnets have a UID cap (maxUids). Free slots → register via burn or PoW.
   711→//   • FULL subnet → registration STILL works: a burn registration pays the
   712→//     dynamic recycle cost and the chain replaces the worst-performing
   713→//     NON-IMMUNE uid. PoW registrations compete for per-block slots the same
   714→//     way. You are only locked out if every uid is immune (rare — immunity
   715→//     covers ~14h by default) or registration is disabled on the subnet.
   716→//   • immunityBlocks = how long YOUR new seat cannot be deregistered — the
   717→//     runway to start earning before becoming replaceable yourself.
   718→// ---------------------------------------------------------------------------
   719→
   720→/**
   721→ * Burn-quote display precision — the chain's burn hyperparam can sit at the
   722→ * floor (e.g. Chutes ~0.0005 TAO); 3 decimals would show "0.001" — a 2x
   723→ * misquote shown to a user deciding whether to burn. <0.01 → 4 decimals,
   724→ * <1 → 3, else 2.
   725→ */
   726→export function formatBurnTao(n: number): string {
   727→  if (n < 0.01) return n.toFixed(4);
   728→  if (n < 1) return n.toFixed(3);
   729→  return n.toFixed(2);
   730→}

   732→export type SeatVerdict = "open" | "burn-entry" | "waitlist" | "unknown";
   733→
   734→export interface SeatChance {
   735→  verdict: SeatVerdict;
   736→  /** Registration is possible right now (burn path or free slot). */
   737→  canRegister: boolean;
   738→  slotsFree: number | null;
   739→  totalSlots: number | null;
   740→  /** 0–100 — how full the uid space is. */
   741→  fillPct: number | null;
   742→  burnCostTao: number | null;
   743→  /** Protection window for a NEW seat, in hours (blocks × 12s). */
   744→  immunityHours: number | null;
   745→  /** Share of uids that earned NOTHING last epoch — the replaceable bottom
   746→   *  a newcomer displaces from. High = churn churns in your favor. */
   747→  replaceableShare: number | null;
   748→  headline: string;
   749→  detail: string;
   750→}
   751→
   752→export function assessSeatChance(inputs: {
   753→  minersCount: number | null;
   754→  maxUids: number | null;
   755→  /** Already-computed free slots (skip when you only have counts). */
   756→  freeSlots?: number | null;
   757→  burnCostTao: number | null;
   758→  immunityBlocks: number | null;
   759→  /** Rewarded uids (either give the ratio directly or the absolute count). */
   760→  rewardedRatio?: number | null;
   761→  rewardedMiners?: number | null;
   762→}): SeatChance {
   763→  const { maxUids, burnCostTao, immunityBlocks } = inputs;
   764→  const minersCount = inputs.minersCount ?? 0;
   765→  const slotsFree =
   766→    inputs.freeSlots != null
   767→      ? Math.max(0, inputs.freeSlots)
   768→      : maxUids != null && maxUids > 0
   769→        ? Math.max(0, maxUids - minersCount)
   770→        : null;
   771→  const fillPct =
   772→    maxUids != null && maxUids > 0
   773→      ? Math.min(100, Math.round((minersCount / maxUids) * 100))
   774→      : null;
   775→  let replaceableShare: number | null = null;
   776→  if (inputs.rewardedRatio != null) {
   777→    replaceableShare = Math.max(0, Math.min(1, 1 - inputs.rewardedRatio));
   778→  } else if (inputs.rewardedMiners != null && minersCount > 0) {
   779→    replaceableShare = Math.max(0, Math.min(1, 1 - inputs.rewardedMiners / minersCount));
   780→  }
   781→  const immunityHours =
   782→    immunityBlocks != null && immunityBlocks > 0
   783→      ? Math.round(((immunityBlocks * 12) / 3600) * 10) / 10
   784→      : null;
   785→  const immunityNote = immunityHours != null
   786→    ? `a new seat is immune for ~${immunityHours}h — your runway to start earning before you're replaceable`
   787→    : null;
   788→  const deadNote = replaceableShare != null && replaceableShare > 0.05
   789→    ? `${Math.round(replaceableShare * 100)}% of uids earned nothing last epoch — a deep replaceable bottom`
   790→    : replaceableShare != null
   791→      ? "almost every seat earned last epoch — displacement targets are scarce and strong"
   792→      : null;
   793→
   794→  const unknown: SeatChance = {
   795→    verdict: "unknown",
   796→    canRegister: true,
   797→    slotsFree,
   798→    totalSlots: maxUids ?? null,
   799→    fillPct,
   800→    burnCostTao,
   801→    immunityHours,
   802→    replaceableShare,
   803→    headline: "slot data unavailable",
   804→    detail:
   805→      "UID capacity was not returned by the chain snapshot this pass — refresh the network data. " +
   806→      (burnCostTao != null ? `Burn registration quote: ~${formatBurnTao(burnCostTao)} TAO.` : ""),
   807→  };
   808→
   809→  if (slotsFree == null || maxUids == null || maxUids <= 0) return unknown;
   810→
   811→  // --- Open: free slots exist ---------------------------------------------
   812→  if (slotsFree > 0) {
   813→    return {
   814→      verdict: "open",
   815→      canRegister: true,
   816→      slotsFree,
   817→      totalSlots: maxUids,
   818→      fillPct,
   819→      burnCostTao,
   820→      immunityHours,
   821→      replaceableShare,
   822→      headline: `${slotsFree} of ${maxUids} slots free`,
   823→      detail:
   824→        `This subnet is NOT full — ${slotsFree} uid${slotsFree === 1 ? "" : "s"} open (${fillPct}% filled). ` +
   825→        `Register now via burn${burnCostTao != null ? ` (~${formatBurnTao(burnCostTao)} TAO)` : ""} or PoW. ` +
   826→        (immunityNote ? `${immunityNote}. ` : "") +
   827→        (deadNote ? `${deadNote}.` : ""),
   828→    };
   829→  }
   830→
   831→  // --- Full: burn-entry is the deterministic path ---------------------------
   832→  if (burnCostTao != null && burnCostTao > 0) {
   833→    return {
   834→      verdict: "burn-entry",
   835→      canRegister: true,
   836→      slotsFree: 0,
   837→      totalSlots: maxUids,
   838→      fillPct,
   839→      burnCostTao,
   840→      immunityHours,
   841→      replaceableShare,
   842→      headline: `full (${minersCount}/${maxUids}) — burn-entry ~${formatBurnTao(burnCostTao)} TAO`,
   843→      detail:
   844→        `Every uid is taken, but registration still works: paying the burn (~${formatBurnTao(burnCostTao)} TAO, floats with demand) ` +
   845→        `immediately replaces the WORST-performing non-immune uid. ` +
   846→        (deadNote ? `${deadNote}. ` : "The bottom of this cohort is well-defended — displacement may take several attempts. ") +
   847→        (immunityNote ? `${immunityNote}.` : ""),
   848→    };
   849→  }
   850→
   851→  // --- Full, no burn quote: PoW/competitive path ----------------------------
   852→  return {
   853→    verdict: "waitlist",
   854→    canRegister: true,
   855→    slotsFree: 0,
   856→    totalSlots: maxUids,
   857→    fillPct,
   858→    burnCostTao: null,
   859→    immunityHours,
   860→    replaceableShare,
   861→    headline: `full (${minersCount}/${maxUids}) — entry is competitive`,
   862→    detail:
   863→      `No free slots and no burn quote from this snapshot. Entry means winning a PoW registration ` +
   864→      `and replacing a zero-earning uid — expect per-block competition. ` +
   865→      (deadNote ? `${deadNote}. ` : "") +
   866→      (immunityNote ? `${immunityNote}.` : ""),
   867→  };
   868→}
   869→
