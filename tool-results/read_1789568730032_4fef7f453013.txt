     2→     2→// Miner's Ledger scoring v2 — models how an experienced GPU miner actually
     3→     3→// picks a subnet. Research-backed (Bittensor docs, taostats miner guide,
     4→     4→// miner-economics deep dives, dTAO analyst frameworks):
     5→     5→//
     6→     6→//   Q1 FIT        — can my hardware run this? (subnet type → GPU tier/cost)
     7→     7→//   Q2 TOP LINE   — what does it pay? (emission × 41% miner share ÷ EARNING
     8→     8→//                   miners, priced in alpha → TAO → USD)
     9→     9→//   Q3 BOTTOM LINE— what do I keep? (minus GPU rental + infra → net monthly)
    10→    10→//   Q4 SEAT SAFETY— can I keep the slot? (saturation, reward concentration,
    11→    11→//                   burn cost, immunity runway)
    12→    12→//   Q5 HOLD VALUE — will revenue persist? (alpha price trend, pool liquidity,
    13→    13→//                   sell-pressure slippage, owner identity, maturity)
    14→    14→//
    15→    15→// Key constants from the chain: per-tempo split is 18% owner / 41% miners /
    16→    16→// 41% validators+stakers. "Top 10-20% of miners take 60-80% of emissions;
    17→    17→// median miners hover near breakeven" — so the ledger scores NET numbers,
    18→    18→// not gross, and treats reward concentration as a first-class risk.

    75→    75→export function deriveFactors(c: ScoreComponents): OpportunityFactor[] {
    76→    76→  return (Object.keys(c) as (keyof ScoreComponents)[]).map((k) => {
    77→    77→    const raw = c[k];
    78→    78→    const weight = SCORE_WEIGHTS[k];
    79→    79→    return {
    80→    80→      name: FACTOR_LABELS[k],
    81→    81→      value: Math.round(raw * weight * 100) / 100,
    82→    82→      weight,
    83→    83→      raw,
    84→    84→      impact: raw >= 60 ? "positive" : raw <= 40 ? "negative" : "neutral",
    85→    85→      description: FACTOR_DESC[k],
    86→    86→    };
    87→    87→  });
    88→    88→}
    89→    89→
    90→    90→export function totalScore(c: ScoreComponents): number {
    91→    91→  return Math.round(
    92→    92→    (Object.keys(c) as (keyof ScoreComponents)[]).reduce(
    93→    93→      (acc, k) => acc + c[k] * SCORE_WEIGHTS[k],
    94→    94→      0
    95→    95→    ) * 10
    96→    96→  ) / 10;
    97→    97→}
    98→    98→
    99→    99→export function riskLevel(score: number): "low" | "medium" | "high" {
   100→   100→  if (score >= 60) return "low";
   101→   101→  if (score >= 40) return "medium";
   102→   102→  return "high";
   103→   103→}

   134→   134→/**
   135→   135→ * Resolve a GPU tier from a model string found in a repo README
   136→   136→ * (github-scraper parseGpuRequirement output). Ground truth from the
   137→   137→ * subnet's own docs beats keyword classification.
   138→   138→ */
   139→   139→export function resolveGpuTierFromModel(model: string): GpuTier | null {
   140→   140→  const m = model.toLowerCase();
   141→   141→  if (/\bb300\b/.test(m)) return GPU_TIERS.b200; // same Blackwell HGX class
   142→   142→  if (/\bb200\b/.test(m)) return GPU_TIERS.b200;
   143→   143→  if (/\bh200\b/.test(m)) return GPU_TIERS.h200;
   144→   144→  if (/\bh100\b/.test(m)) return GPU_TIERS.h100;
   145→   145→  if (/\b(?:rtx\s*)?pro\s*6000\b/.test(m) || /\brtx\s*6000\s*ada\b/.test(m)) return GPU_TIERS.pro6000;
   146→   146→  if (/\ba100\b/.test(m)) return GPU_TIERS.a100;
   147→   147→  if (/\bl40s?\b/.test(m)) return GPU_TIERS.a6000; // 48GB Ada-class
   148→   148→  if (/\ba40\b/.test(m)) return GPU_TIERS.a6000;
   149→   149→  if (/\ba6000\b/.test(m)) return GPU_TIERS.a6000;
   150→   150→  if (/\ba5000\b/.test(m)) return GPU_TIERS.vram16; // 24GB but older — mid tier
   151→   151→  if (/\ba10\b/.test(m)) return GPU_TIERS.vram16;
   152→   152→  if (/\bt4\b/.test(m)) return GPU_TIERS.entry;
   153→   153→  if (/\b(?:rtx\s*)?(?:4090|3090|5090)\b/.test(m)) return GPU_TIERS.consumer24;
   154→   154→  if (/\bmi300x\b/.test(m)) return GPU_TIERS.h200; // 192GB HBM3 accelerator class
   155→   155→  return null;
   156→   156→}

   187→   187→/** Deterministic GPU tier from the reward stream — fallback for subnets
   188→   188→ *  with no registered identity. What per-miner revenue can plausibly fund. */
   189→   189→export function estimateGpuTierFromRevenue(monthlyUsdPerMiner: number): GpuTier {
   190→   190→  if (monthlyUsdPerMiner >= 1200) return GPU_TIERS.h200;
   191→   191→  if (monthlyUsdPerMiner >= 600) return GPU_TIERS.h100;
   192→   192→  if (monthlyUsdPerMiner >= 300) return GPU_TIERS.a100;
   193→   193→  if (monthlyUsdPerMiner >= 140) return GPU_TIERS.a6000;
   194→   194→  if (monthlyUsdPerMiner >= 60) return GPU_TIERS.consumer24;
   195→   195→  if (monthlyUsdPerMiner >= 25) return GPU_TIERS.vram16;
   196→   196→  if (monthlyUsdPerMiner >= 8) return GPU_TIERS.entry;
   197→   197→  return GPU_TIERS.cpu;
   198→   198→}

   224→   224→export function classifySubnetHardware(
   225→   225→  name: string | null | undefined,
   226→   226→  description: string | null | undefined,
   227→   227→  fallbacks?: {
   228→   228→    /** Curated category text (e.g. "Inference") when the chain identity is absent. */
   229→   229→    fallbackCategory?: string;
   230→   230→    /** Curated/user GPU figures win when the classifier has no keyword match. */
   231→   231→    fallbackVramGb?: number;
   232→   232→    fallbackGpu?: string;
   233→   233→    /** Per-earning-miner monthly USD — revenue-based fallback tier. */
   234→   234→    fallbackMonthlyUsd?: number;
   235→   235→    /** GitHub-scraped requirements — GROUND TRUTH, wins over everything. */
   236→   236→    scraped?: {
   237→   237→      recommendedGpu?: string | null;
   238→   238→      gpuCount?: number | null;
   239→   239→      minVramGb?: number | null;
   240→   240→      hosting?: HostingRequirements | null;
   241→   241→      requirementsSource?: string | null;
   242→   242→    } | null;
   243→   243→  }
   244→   244→): SubnetHardwareProfile {
   245→   245→  // --- Base classification: work-type keywords → tier (existing logic) ----
   246→   246→  const base = (() => {
   247→   247→    const text = `${name ?? ""} ${description ?? ""} ${fallbacks?.fallbackCategory ?? ""}`;
   248→   248→    for (const rule of CATEGORY_RULES) {
   249→   249→      if (rule.keywords.test(text)) {
   250→   250→        return {
   251→   251→          profile: {
   252→   252→            category: rule.category,
   253→   253→            tier: rule.tier,
   254→   254→            minVramGb: rule.tier.minVramGb,
   255→   255→            recommendedGpu: rule.tier.recommendedGpu,
   256→   256→            monthlyCostUsd: rule.tier.monthlyRentUsd + (rule.infra ?? INFRA_BASE_USD),
   257→   257→            isGpuWorkload: rule.tier.minVramGb > 0,
   258→   258→            classified: true,
   259→   259→          } as SubnetHardwareProfile,
   260→   260→          infraUsd: rule.infra ?? INFRA_BASE_USD,
   261→   261→        };
   262→   262→      }
   263→   263→    }
   264→   264→    // No keyword match — fall back to curated VRAM data, else revenue tier.
   265→   265→    if (fallbacks?.fallbackVramGb && fallbacks.fallbackVramGb > 0) {
   266→   266→      const tier = Object.values(GPU_TIERS).find(
   267→   267→        (t) => t.minVramGb === fallbacks.fallbackVramGb
   268→   268→      ) ?? GPU_TIERS.consumer24;
   269→   269→      return {
   270→   270→        profile: {
   271→   271→          category: fallbacks.fallbackCategory || "Unclassified workload",
   272→   272→          tier,
   273→   273→          minVramGb: tier.minVramGb,
   274→   274→          recommendedGpu: fallbacks.fallbackGpu ?? tier.recommendedGpu,
   275→   275→          monthlyCostUsd: tier.monthlyRentUsd + INFRA_BASE_USD,
   276→   276→          isGpuWorkload: true,
   277→   277→          classified: false,
   278→   278→        } as SubnetHardwareProfile,
   279→   279→        infraUsd: INFRA_BASE_USD,
   280→   280→      };
   281→   281→    }
   282→   282→    const tier = estimateGpuTierFromRevenue(fallbacks?.fallbackMonthlyUsd ?? 0);
   283→   283→    return {
   284→   284→      profile: {
   285→   285→        category: fallbacks?.fallbackCategory || "Unclassified workload",
   286→   286→        tier,
   287→   287→        minVramGb: tier.minVramGb,
   288→   288→        recommendedGpu: tier.recommendedGpu,
   289→   289→        monthlyCostUsd: tier.monthlyRentUsd + INFRA_BASE_USD,
   290→   290→        isGpuWorkload: tier.minVramGb > 0,
   291→   291→        classified: false,
   292→   292→      } as SubnetHardwareProfile,
   293→   293→      infraUsd: INFRA_BASE_USD,
   294→   294→    };
   295→   295→  })();
   296→   296→
   297→   297→  // --- Scraped repo requirements layer on top: the subnet's own README ----
   298→   298→  // (or its miner repo) is ground truth for GPU model + hosting rules.
   299→   299→  // It overrides ONLY what it actually knows — a hosting-only README keeps
   300→   300→  // the work-type GPU tier; a GPU-only README keeps the classifier category.
   301→   301→  const scraped = fallbacks?.scraped;
   302→   302→  const scrapedTier = scraped?.recommendedGpu
   303→   303→    ? resolveGpuTierFromModel(scraped.recommendedGpu)
   304→   304→    : null;
   305→   305→  if (!scraped || (!scrapedTier && !scraped.hosting)) return base.profile;
   306→   306→
   307→   307→  const count = scraped.gpuCount && scraped.gpuCount > 1 ? scraped.gpuCount : 1;
   308→   308→  const tier = scrapedTier ?? base.profile.tier;
   309→   309→  // The scraper's raw string already carries the count prefix ("8x H200").
   310→   310→  const gpuLabel = scraped.recommendedGpu?.trim() ?? base.profile.recommendedGpu;
   311→   311→  return {
   312→   312→    ...base.profile,
   313→   313→    tier,
   314→   314→    minVramGb: tier.minVramGb,
   315→   315→    recommendedGpu: gpuLabel,
   316→   316→    monthlyCostUsd: tier.monthlyRentUsd * count + base.infraUsd,
   317→   317→    isGpuWorkload: tier.minVramGb > 0,
   318→   318→    gpuCount: count,
   319→   319→    hosting: scraped.hosting ?? null,
   320→   320→    requirementsSource: scraped.requirementsSource ?? null,
   321→   321→    sourceScraped: true,
   322→   322→  };
   323→   323→}

   383→   383→
   384→   384→/**
   385→   385→ * Score one subnet the way a miner reads it. Deterministic per snapshot —
   386→   386→ * the same inputs always produce the same score so ranks stay stable.
   387→   387→ */
   388→   388→export function scoreMinersLedger(inputs: {
   389→   389→  live: LiveSubnetMetrics;
   390→   390→  taoUsd: number;
   391→   391→  hardware?: SubnetHardwareProfile | null;
   392→   392→  taoChange24h?: number;
   393→   393→  /** Blocks since the subnet was registered (for the maturity bonus). */
   394→   394→  liveAgeBlocks?: number | null;
   395→   395→  /** User-configurable cost lines (Profitability Engine settings). When
   396→   396→   *  absent, market defaults apply — same engine, same shape. */
   397→   397→  costs?: {
   398→   398→    hardwareMode?: "rent" | "owned";
   399→   399→    electricityUsdPerKwh?: number;
   400→   400→    storageMonthlyUsd?: number;
   401→   401→    /** 0/undefined = auto infra from the work-type classifier. */
   402→   402→    infraMonthlyUsd?: number;
   403→   403→    otherOpexMonthlyUsd?: number;
   404→   404→    includeBurnAmortization?: boolean;
   405→   405→    amortizeBurnMonths?: number;
   406→   406→  };
   407→   407→}): MinerLedger {
   408→   408→  const { live, taoUsd } = inputs;
   409→   409→  const usd = taoUsd || 0;
   410→   410→
   411→   411→  // --- Top line: newcomer-adjusted per-miner revenue ----------------------
   412→   412→  // Miners get 41% of the subnet's emission (docs constant); the chain-
   413→   413→  // measured minerEmissionTaoPerDay already reflects the realized split.
   414→   414→  //
   415→   415→  // The decision a miner makes is "if I register TODAY, what does a
   416→   416→  // mid-pack seat pay?" — so the ledger starts from the per-REGISTERED
   417→   417→  // slot (zero-earning seats included: on many subnets 250+ UIDs are
   418→   418→  // registered and only a handful earn), then applies the median/mean
   419→   419→  // spread share so a whale UID can't inflate the estimate.
   420→   420→  // perEarningDailyTao (mean over rewarded UIDs) is kept as the upside
   421→   421→  // reference — what current earners make.
   422→   422→  const rewarded = Math.max(live.rewardedMiners ?? 0, 0);
   423→   423→  const registered = Math.max(live.minersCount, 1);
   424→   424→  const minerEmissionTaoPerDay = live.minerEmissionTaoPerDay ?? 0;
   425→   425→  const perRegisteredDailyTao =
   426→   426→    minerEmissionTaoPerDay > 0 ? minerEmissionTaoPerDay / registered : 0;
   427→   427→  const perEarningDailyTao =
   428→   428→    rewarded > 0 && minerEmissionTaoPerDay > 0
   429→   429→      ? minerEmissionTaoPerDay / rewarded
   430→   430→      : minerEmissionTaoPerDay > 0
   431→   431→        ? perRegisteredDailyTao // vec unavailable — no zero-slot discount available
   432→   432→        : 0;
   433→   433→  const rewardMedianShare = live.incentiveMedianShare ?? 1;
   434→   434→  // Newcomer-adjusted mid-pack revenue:
   435→   435→  //  - reward vec available  → perRegistered × medianShare
   436→   436→  //  - reward vec unavailable→ perRegistered × 0.6 (conservative)
   437→   437→  const expectedDailyTao =
   438→   438→    rewarded > 0 && minerEmissionTaoPerDay > 0
   439→   439→      ? perRegisteredDailyTao * rewardMedianShare
   440→   440→      : minerEmissionTaoPerDay > 0
   441→   441→        ? perRegisteredDailyTao * 0.6
   442→   442→        : 0;
   443→   443→  const grossMonthlyUsd = Math.round(expectedDailyTao * 30 * usd);
   444→   444→  const dailyAlpha = live.movingPrice > 0 ? expectedDailyTao / live.movingPrice : 0;
   445→   445→
   446→   446→  // --- Hardware profile (work type → GPU cost) ---------------------------
   447→   447→  const fallbackMonthly = grossMonthlyUsd > 0 ? grossMonthlyUsd : undefined;
   448→   448→  const hardware =
   449→   449→    inputs.hardware ??
   450→   450→    classifySubnetHardware(live.name, live.identityDescription, {
   451→   451→      fallbackMonthlyUsd: fallbackMonthly,
   452→   452→    });
   453→   453→  // --- Cost stack (Profitability Engine lines) ----------------------------
   454→   454→  // gpuCount > 1 (scraped, e.g. "8x H200") scales the GPU line: the requirement
   455→   455→  // is a multi-GPU server, not one card. Owned mode scales the power draw.
   456→   456→  const costs = inputs.costs ?? {};
   457→   457→  const gpuCount = hardware.gpuCount && hardware.gpuCount > 1 ? hardware.gpuCount : 1;
   458→   458→  const gpuCost =
   459→   459→    costs.hardwareMode === "owned"
   460→   460→      ? electricityMonthlyUsd(
   461→   461→          hardware.tier.powerWatts * gpuCount,
   462→   462→          costs.electricityUsdPerKwh ?? 0.12
   463→   463→        )
   464→   464→      : hardware.tier.monthlyRentUsd * gpuCount;
   465→   465→  const infraCost =
   466→   466→    costs.infraMonthlyUsd != null && costs.infraMonthlyUsd > 0
   467→   467→      ? costs.infraMonthlyUsd
   468→   468→      : Math.max(hardware.monthlyCostUsd - hardware.tier.monthlyRentUsd * gpuCount, INFRA_BASE_USD);
   469→   469→  const storageCost = costs.storageMonthlyUsd ?? 0;
   470→   470→  const otherOpex = costs.otherOpexMonthlyUsd ?? 0;
   471→   471→  const burnUsd =
   472→   472→    live.burnCostTao != null && costs.includeBurnAmortization
   473→   473→      ? Math.max(live.burnCostTao, 0) * usd
   474→   474→      : 0;
   475→   475→  const amortizedBurn = burnUsd / Math.max(costs.amortizeBurnMonths ?? 3, 1);
   476→   476→  const totalCosts = gpuCost + infraCost + storageCost + otherOpex + amortizedBurn;
   477→   477→  // Zero-revenue subnets are honest losers: the cost stack still applies.
   478→   478→  const netMonthlyUsd = grossMonthlyUsd > 0
   479→   479→    ? Math.round(grossMonthlyUsd - totalCosts)
   480→   480→    : -Math.round(totalCosts);
   481→   481→
   482→   482→  // --- Alpha economics ----------------------------------------------------
   483→   483→  const alphaPriceTao = live.movingPrice;
   484→   484→  const alphaPriceUsd = alphaPriceTao * usd;
   485→   485→  const liquidityTao = live.subnetTao; // TAO side of the pool = exit liquidity
   486→   486→  const slippagePct =
   487→   487→    live.alphaOut > 0 && dailyAlpha > 0
   488→   488→      ? Math.round((dailyAlpha / live.alphaOut) * 10000) / 100
   489→   489→      : null;
   490→   490→
   491→   491→  // --- Seat safety inputs -------------------------------------------------
   492→   492→  const maxUids = live.maxUids ?? 0;
   493→   493→  const freeSlots = maxUids > 0 ? Math.max(0, maxUids - live.minersCount) : null;
   494→   494→  const totalSlots = maxUids > 0 ? maxUids : null;
   495→   495→  const rewardedRatio = live.rewardedMiners != null && live.rewardedMiners > 0
   496→   496→    ? Math.min(1, live.rewardedMiners / registered)
   497→   497→    : live.rewardedMiners === 0
   498→   498→      ? 0
   499→   499→      : null;
   500→   500→  const top10 = live.top10IncentiveShare;
   501→   501→
   502→   502→  // Newcomer ramp: bonds build via EMA (α=0.1), and on concentrated subnets
   503→   503→  // a fresh UID starts at zero. Research: 1-3 months of tuning is typical.
   504→   504→  const rampWeeks =
   505→   505→    3 +
   506→   506→    (rewardedRatio != null ? (1 - rewardedRatio) * 6 : 3) +
   507→   507→    (top10 != null && top10 > 0.7 ? 3 : 0) +
   508→   508→    (freeSlots != null && freeSlots === 0 ? 2 : 0);
   509→   509→
   510→   510→  // --- Pillar 1: Net ROI --------------------------------------------------
   511→   511→  // $0 → 10, +$100 → 54, +$500 → 69, +$1k → 76, +$5k → 92; losses slide to 2.
   512→   512→  const netRoi =
   513→   513→    netMonthlyUsd >= 0
   514→   514→      ? clampScore(10 + 22 * Math.log10(netMonthlyUsd + 1), 2, 97)
   515→   515→      : clampScore(10 + netMonthlyUsd / 100, 2, 10);
   516→   516→
   517→   517→  // --- Pillar 2: Seat safety ----------------------------------------------
   518→   518→  const roomRatio = maxUids > 0 ? freeSlots! / maxUids : 0.5;
   519→   519→  const scoreRoom = clampScore(10 + roomRatio * 45, 10, 55); // full subnet → 10
   520→   520→  const scoreSpread =
   521→   521→    top10 != null ? clampScore(100 - (top10 - 0.2) * 100, 10, 95) : 50;
   522→   522→  const scoreBurn =
   523→   523→    live.burnCostTao != null
   524→   524→      ? clampScore(95 - Math.log10(live.burnCostTao + 1) * 25, 30, 95)
   525→   525→      : 55;
   526→   526→  const scoreImmunity =
   527→   527→    live.immunityBlocks != null
   528→   528→      ? clampScore(30 + (live.immunityBlocks / 7200) * 50, 30, 80)
   529→   529→      : 45;
   530→   530→  const seatSafety = !live.emissionEnabled
   531→   531→    ? 12
   532→   532→    : clampScore(
   533→   533→        scoreRoom * 0.35 + scoreSpread * 0.35 + scoreBurn * 0.15 + scoreImmunity * 0.15,
   534→   534→        5,
   535→   535→        96
   536→   536→      );
   537→   537→
   538→   538→  // --- Pillar 3: Alpha economics -------------------------------------------
   539→   539→  const change = live.alphaPriceChange24h;
   540→   540→  const scoreTrend = change != null ? clampScore(50 + change * 1.5, 10, 95) : 50;
   541→   541→  const scoreLiq =
   542→   542→    liquidityTao > 0
   543→   543→      ? clampScore(20 + 19 * Math.log10(liquidityTao / 500 + 1), 10, 95)
   544→   544→      : 10;
   545→   545→  const scoreSlip =
   546→   546→    slippagePct != null ? clampScore(95 - slippagePct * 1200, 15, 95) : 50;
   547→   547→  const alphaEconomics = usd <= 0
   548→   548→    ? 50
   549→   549→    : clampScore(scoreTrend * 0.4 + scoreLiq * 0.3 + scoreSlip * 0.3, 5, 96);
   550→   550→
   551→   551→  // --- Pillar 4: Earning reality -------------------------------------------
   552→   552→  const scoreRewarded =
   553→   553→    rewardedRatio != null ? clampScore(15 + rewardedRatio * 70, 15, 85) : 50;
   554→   554→  const scoreConc =
   555→   555→    top10 != null ? clampScore(100 - (top10 - 0.2) * 90, 15, 92) : 50;
   556→   556→  const scoreRamp = clampScore(90 - rampWeeks * 5, 20, 90);
   557→   557→  const earningReality = !live.emissionEnabled
   558→   558→    ? 8
   559→   559→    : clampScore(scoreRewarded * 0.4 + scoreConc * 0.3 + scoreRamp * 0.3, 5, 96);
   560→   560→
   561→   561→  // --- Pillar 5: Fit & feasibility ------------------------------------------
   562→   562→  let fitFeasibility = 25; // emission disabled base
   563→   563→  if (live.emissionEnabled) {
   564→   564→    fitFeasibility = 40;
   565→   565→    const validators = live.validatorsCount ?? 0;
   566→   566→    fitFeasibility += clampScore(validators * 2.5, 0, 20); // validator coverage
   567→   567→    if (live.identityGithub) fitFeasibility += 12;
   568→   568→    if (live.identityDescription) fitFeasibility += 6;
   569→   569→    if (live.registeredAt != null && inputs.liveAgeBlocks != null) {
   570→   570→      const ageDays = inputs.liveAgeBlocks / 7200;
   571→   571→      fitFeasibility += clampScore(Math.min(ageDays / 30, 1) * 12, 0, 12); // maturity
   572→   572→    }
   573→   573→    if (live.immunityBlocks != null && live.immunityBlocks < 4096) fitFeasibility -= 5;
   574→   574→  }
   575→   575→  fitFeasibility = clampScore(fitFeasibility, 5, 92);
   576→   576→
   577→   577→  const components: ScoreComponents = {
   578→   578→    net_roi: Math.round(netRoi * 10) / 10,
   579→   579→    seat_safety: Math.round(seatSafety * 10) / 10,
   580→   580→    alpha_economics: Math.round(alphaEconomics * 10) / 10,
   581→   581→    earning_reality: Math.round(earningReality * 10) / 10,
   582→   582→    fit_feasibility: Math.round(fitFeasibility * 10) / 10,
   583→   583→  };
   584→   584→
   585→   585→  const diag: MinerLedgerDiagnostics = {
   586→   586→    perEarningDailyTao: Math.round(perEarningDailyTao * 10000) / 10000,
   587→   587→    expectedDailyTao: Math.round(expectedDailyTao * 10000) / 10000,
   588→   588→    rewardMedianShare: live.incentiveMedianShare,
   589→   589→    perRegisteredDailyTao: Math.round(perRegisteredDailyTao * 10000) / 10000,
   590→   590→    grossMonthlyUsd,
   591→   591→    alphaPriceTao,
   592→   592→    alphaPriceUsd: Math.round(alphaPriceUsd * 1000) / 1000,
   593→   593→    alphaChange24h: change ?? null,
   594→   594→    liquidityTao: Math.round(liquidityTao),
   595→   595→    slippagePct,
   596→   596→    category: hardware.category,
   597→   597→    gpuLabel: hardware.tier.label,
   598→   598→    recommendedGpu: hardware.recommendedGpu,
   599→   599→    minVramGb: hardware.minVramGb,
   600→   600→    gpuCostMonthlyUsd: Math.round(gpuCost),
   601→   601→    gpuCount: hardware.gpuCount ?? null,
   602→   602→    hosting: hardware.hosting ?? null,
   603→   603→    requirementsSource: hardware.requirementsSource ?? null,
   604→   604→    infraCostMonthlyUsd: Math.round(infraCost),
   605→   605→    storageCostMonthlyUsd: Math.round(storageCost),
   606→   606→    otherOpexMonthlyUsd: Math.round(otherOpex),
   607→   607→    amortizedBurnUsd: Math.round(amortizedBurn),
   608→   608→    totalCostMonthlyUsd: Math.round(totalCosts),
   609→   609→    gpuPowerWatts: hardware.tier.powerWatts * gpuCount,
   610→   610→    netMonthlyUsd,
   611→   611→    netDailyTao:
   612→   612→      usd > 0 ? Math.round((netMonthlyUsd / 30 / usd) * 10000) / 10000 : 0,
   613→   613→    freeSlots,
   614→   614→    totalSlots,
   615→   615→    rewardedRatio: rewardedRatio != null ? Math.round(rewardedRatio * 1000) / 1000 : null,
   616→   616→    top10IncentiveShare: top10,
   617→   617→    burnCostTao: live.burnCostTao,
   618→   618→    immunityBlocks: live.immunityBlocks,
   619→   619→    rampWeeks: Math.round(rampWeeks * 10) / 10,
   620→   620→    emissionEnabled: live.emissionEnabled,
   621→   621→    hardwareClassified: hardware.classified,
   622→   622→  };
   623→   623→
   624→   624→  return { components, factors: deriveFactors(components), diag };
   625→   625→}

   627→   627→// ---------------------------------------------------------------------------
   628→   628→// Chance to earn · month 1 — a newcomer's honest odds of earning ANY reward
   629→   629→// during their first month of continuous, good-uptime mining. Uptime alone
   630→   630→// keeps the seat; the share of slots that actually earn decides whether the
   631→   631→// wallet moves. Dominant factor: rewardedRatio (share of registered slots
   632→   632→// that earned last epoch), discounted for whale dominance, full subnets
   633→   633→// (must displace an incumbent) and long bond-EMA ramps.
   634→   634→// ---------------------------------------------------------------------------
   635→   635→
   636→   636→export type EarnChanceLevel = "high" | "medium" | "low" | "none";
   637→   637→
   638→   638→export interface EarnChance {
   639→   639→  level: EarnChanceLevel;
   640→   640→  /** 0–100 modeled chance that a newcomer earns something in month 1. */
   641→   641→  pct: number;
   642→   642→  /** Human explanation of the dominant factors. */
   643→   643→  note: string;
   644→   644→}
   645→   645→
   646→   646→export function computeEarnChance(inputs: {
   647→   647→  /** Share of registered slots that earned reward last epoch (0..1). */
   648→   648→  rewardedRatio: number | null;
   649→   649→  /** Free UID slots (null = unknown). 0 means you must displace an incumbent. */
   650→   650→  freeSlots: number | null;
   651→   651→  /** Top-10% share of epoch incentive (0..1) — whale dominance. */
   652→   652→  top10IncentiveShare: number | null;
   653→   653→  /** Newcomer bond-EMA ramp estimate in weeks. */
   654→   654→  rampWeeks: number | null;
   655→   655→}): EarnChance {
   656→   656→  const { rewardedRatio, freeSlots, top10IncentiveShare, rampWeeks } = inputs;
   657→   657→
   658→   658→  // Base: the fraction of slots that actually earn — the single strongest
   659→   659→  // predictor of whether a newcomer earns anything at all.
   660→   660→  let pct = rewardedRatio != null ? rewardedRatio * 100 : 0;
   661→   661→  const factors: string[] = [];
   662→   662→
   663→   663→  // Whale dominance: when the top 10% of UIDs take nearly everything, the
   664→   664→  // rewarded tail is scraps — being "rewarded" still pays almost nothing.
   665→   665→  if (top10IncentiveShare != null && top10IncentiveShare > 0.5) {
   666→   666→    pct *= 1 - Math.min(1, top10IncentiveShare - 0.5) * 0.6;
   667→   667→    factors.push(
   668→   668→      `top-10% of UIDs take ${Math.round(top10IncentiveShare * 100)}% of rewards`
   669→   669→    );
   670→   670→  }
   671→   671→
   672→   672→  // Full subnet: no free seats — you must outperform an incumbent to get in,
   673→   673→  // and you start from zero bonds against their established ones.
   674→   674→  if (freeSlots === 0) {
   675→   675→    pct *= 0.7;
   676→   676→    factors.push("subnet full — must displace an incumbent");
   677→   677→  }
   678→   678→
   679→   679→  // Long ramp: month 1 is mostly bond-building; long ramps leave little of it.
   680→   680→  if (rampWeeks != null && rampWeeks > 8) {
   681→   681→    pct *= 0.8;
   682→   682→    factors.push(`~${Math.round(rampWeeks)} wk ramp to full bonds`);
   683→   683→  }
   684→   684→
   685→   685→  pct = Math.round(Math.max(0, Math.min(100, pct)));
   686→   686→  const level: EarnChanceLevel =
   687→   687→    rewardedRatio != null && rewardedRatio <= 0
   688→   688→      ? "none"
   689→   689→      : pct >= 50
   690→   690→        ? "high"
   691→   691→        : pct >= 20
   692→   692→          ? "medium"
   693→   693→          : pct > 0
   694→   694→            ? "low"
   695→   695→            : "none";
   696→   696→
   697→   697→  const base = rewardedRatio != null
   698→   698→    ? `${rewardedRatio < 0.1 && rewardedRatio > 0 ? (rewardedRatio * 100).toFixed(1) : Math.round(rewardedRatio * 100)}% of slots earned last epoch`
   699→   699→    : "reward spread unknown";
   700→   700→  const note =
   701→   701→    factors.length > 0 ? `${base} · ${factors.join(" · ")}` : base;
   702→   702→
   703→   703→  return { level, pct, note };
   704→   704→}
   705→   705→
   706→   706→// ---------------------------------------------------------------------------
   707→   707→// Seat Chance — "can I actually GET IN?" (vs EarnChance's "will I earn?").
   708→   708→//
   709→   709→// Bittensor registration mechanics:
   710→   710→//   • Subnets have a UID cap (maxUids). Free slots → register via burn or PoW.
   711→   711→//   • FULL subnet → registration STILL works: a burn registration pays the
   712→   712→//     dynamic recycle cost and the chain replaces the worst-performing
   713→   713→//     NON-IMMUNE uid. PoW registrations compete for per-block slots the same
   714→   714→//     way. You are only locked out if every uid is immune (rare — immunity
   715→   715→//     covers ~14h by default) or registration is disabled on the subnet.
   716→   716→//   • immunityBlocks = how long YOUR new seat cannot be deregistered — the
   717→   717→//     runway to start earning before becoming replaceable yourself.
   718→   718→// ---------------------------------------------------------------------------
   719→   719→
   720→   720→/**
   721→   721→ * Burn-quote display precision — the chain's burn hyperparam can sit at the
   722→   722→ * floor (e.g. Chutes ~0.0005 TAO); 3 decimals would show "0.001" — a 2x
   723→   723→ * misquote shown to a user deciding whether to burn. <0.01 → 4 decimals,
   724→   724→ * <1 → 3, else 2.
   725→   725→ */
   726→   726→export function formatBurnTao(n: number): string {
   727→   727→  if (n < 0.01) return n.toFixed(4);
   728→   728→  if (n < 1) return n.toFixed(3);
   729→   729→  return n.toFixed(2);
   730→   730→}
   731→   731→
   732→   732→export type SeatVerdict = "open" | "burn-entry" | "waitlist" | "unknown";
   733→   733→
   734→   734→export interface SeatChance {
   735→   735→  verdict: SeatVerdict;
   736→   736→  /** Registration is possible right now (burn path or free slot). */
   737→   737→  canRegister: boolean;
   738→   738→  slotsFree: number | null;
   739→   739→  totalSlots: number | null;
   740→   740→  /** 0–100 — how full the uid space is. */
   741→   741→  fillPct: number | null;
   742→   742→  burnCostTao: number | null;
   743→   743→  /** Protection window for a NEW seat, in hours (blocks × 12s). */
   744→   744→  immunityHours: number | null;
   745→   745→  /** Share of uids that earned NOTHING last epoch — the replaceable bottom
   746→   746→   *  a newcomer displaces from. High = churn churns in your favor. */
   747→   747→  replaceableShare: number | null;
   748→   748→  headline: string;
   749→   749→  detail: string;
   750→   750→}
   751→   751→
   752→   752→export function assessSeatChance(inputs: {
   753→   753→  minersCount: number | null;
   754→   754→  maxUids: number | null;
   755→   755→  /** Already-computed free slots (skip when you only have counts). */
   756→   756→  freeSlots?: number | null;
   757→   757→  burnCostTao: number | null;
   758→   758→  immunityBlocks: number | null;
   759→   759→  /** Rewarded uids (either give the ratio directly or the absolute count). */
   760→   760→  rewardedRatio?: number | null;
   761→   761→  rewardedMiners?: number | null;
   762→   762→}): SeatChance {
   763→   763→  const { maxUids, burnCostTao, immunityBlocks } = inputs;
   764→   764→  const minersCount = inputs.minersCount ?? 0;
   765→   765→  const slotsFree =
   766→   766→    inputs.freeSlots != null
   767→   767→      ? Math.max(0, inputs.freeSlots)
   768→   768→      : maxUids != null && maxUids > 0
   769→   769→        ? Math.max(0, maxUids - minersCount)
   770→   770→        : null;
   771→   771→  const fillPct =
   772→   772→    maxUids != null && maxUids > 0
   773→   773→      ? Math.min(100, Math.round((minersCount / maxUids) * 100))
   774→   774→      : null;
   775→   775→  let replaceableShare: number | null = null;
   776→   776→  if (inputs.rewardedRatio != null) {
   777→   777→    replaceableShare = Math.max(0, Math.min(1, 1 - inputs.rewardedRatio));
   778→   778→  } else if (inputs.rewardedMiners != null && minersCount > 0) {
