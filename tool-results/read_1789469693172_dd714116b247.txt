    68→

    70→    (d) => d.status !== "terminated" && d.status !== "failed"
    71→  ) ?? [];

    73→    (d) => d.status === "terminated" || d.status === "failed"
    74→  ) ?? [];
    75→
    76→  const selected = deployments?.find((d) => d.id === selectedId) ?? null;
    77→
    78→  return (
    79→    <div className="space-y-6">
    80→      <header className="flex flex-col gap-4 sm:flex-row sm:items-end sm:justify-between">
    81→        <div>
    82→          <p className="text-eyebrow text-muted-foreground">
    83→            Section · 07 · <span className="text-primary">Step 3 — onboard, validate &amp; deploy</span>

    86→            Deployments
    87→          </h1>
    88→          <p className="mt-1 text-sm text-muted-foreground">
    89→            One guided flow: choose a subnet → check its required GPU → rent from a provider →
    90→            requirements install automatically → add the hotkey &amp; register last. The DevOps
    91→            Engine below manages YOUR OWN GPU hosts; rental deployments appear as cards here.
    92→          </p>
    93→        </div>
    94→        <Button className="gap-2 self-start sm:self-end" onClick={() => onOpenWizard()}>
    95→          <Plus className="h-4 w-4" />
    96→          Deploy a miner
    97→        </Button>
    98→      </header>
    99→
   100→      {/* --- DevOps Engine: real GPU host onboarding + 10-step pipeline --- */}
   101→      <DevOpsEngineSection onRent={() => onOpenWizard()} />
   102→
   103→      <Separator />
   104→
   105→      {isLoading ? (
   106→        <Card className="border-border/60 bg-card/40 backdrop-blur-sm">
   107→          <CardContent className="flex items-center justify-center gap-2 py-16 text-muted-foreground">
   108→            <Loader2 className="h-5 w-5 animate-spin" />
   109→            <span className="text-sm">Loading deployments…</span>
   110→          </CardContent>
   111→        </Card>
   112→      ) : deployments && deployments.length === 0 ? (
   113→        <Card className="border-dashed border-border/60 bg-card/20">
   114→          <CardContent className="flex flex-col items-center gap-3 py-16 text-center">
   115→            <Rocket className="h-8 w-8 text-muted-foreground" />
   116→            <div>
   117→              <p className="text-display text-lg font-medium">No deployments yet</p>
   118→              <p className="mt-1 text-sm text-muted-foreground">
   119→                Create your first deployment to provision a GPU and launch a miner.
   120→              </p>
   121→            </div>
   122→            <Button className="mt-2 gap-2" onClick={() => onOpenWizard()}>
   123→              <Plus className="h-4 w-4" />
   124→              Deploy a miner

   126→          </CardContent>
   127→        </Card>

   131→          {active.length > 0 && (
   132→            <section className="space-y-3">
   133→              <p className="text-eyebrow text-muted-foreground">

   139→                    key={d.id}

   190→
   191→function DeploymentCard({
   192→  deployment: d,
   193→  onSelect,
   194→  isSelected,
   195→  onOpenWizard,
   196→}: {
   197→  deployment: DeploymentRecord;
   198→  onSelect: () => void;
   199→  isSelected: boolean;
   200→  onOpenWizard: (ctx: RegistrationWizardContext) => void;
   201→}) {
   202→  const tickMut = useTickDeployment();
   203→  const termMut = useTerminateDeployment();
   204→  const delMut = useDeleteDeployment();
   205→  const regAction = useRegistrationAction();
   206→  const qc = useQueryClient();
   207→  const [devopsNote, setDevopsNote] = useState<string | null>(null);
   208→  const [devopsBusy, setDevopsBusy] = useState(false);
   209→  const [wizardBusy, setWizardBusy] = useState(false);
   210→  const [revOpen, setRevOpen] = useState(false);
   211→  const [migOpen, setMigOpen] = useState(false);
   212→  const [daemonOpen, setDaemonOpen] = useState(false);
   213→
   214→  const isStarted = d.status === "started";
   215→  const hasHotkey = SS58_RE.test(d.hotkey ?? "");
   216→
   217→  // Phase 2 background watcher: for a STARTED deployment with a plausible
   218→  // hotkey, poll the registration check every 45 s (server-throttled).
   219→  const regWatch = useRegistrationWatcher(d.id, isStarted && hasHotkey);
   220→  useEffect(() => {
   221→    if (regWatch.data?.changed) {
   222→      qc.invalidateQueries({ queryKey: ["deployments"] });
   223→    }
   224→  }, [regWatch.data?.changed, regWatch.data?.checkedAt, qc]);
   225→
   226→  // Open the wizard pre-bound to THIS deployment (fetch its hand-off context).
   227→  const openWizard = async () => {
   228→    setWizardBusy(true);
   229→    try {
   230→      const res = await fetch(`/api/deployments/${d.id}/registration`, { cache: "no-store" });
   231→      const j = await res.json();
   232→      if (res.ok && j.wizard) onOpenWizard(j.wizard as RegistrationWizardContext);
   233→      else setDevopsNote(j?.error ?? "Could not load the wizard context");
   234→    } catch {
   235→      setDevopsNote("Could not reach the registration API");
   236→    } finally {
   237→      setWizardBusy(false);
   238→    }
   239→  };
   240→
   241→  const restartAfterReg = () => regAction.mutate({ id: d.id, action: "restart" });
   242→
   243→  const registerToDevOps = async () => {
   244→    setDevopsBusy(true);
   245→    setDevopsNote(null);
   246→    try {
   247→      const res = await fetch(`/api/deployments/${d.id}/to-devops`, { method: "POST" });
   248→      const j = await res.json();
   249→      if (!res.ok) throw new Error(j?.error ?? `HTTP ${res.status}`);
   250→      setDevopsNote(`Registered as host ${String(j.hostName ?? j.hostId)} — run validation in DevOps.`);
   251→    } catch (e) {
   252→      setDevopsNote(e instanceof Error ? e.message : "Register failed");
   253→    } finally {
   254→      setDevopsBusy(false);
   255→    }
   256→  };
   257→
   258→  const isTerminal = d.status === "terminated" || d.status === "failed";
   259→  const isInProgress = !isTerminal && d.status !== "started";
   260→
   261→  const statusColor =
   262→    d.status === "started"
   263→      ? "bg-success/10 text-success"
   264→      : isTerminal
   265→        ? "bg-muted/50 text-muted-foreground"
   266→        : "bg-primary/10 text-primary";
   267→
   268→  return (
   269→    <Card
   270→      className={cn(
   271→        "border-border/60 bg-card/40 backdrop-blur-sm transition-all",
   272→        isSelected && "border-primary/40",
   273→        !isTerminal && "hover:border-primary/30"
   274→      )}
   275→    >
   276→      <CardContent className="p-4">
   277→        <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
   278→          <div className="flex min-w-0 items-start gap-3">
   279→            <div className="flex flex-col items-center gap-1 pt-1">
   280→              <span className={cn("pulse-dot", isStarted ? "text-success" : isInProgress ? "text-primary" : "text-muted-foreground")} />
   281→            </div>
   282→            <div className="min-w-0">
   283→              <div className="flex flex-wrap items-center gap-2">
   284→                <p className="font-medium">{d.minerName}</p>
   285→                <Badge variant="outline" className="mono text-[10px]">α{d.netuid}</Badge>
   286→                <Badge variant="outline" className="text-[10px]">{d.subnetName}</Badge>
   287→                <span className={cn("badge-status capitalize", statusColor)}>{d.status}</span>
   288→                <Badge variant="outline" className={cn("text-[9px]", d.mode === "runpod" ? "border-primary/30 text-primary" : d.mode === "vast" ? "border-warning/40 text-warning" : "text-muted-foreground")}>
   289→                  {d.mode}
   290→                </Badge>
   291→                {/* TIER4 — creator attribution */}
   292→                {d.createdByLabel && (
   293→                  <Badge variant="outline" className="text-[9px] text-muted-foreground">
   294→                    by {d.createdByLabel}
   295→                  </Badge>
   296→                )}
   297→                {/* Phase 2: registration lifecycle chip */}
   298→                {isStarted && d.registrationState === "registered" && (
   299→                  <Badge variant="outline" className="gap-1 border-success/40 text-[10px] text-success">
   300→                    <ShieldCheck className="h-3 w-3" />
   301→                    UID {d.registeredUid ?? "?"}
   302→                  </Badge>
   303→                )}
   304→                {isStarted && d.registrationState === "unregistered" && (
   305→                  <Badge variant="outline" className="gap-1 border-amber-500/40 text-[10px] text-amber-600 dark:text-amber-400">
   306→                    <KeyRound className="h-3 w-3" />
   307→                    unregistered
   308→                  </Badge>
   309→                )}
   310→                {isStarted && d.registrationState === null && (
   311→                  <Badge variant="outline" className={cn("gap-1 text-[10px]", hasHotkey ? "border-amber-500/40 text-amber-600 dark:text-amber-400" : "text-muted-foreground")}>
   312→                    <KeyRound className="h-3 w-3" />
   313→                    {hasHotkey ? "registration unverified" : "no hotkey"}
   314→                  </Badge>
   315→                )}
   316→              </div>
   317→              <p className="mt-1 text-xs text-muted-foreground">
   318→                {d.gpuModel} · {d.provider}
   319→                {d.providerPodId && <span className="mono"> · pod {d.providerPodId.slice(0, 12)}</span>}
   320→                {" · "}created {formatRelativeTime(new Date(d.createdAt))}
   321→              </p>
   322→            </div>
   323→          </div>
   324→
   325→          <div className="flex items-center gap-4">
   326→            <div className="text-right">
   327→              <p className="text-[10px] text-muted-foreground">Cost/mo</p>
   328→              <p className="tabular text-sm font-medium">{formatCurrency(d.monthlyCost)}</p>
   329→            </div>
   330→            <div className="text-right">
   331→              <p className="text-[10px] text-muted-foreground">Rev/mo</p>
   332→              <p className="tabular text-sm font-medium text-success">{formatCurrency(d.estimatedRevenue)}</p>
   333→            </div>
   334→          </div>
   335→        </div>
   336→
   337→        {/* Progress bar */}
   338→        <div className="mt-3">
   339→          <div className="flex items-center justify-between text-[10px] text-muted-foreground">
   340→            <span>Progress</span>
   341→            <span className="tabular">{d.progress}%</span>
   342→          </div>
   343→          <Progress
   344→            value={d.progress}
   345→            className="mt-1 h-1"
   346→            indicatorClassName={cn(
   347→              isStarted ? "bg-success" : isTerminal ? "bg-muted-foreground" : "bg-primary"
   348→            )}
   349→          />
   350→        </div>
   351→
   352→        {/* Step indicators */}
   353→        <div className="mt-3 flex flex-wrap items-center gap-x-3 gap-y-1">
   354→          {d.steps.map((s) => {
   355→            const Icon =
   356→              s.status === "done" ? CheckCircle2
   357→              : s.status === "running" ? Loader2
   358→              : s.status === "failed" ? XCircle
   359→              : Circle;
   360→            return (
   361→              <div key={s.name} className="flex items-center gap-1 text-[10px]">
   362→                <Icon
   363→                  className={cn(
   364→                    "h-3 w-3",
   365→                    s.status === "done" ? "text-success"
   366→                    : s.status === "running" ? "animate-spin text-primary"
   367→                    : s.status === "failed" ? "text-destructive"
   368→                    : "text-muted-foreground/40"
   369→                  )}
   370→                />
   371→                <span className={cn(s.status === "pending" ? "text-muted-foreground/60" : "text-foreground/80")}>
   372→                  {s.label}
   373→                </span>
   374→              </div>
   375→            );
   376→          })}
   377→        </div>
   378→
   379→        {/* Actions */}
   380→        <Separator className="my-3" />
   381→        <div className="flex items-center gap-2">
   382→          <Button variant="outline" size="sm" className="gap-1.5" onClick={onSelect}>
   383→            <Terminal className="h-3.5 w-3.5" />
   384→            View logs
   385→          </Button>
   386→          {/* TIER2 — config revision history + rollback (spec §20/§25) */}
   387→          <Button
   388→            variant="outline"
   389→            size="sm"
   390→            className="gap-1.5"
   391→            onClick={() => setRevOpen(true)}
   392→          >
   393→            <History className="h-3.5 w-3.5" />
   394→            Revisions
   395→          </Button>
   396→          {/* DAEMON-INSTALL — node daemon setup script (Judge Apply / DevOps pushes).
   397→              Any live pod can host the daemon — not just fully "started" ones. */}
   398→          {!isTerminal && (
   399→            <Button
   400→              variant="outline"
   401→              size="sm"
   402→              className="gap-1.5"
   403→              onClick={() => setDaemonOpen(true)}
   404→            >
   405→              <Server className="h-3.5 w-3.5" />
   406→              Install daemon
   407→            </Button>
   408→          )}
   409→          {/* TIER3 — migrate to a different GPU offer (spec §20) */}
   410→          {isStarted && (
   411→            <Button
   412→              variant="outline"
   413→              size="sm"
   414→              className="gap-1.5"
   415→              onClick={() => setMigOpen(true)}
   416→            >
   417→              <ArrowRightLeft className="h-3.5 w-3.5" />
   418→              Migrate
   419→            </Button>
   420→          )}
   421→          {isInProgress && (
   422→            <Button
   423→              variant="outline"
   424→              size="sm"
   425→              className="gap-1.5"
   426→              onClick={() => tickMut.mutate(d.id)}
   427→              disabled={tickMut.isPending}
   428→            >
   429→              <Play className="h-3.5 w-3.5" />
   430→              {tickMut.isPending
   431→                ? "Advancing…"
   432→                : d.installStatus === "failed"
   433→                  ? "Retry install"
   434→                  : d.installStatus === "running"
   435→                    ? "Installing…"
   436→                    : "Advance step"}
   437→            </Button>
   438→          )}
   439→          {isStarted && (
   440→            <Button
   441→              variant="outline"
   442→              size="sm"
   443→              className="gap-1.5"
   444→              onClick={openWizard}
   445→              disabled={wizardBusy}
   446→            >
   447→              {wizardBusy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <KeyRound className="h-3.5 w-3.5" />}
   448→              {d.registrationState === "registered" ? "Wallet & registration" : "Connect & register"}
   449→            </Button>
   450→          )}
   451→          {isStarted && d.registrationState === "registered" && !d.restartedAfterRegistration && (
   452→            <Button
   453→              variant="outline"
   454→              size="sm"
   455→              className="gap-1.5 border-success/40 text-success hover:text-success"
   456→              onClick={restartAfterReg}
   457→              disabled={regAction.isPending}
   458→            >
   459→              {regAction.isPending ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <RotateCw className="h-3.5 w-3.5" />}
   460→              {regAction.isPending ? "Restarting…" : "Restart miner (approval)"}
   461→            </Button>
   462→          )}
   463→          {(d.mode === "runpod" || d.mode === "vast") && !isTerminal && d.providerPodId && (
   464→            <Button
   465→              variant="outline"
   466→              size="sm"
   467→              className="gap-1.5"
   468→              onClick={registerToDevOps}
   469→              disabled={devopsBusy}
   470→            >
   471→              {devopsBusy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Wrench className="h-3.5 w-3.5" />}
   472→              Register to DevOps
   473→            </Button>
   474→          )}
   475→          {!isTerminal && (
   476→            <Button
   477→              variant="outline"
   478→              size="sm"
   479→              className="gap-1.5 text-destructive hover:text-destructive"
   480→              onClick={() => termMut.mutate(d.id)}
   481→              disabled={termMut.isPending}
   482→            >
   483→              <XCircle className="h-3.5 w-3.5" />
   484→              {termMut.isPending ? "Terminating…" : "Terminate"}
   485→            </Button>
   486→          )}
   487→          {isTerminal && (
   488→            <Button
   489→              variant="ghost"
   490→              size="sm"
   491→              className="gap-1.5 text-muted-foreground"
   492→              onClick={() => delMut.mutate(d.id)}
   493→              disabled={delMut.isPending}
   494→            >
   495→              <Trash2 className="h-3.5 w-3.5" />
   496→              Delete
   497→            </Button>
   498→          )}
   499→        </div>
   500→        {regAction.error && (
   501→          <p className="mt-2 rounded-lg border border-destructive/25 bg-destructive/5 px-3 py-2 text-xs text-destructive">
   502→            {regAction.error instanceof Error ? regAction.error.message : "Registration action failed"}
   503→          </p>
   504→        )}
   505→        {devopsNote && (
   506→          <p className="mt-2 rounded-lg border border-primary/25 bg-primary/5 px-3 py-2 text-xs text-primary/90">
   507→            {devopsNote}
   508→          </p>
   509→        )}
   510→      </CardContent>
   511→      <RevisionsDialog
   512→        deploymentId={d.id}
   513→        minerName={d.minerName}
   514→        open={revOpen}
   515→        onOpenChange={setRevOpen}
   516→      />
   517→      <MigrateDialog
   518→        deploymentId={d.id}
   519→        minerName={d.minerName}
   520→        mode={d.mode}
   521→        currentGpuModel={d.gpuModel}
   522→        currentHourlyCost={d.hourlyCost}
   523→        open={migOpen}
   524→        onOpenChange={setMigOpen}
   525→      />
   526→      <DaemonInstallDialog
   527→        deploymentId={d.id}
   528→        minerName={d.minerName}
   529→        open={daemonOpen}
   530→        onOpenChange={setDaemonOpen}
   531→      />
   532→    </Card>
   533→  );
   534→}

   536→function DeploymentDetail({
   537→  deployment: d,
   538→  onClose,
   539→}: {
   540→  deployment: DeploymentRecord;
   541→  onClose: () => void;
   542→}) {
   543→  const [tab, setTab] = useState<"logs" | "config">("logs");
   544→  const cfg = d.config;
   545→

   604→                    s.status === "done" && "border-success/30 text-success",

   607→                  )}>
   608→                    {s.status}
   609→                  </Badge>
   610→                </div>
   611→                {s.output.length > 0 && (
