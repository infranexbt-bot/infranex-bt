    87→function parseGithubUrl(url: string): RepoInfo | null {
    88→  try {
    89→    const u = new URL(url);
    90→    if (!u.hostname.includes("github.com")) return null;
    91→    let parts = u.pathname.split("/").filter(Boolean);
    92→    // github.com/orgs/X/repositories -> org repo listing
    93→    if (parts[0] === "orgs" && parts.length >= 2) parts = [parts[1], "repositories"];
    94→    // strip /tree/<branch>, /blob/<branch>/... suffixes
    95→    if (parts.length >= 3 && ["tree", "blob"].includes(parts[2])) parts = parts.slice(0, 2);
    96→    if (parts.length === 0) return null;
    97→    if (parts.length === 1 || parts[1] === "repositories") {
    98→      // org-only URL — repo resolved later via the org's repositories page
    99→      return { owner: parts[0], repo: "", branch: "main" };
   100→    }
   101→    return { owner: parts[0], repo: parts[1].replace(/\.git$/i, ""), branch: parts[3] || "main" };
   102→  } catch {
   103→    return null;
   104→  }
   105→}

   107→/**
   108→ * Resolve an org-only identity (identityGithub = "github.com/Org" or
   109→ * "/orgs/Org/repositories") to the org's most likely subnet repos by
   110→ * scraping the repositories page HTML. Ordered: miner/subnet-ish names,
   111→ * then program/mechanism names, then everything else. Callers try each
   112→ * until a README is found (some org repos have no root README).
   113→ */
   114→async function resolveOrgRepos(owner: string): Promise<RepoInfo[]> {
   115→  try {
   116→    const res = await fetch(`https://github.com/orgs/${owner}/repositories?q=&type=all`, {
   117→      headers: { "User-Agent": "infranex-bt/1.0" },
   118→      cache: "no-store",
   119→    });
   120→    if (!res.ok) return [];
   121→    const html = await res.text();
   122→    const re = new RegExp(`href="/${owner}/([A-Za-z0-9_.-]+)"`, "gi");
   123→    const found: string[] = [];
   124→    let m: RegExpExecArray | null;
   125→    while ((m = re.exec(html)) !== null) {
   126→      const slug = m[1];
   127→      if (!found.includes(slug)) found.push(slug);
   128→    }
   129→    if (!found.length) return [];
   130→    const score = (r: string) =>
   131→      /miner|subnet|sn[-_]?\d+/i.test(r) ? 0 : /incentive|mechanism|program|partner/i.test(r) ? 1 : 2;
   132→    return found
   133→      .sort((a, b) => score(a) - score(b))
   134→      .slice(0, 3)
   135→      .map((repo) => ({ owner, repo, branch: "main" }));
   136→  } catch {
   137→    return [];
   138→  }
   139→}

   141→async function fetchRaw(
   142→  owner: string,
   143→  repo: string,
   144→  branch: string,
   145→  path: string
   146→): Promise<string | null> {
   147→  const url = `https://raw.githubusercontent.com/${owner}/${repo}/${branch}/${path}`;
   148→  try {
   149→    const res = await fetch(url, {
   150→      headers: { "User-Agent": "infranex-bt/1.0" },
   151→      cache: "no-store",
   152→    });
   153→    if (!res.ok) return null;
   154→    return await res.text();
   155→  } catch {
   156→    return null;
   157→  }
   158→}

   162→async function fetchReadme(info: RepoInfo): Promise<{ content: string; url: string } | null> {
   163→  const branches = ["HEAD", info.branch, "main", "master"];
   164→  // README variants first, then the doc conventions that agent-era subnets
   165→  // use INSTEAD of a root README: AGENTS.md / CLAUDE.md (repo context docs)
   166→  // and docs/MINING.md / docs/README.md (Bittensor mining-doc convention —
   167→  // e.g. SN120 Affine has no root README but a substantive AGENTS.md;
   168→  // SN97 albedo keeps its mining rules in docs/MINING.md).
   169→  const paths = [
   170→    "README.md",
   171→    "readme.md",
   172→    "README.rst",
   173→    "README.txt",
   174→    "README",
   175→    "Readme.md",
   176→    "AGENTS.md",
   177→    "CLAUDE.md",
   178→    "docs/README.md",
   179→    "docs/MINING.md",
   180→  ];
   181→  for (const branch of branches) {
   182→    for (const path of paths) {
   183→      const content = await fetchRaw(info.owner, info.repo, branch, path);
   184→      if (content) {
   185→        return { content, url: `https://github.com/${info.owner}/${info.repo}/blob/${branch}/${path}` };
   186→      }
   187→    }
   188→  }
   189→  return null;
   190→}

   212→// ---------------------------------------------------------------------------
   213→// GPU requirement parsing — ordered strongest-first. Multi-GPU counts
   214→// ("8x H200", "4×A100") are captured too: hosting cost scales with count.
   215→// ---------------------------------------------------------------------------
   216→
   217→interface GpuMatch {
   218→  model: string;
   219→  count: number;
   220→  raw: string;
   221→}
   222→
   223→const GPU_MODEL_PATTERNS: Array<[RegExp, string]> = [
   224→  [/\b(B300)\b/i, "B300"],
   225→  [/\b(B200)\b/i, "B200"],
   226→  [/\b(H200)\b/i, "H200"],
   227→  [/\b(H100)\b/i, "H100"],
   228→  [/\b(RTX\s*Pro\s*6000)\b/i, "RTX Pro 6000"],
   229→  [/\b(RTX\s*6000\s*Ada)\b/i, "RTX 6000 Ada"],
   230→  [/\b(A100)\b/i, "A100"],
   231→  [/\b(L40S)\b/i, "L40S"],
   232→  [/\b(L40)\b/i, "L40"],
   233→  [/\b(A40)\b/i, "A40"],
   234→  [/\b(A6000)\b/i, "RTX A6000"],
   235→  [/\b(A5000)\b/i, "RTX A5000"],
   236→  [/\b(L4)\b/i, "L4"],
   237→  [/\b(A10)\b/i, "A10"],
   238→  [/\b(T4)\b/i, "T4"],
   239→  [/\b(RTX\s*5090)\b/i, "RTX 5090"],
   240→  [/\b(RTX\s*4090)\b/i, "RTX 4090"],
   241→  [/\b(RTX\s*3090)\b/i, "RTX 3090"],
   242→  [/\b(MI300X)\b/i, "MI300X"],
   243→];
   244→
   245→function parseGpuRequirement(text: string): GpuMatch | null {
   246→  for (const [pattern, model] of GPU_MODEL_PATTERNS) {
   247→    const m = text.match(pattern);
   248→    if (!m) continue;
   249→    const idx = m.index ?? 0;
   250→    // Window around the match to catch "8x H200" / "4 × H100" / "8 GPUs (H200)".
   251→    const start = Math.max(0, idx - 24);
   252→    const window = text.slice(start, idx + m[0].length + 8);
   253→    // Chars after the model — table-cell counts: "| H200 | 8 | Validated |".
   254→    const after = text.slice(idx + m[0].length, idx + m[0].length + 24);
   255→    let count = 1;
   256→    // Nearest "N×" BEFORE the model wins: "32× H100 + 24× H200" → 24 (the
   257→    // H200 count), not 56 (the mixed-fleet total). Then table cells, then
   258→    // "8 GPUs (...)" prefixes.
   259→    const xAll = [...window.matchAll(/(\d{1,2})\s*[x×]\s*(?:NVIDIA\s*)?/gi)];
   260→    const countM =
   261→      xAll.length > 0
   262→        ? xAll[xAll.length - 1]
   263→        : after.match(/\|\s*(\d{1,2})\s*\|/) ??
   264→          window.match(/^(\d{1,2})\s*(?:GPUs?|gpu workers?)/i);
   265→    if (countM) {
   266→      const n = parseInt(countM[1], 10);
   267→      if (n >= 2 && n <= 64) count = n;
   268→    }
   269→    return {
   270→      model,
   271→      count,
   272→      raw: count > 1 ? `${count}x ${model}` : model,
   273→    };
   274→  }
   275→  return null;
   276→}

   278→/**
   279→ * GPU requirement across a WHOLE document, strongest evidence first:
   280→ *   1. a line declaring validated/required hardware ("| H200 | 8 | Validated |")
   281→ *   2. a count-qualified mention ("8x H200")
   282→ *   3. a bare model mention (feature lists — weakest)
   283→ * Incentive prose ("run a wide VARIETY of GPUs ... to very powerful (8x h100)")
   284→ * is excluded — it describes the reward spread, not a hosting requirement.
   285→ */
   286→const GPU_LINE_EXCLUDE = /variety|from .* to |cheap|etc\.|such as|e\.g\.|can be found/i;
   287→const GPU_STRONG_CONTEXT = /validated|required|minimum spec|officially supported/i;
   288→
   289→function parseGpuRequirementSmart(text: string): GpuMatch | null {
   290→  let firstCounted: GpuMatch | null = null;
   291→  let firstBare: GpuMatch | null = null;
   292→  for (const line of text.split("\n")) {
   293→    if (GPU_LINE_EXCLUDE.test(line)) continue;
   294→    const m = parseGpuRequirement(line);
   295→    if (!m) continue;
   296→    if (GPU_STRONG_CONTEXT.test(line)) return m;
   297→    if (m.count > 1 && !firstCounted) firstCounted = m;
   298→    if (!firstBare) firstBare = m;
   299→  }
   300→  return firstCounted ?? firstBare;
   301→}

   303→// ---------------------------------------------------------------------------
   304→// Hosting-constraint parsing — the rules that decide WHERE a miner may run.
   305→// Evidence lines are quoted so the UI can show WHY.
   306→// ---------------------------------------------------------------------------
   307→
   308→const HOSTING_PATTERNS: Array<[keyof Omit<HostingRequirements, "notes">, RegExp]> = [
   309→  ["bareMetalOnly", /bare[ -]?metal(\/VM| or VM| and VM)?/i],
   310→  ["bareMetalOnly", /will not work on\s+(runpod|vast)/i],
   311→  ["bareMetalOnly", /not (?:supported|supported on|work(?:ing)?) on\s+(?:runpod|vast|container)/i],
   312→  ["bareMetalOnly", /containers? (?:are )?not supported/i],
   313→  ["teeRequired", /\bTEE\b/],
   314→  ["teeRequired", /\bTDX\b/],
   315→  ["teeRequired", /confidential (?:VM|computing|compute)/i],
   316→  ["teeRequired", /attestation (?:service|required|is required)/i],
   317→  ["staticIpRequired", /static IPs?/i],
   318→  ["staticIpRequired", /1:1 port mapping/i],
   319→  ["staticIpRequired", /shared or dynamic IPs?/i],
   320→];
   321→
   322→/**
   323→ * Hosting-constraint parsing — the rules that decide WHERE a miner may run.
   324→ * Evidence lines are quoted so the UI can show WHY.
   325→ *
   326→ * Guards (audit of all 129 subnets, 2026-09):
   327→ *  - Roadmap sections ("Future Roadmap", "planned") are NOT current
   328→ *    requirements — SN4 Targon lists "bare metal access" as roadmap.
   329→ *  - Negated mentions ("No Nitro enclave or KMS" — SN71) never set flags.
   330→ *  - Validator-only lines ("Validators run inside Phala Cloud" — SN38) do
   331→ *    not constrain the MINER's hosting.
   332→ *  - TEE/static-IP mentions must read as requirements ("must", "runs
   333→ *    inside", "deploy") or sit under a requirements-style heading — a bare
   334→ *    feature-list mention ("TEE Attestation Verification") is not a rule.
   335→ */
   336→const SECTION_HEADING_OK =
   337→  /current implementation|requirement|hardware|deployment|deploy|setup|installation|hosting|infrastructure/;
   338→const LINE_REQUIREMENTISH =
   339→  /(must(?: be| run)?|required to|runs?\s+(?:inside|in|on|within|as)|\b(?:inside|within)\b|deploy|provision|exclusiv|mandatory|enforc|gated?\b|launch(?:es|ed)?|hardware|bare[ -]?metal)/i;
   340→const NEGATION_BEFORE =
   341→  /\b(no|not|without|skip|avoid|neither|nor|don'?t|doesn'?t|can'?t|cannot|never)\s+(?:longer\s+|be\s+)?$/i;
   342→const ROADMAP_HEADING = /roadmap|future|planned|coming soon|not yet|backlog|vision|design/;
   343→
   344→function parseHosting(text: string): HostingRequirements | null {
   345→  const lines = text.split("\n");
   346→  const notes = new Set<string>();
   347→  const flags = { bareMetalOnly: false, teeRequired: false, staticIpRequired: false };
   348→  let any = false;
   349→  for (let i = 0; i < lines.length; i++) {
   350→    const line = lines[i];
   351→    // Nearest markdown heading above (≤30 lines) — section context.
   352→    let heading = "";
   353→    for (let h = i - 1; h >= Math.max(0, i - 30); h--) {
   354→      const t = lines[h].trim();
   355→      if (/^#{1,6} /.test(t)) {
   356→        heading = t.toLowerCase();
   357→        break;
   358→      }
   359→    }
   360→    if (ROADMAP_HEADING.test(heading)) continue; // roadmap ≠ current rule
   361→    for (const [flag, pattern] of HOSTING_PATTERNS) {
   362→      if (flags[flag]) continue;
   363→      const m = line.match(pattern);
   364→      if (!m) continue;
   365→      const idx = m.index ?? 0;
   366→      // Negation guard: "No Nitro enclave or KMS", "without TEE".
   367→      if (NEGATION_BEFORE.test(line.slice(Math.max(0, idx - 48), idx))) continue;
   368→      // Validator-only lines describe the VALIDATOR's hosting, not the miner's.
   369→      if (/\bvalidators?\b/i.test(line) && !/\bminers?\b/i.test(line)) continue;
   370→      // TEE / static-IP need requirement semantics; bare-metal patterns are
   371→      // already requirement-phrased ("must be", "will not work on").
   372→      if (flag !== "bareMetalOnly") {
   373→        const requirementish =
   374→          LINE_REQUIREMENTISH.test(line) || SECTION_HEADING_OK.test(heading);
   375→        if (!requirementish) continue;
   376→      }
   377→      flags[flag] = true;
   378→      any = true;
   379→      // Quote the line containing the evidence (trimmed).
   380→      const cleaned = line
   381→        .trim()
   382→        .replace(/[#*`>]/g, "")
   383→        .slice(0, 220);
   384→      if (cleaned) notes.add(cleaned);
   385→      // NO break — one line can evidence several constraints
   386→      // (e.g. Chutes: "ALL servers must be bare metal/VM ... the IPs must
   387→      // be unique, static, and provide a 1:1 port mapping" sets BOTH
   388→      // bareMetalOnly and staticIpRequired on the same line).
   389→    }
   390→  }
   391→  if (!any) return null;
   392→  return { ...flags, notes: [...notes].slice(0, 5) };
   393→}

   395→// --- INFRA-STACK: service-level infrastructure the subnet's miner stack ----

   425→// service key → detection regex. Conservative: name must appear as a word.
   426→const INFRA_SERVICES: Array<[string, RegExp, RegExp]> = [
   427→  // key, detector, role-extractor (first capture = purpose clause)
   428→  ["kubernetes", /\bk(ubernetes|8s|3s)\b/i, /(?:run(?:ning)?|provision(?:ed)?|deploy(?:ed)?|orchestrat\w*)\s+(?:with|within|by|via|using)?\s*(?:the\s*)?(?:entirety of the\s*)?(k(ubernetes|8s|3s)\b[^\n.]{0,140})/i],
   429→  ["postgres", /\bpostgres(ql|)\b/i, /(?:tracked in|deployed with|uses?|making? use of|requires?)\s+(?:\w+[^\n.]{0,40})?(postgres(?:ql|)\b[^\n.]{0,140})/i],
   430→  ["redis", /\bredis\b/i, /((?:redis\b[^\n.]{0,160}?)(?:used for|triggers?|pubsub)[^\n.]{0,160})/i],
   431→  ["gepetto", /\bgepetto\b/i, /(gepetto\b[^\n.]{0,160})/i],
   432→  // Chutes-class TEE stacks: worker nodes verified via hardware attestation
   433→  // (Intel TDX confidential VMs) instead of software GPU challenges.
   434→  ["tee-attestation", /\b(intel\s+tdx|tee[- ]?(?:worker|node|vm)|hardware\s+attestation|attestation\s+service)\b/i, /((?:intel\s+tdx|tee[- ]?(?:worker|node|vm)|hardware\s+attestation|attestation\s+service)\b[^\n.]{0,160})/i],
   435→  ["rabbitmq", /\brabbitmq\b/i, /(rabbitmq\b[^\n.]{0,140})/i],
   436→  ["nats", /\bnats\b/i, /(nats\b[^\n.]{0,140})/i],
   437→  ["mongodb", /\bmongodb\b/i, /(mongodb\b[^\n.]{0,140})/i],
   438→  ["ipfs", /\bipfs\b/i, /(ipfs\b[^\n.]{0,140})/i],
   439→];
   440→
   441→export function parseInfraStack(text: string): InfraStack | null {
   442→  const lines = text.split("\n");
   443→  const services: InfraService[] = [];
   444→  const seen = new Set<string>();
   445→
   446→  for (const [key, detect, roleRe] of INFRA_SERVICES) {
   447→    if (seen.has(key)) continue;
   448→    let hit: { line: string; role: string | null } | null = null;
   449→    for (let i = 0; i < lines.length; i++) {
   450→      const line = lines[i];
   451→      if (!detect.test(line)) continue;
   452→      // Skip roadmap/speculative sections and validator-only lines.
   453→      let heading = "";
   454→      for (let h = i - 1; h >= Math.max(0, i - 30); h--) {
   455→        const t = lines[h].trim();
   456→        if (/^#{1,6} /.test(t)) { heading = t.toLowerCase(); break; }
   457→      }
   458→      if (ROADMAP_HEADING.test(heading)) continue;
   459→      if (/\bvalidators?\b/i.test(line) && !/\bminers?\b|\bcluster\b|\bstack\b/i.test(line)) continue;
   460→      const roleM = line.match(roleRe);
   461→      hit = { line, role: roleM ? roleM[1].trim().slice(0, 180) : null };
   462→      // Prefer lines with purpose/requirement phrasing over a bare mention.
   463→      if (roleM || /must|requires?|deployed|provisioned|uses?|run(?:s|ning)?\s+within/i.test(line)) break;
   464→    }
   465→    if (hit) {
   466→      seen.add(key);
   467→      services.push({
   468→        name: key,
   469→        role: hit.role,
   470→        quote: hit.line.trim().replace(/[#*`>]/g, "").slice(0, 220),
   471→      });
   472→    }
   473→  }
   474→
   475→  // Orchestration style — what the miner's stack actually runs on.
   476→  let orchestration: InfraStack["orchestration"] = null;
   477→  const joined = text.toLowerCase();
   478→  if (/\bk3s\b|\bkubernetes\b|\bk8s\b/.test(joined)) orchestration = "kubernetes";
   479→  else if (/docker[ -]compose/.test(joined)) orchestration = "docker-compose";
   480→  else if (/\bansible\b/.test(joined)) orchestration = "ansible";
   481→
   482→  // RAM sizing rule — the "RAM per GPU ≥ VRAM" family (Chutes-class). Require
   483→  // both tokens AND per-GPU linkage in one line: generic spec lines ("32 GB
   484→  // RAM, GPU with 8 GB VRAM") are hardware guidance, not sizing RULES.
   485→  let ramRule: InfraStack["ramRule"] = null;
   486→  for (const line of lines) {
   487→    if (
   488→      /\bram\b/i.test(line) &&
   489→      /\bvram\b/i.test(line) &&
   490→      (/per\s+gpu/i.test(line) || /as much ram/i.test(line))
   491→    ) {
   492→      ramRule = { quote: line.trim().replace(/[#*`>]/g, "").slice(0, 220) };
   493→      break;
   494→    }
   495→  }
   496→
   497→  // Networking rule — documented firewall/port obligations ("allow the
   498→  // kubernetes ephemeral port range…", NodePort access, firewall disable).
   499→  // ToC anchor lines ("[Important Networking Note](#…)") don't count. Lines
   500→  // with direct allow/firewall phrasing beat bare NodePort mentions.
   501→  let networkRule: InfraStack["networkRule"] = null;
   502→  const netHit = (t: string) =>
   503→    /firewall|port range|nodeport|port mapping|static ip/i.test(t) &&
   504→    /allow|open|expose|disable|must|need|require|unique|1:1/i.test(t);
   505→  const netPreferred = (t: string) => /firewall|port range|allow the/i.test(t);
   506→  let netFallback: string | null = null;
   507→  for (const line of lines) {
   508→    const t = line.trim();
   509→    if (/^\s*\[/.test(t) || /\]\(#/.test(t)) continue;
   510→    if (!netHit(t)) continue;
   511→    if (netPreferred(t)) {
   512→      networkRule = { quote: t.replace(/[#*`>]/g, "").slice(0, 220) };
   513→      break;
   514→    }
   515→    netFallback ??= t.replace(/[#*`>]/g, "").slice(0, 220);
   516→  }
   517→  networkRule ??= netFallback ? { quote: netFallback } : null;
   518→
   519→  // Storage rule — documented storage prep ("be sure as much as possible is
   520→  // allocated under /var/snap", bind-mount / fstab instructions).
   521→  let storageRule: InfraStack["storageRule"] = null;
   522→  for (const line of lines) {
   523→    const t = line.trim();
   524→    if (/^\s*\[/.test(t) || /\]\(#/.test(t)) continue;
   525→    if (
   526→      /\b(var\/snap|\/etc\/fstab|bind mount)\b/i.test(t) ||
   527→      (/\bstorage\b/i.test(t) && /\bmount|allocat/i.test(t) && /check|be sure|want|should|must/i.test(t))
   528→    ) {
   529→      storageRule = { quote: t.replace(/[#*`>]/g, "").slice(0, 220) };
   530→      break;
   531→    }
   532→  }
   533→
   534→  // Host-class constraints — bare-metal-only / no serverless-rental / static
   535→  // 1:1 IPs ("will not work on Runpod, Vast, etc.").
   536→  let hostClass: InfraStack["hostClass"] = null;
   537→  for (const line of lines) {
   538→    const t = line.trim();
   539→    if (/^\s*\[/.test(t) || /\]\(#/.test(t)) continue;
   540→    if (
   541→      /\b(bare[ -]?metal|will not work on (runpod|vast|hyperstack|lambda)|no shared or dynamic ip|unique, static)\b/i.test(t)
   542→    ) {
   543→      hostClass = { quote: t.replace(/[#*`>]/g, "").slice(0, 220) };
   544→      break;
   545→    }
   546→  }
   547→
   548→  if (!services.length && !ramRule && !networkRule && !storageRule && !hostClass) return null;
   549→  return { services, orchestration, ramRule, networkRule, storageRule, hostClass };
   550→}

   552→// Parse VRAM requirements from text (README or requirements)
   553→function parseVram(text: string): number | null {
   554→  // Look for patterns like "80GB VRAM", "80 GB", "VRAM: 80GB", "min 80GB"
   555→  const patterns = [
   556→    /(\d{2,3})\s*gb\s*vram/i,
   557→    /vram[:\s]*(\d{2,3})\s*gb/i,
   558→    /min(?:imum)?\s*(?:vram|gpu|memory)?[:\s]*(\d{2,3})\s*gb/i,
   559→    /(\d{2,3})\s*gb\s*(?:required|minimum|recommended)/i,
   560→  ];
   561→  for (const p of patterns) {
   562→    const m = text.match(p);
   563→    if (m) {
   564→      const v = parseInt(m[1], 10);
   565→      if (v >= 4 && v <= 200) return v;
   566→    }
   567→  }
   568→  return null;
   569→}

   571→// Parse description from README (first paragraph after the title)
   572→function parseDescription(readme: string): string | null {
   573→  const lines = readme.split("\n");
   574→  let foundTitle = false;
   575→  const descLines: string[] = [];
   576→  for (const line of lines) {
   577→    const trimmed = line.trim();
   578→    if (trimmed.startsWith("#")) {
   579→      foundTitle = true;
   580→      continue;
   581→    }
   582→    if (foundTitle && trimmed && !trimmed.startsWith("!") && !trimmed.startsWith("[") && !trimmed.startsWith("<")) {
   583→      // Skip broken inline-link lines (e.g. Albedo's MINING.md literally
   584→      // contains "in iner/](../miner/).") — the next prose line describes
   585→      // the subnet better than a dangling markdown fragment.
   586→      if (/\]\(/.test(trimmed)) continue;
   587→      descLines.push(trimmed);
   588→      if (descLines.length >= 3) break;
   589→    }
   590→  }
   591→  if (descLines.length === 0) return null;
   592→  const joined = descLines.join(" ").replace(/[#*`]/g, "").trim();
   593→  // Prefer complete sentence(s): list intros ("As a miner you:") and bullet
   594→  // markers ("1.") would otherwise run into their lists and produce a run-on.
   595→  // Split on period+whitespace so version strings (Qwen3.6-35B) survive.
   596→  const parts = joined
   597→    .split(/(?<=[.!?])\s+/)
   598→    .map((s) => s.trim())
   599→    .map((s) => s.replace(/\s+\d+[.)]\s*$/, "").replace(/[:;,]\s*$/, ""))
   600→    .filter((s) => s.length > 0 && !/^\d+[.)]/.test(s));
   601→  const desc = (parts[0] && parts[0].length < 60 && parts[1] ? `${parts[0]} ${parts[1]}` : parts[0] ?? joined).trim();
   602→  return desc.length > 10 && desc.length < 300 ? desc : null;
   603→}

   605→/**
   606→ * Discover a sibling MINER repo from the identity README — many subnets
   607→ * register the product repo on-chain while the mining requirements live
   608→ * in a *-miner / *-validator repo. Only follows same-owner links whose
   609→ * slug mentions miner/validator, so we never wander off.
   610→ */
   611→function discoverMinerRepoUrl(readme: string, identity: RepoInfo): string | null {
   612→  const seen = new Set<string>();
   613→  const re = /github\.com\/([A-Za-z0-9_.-]+)\/([A-Za-z0-9_.-]+)/g;
   614→  let m: RegExpExecArray | null;
   615→  while ((m = re.exec(readme)) !== null) {
   616→    const [full, owner, repoRaw] = m;
   617→    const repo = repoRaw.replace(/\.git$/i, "").replace(/[).,]+$/, "");
   618→    const key = `${owner}/${repo}`.toLowerCase();
   619→    if (seen.has(key)) continue;
   620→    seen.add(key);
   621→    if (owner.toLowerCase() !== identity.owner.toLowerCase()) continue;
   622→    if (`${owner}/${repo}`.toLowerCase() === `${identity.owner}/${identity.repo}`.toLowerCase()) continue;
   623→    if (/miner|validator/i.test(repo)) return `https://github.com/${owner}/${repo}`;
   624→  }
   625→  return null;
   626→}

   628→interface RepoScrape {
   629→  readme: string | null;
   630→  readmeUrl: string | null;
   631→  requirements: string | null;
   632→  requirementsUrl: string | null;
   633→}
   634→
   635→async function scrapeRepo(info: RepoInfo): Promise<RepoScrape> {
   636→  const [readmeResult, reqResult] = await Promise.all([
   637→    fetchReadme(info),
   638→    fetchRequirements(info),
   639→  ]);
   640→  return {
   641→    readme: readmeResult?.content ?? null,
   642→    readmeUrl: readmeResult?.url ?? null,
   643→    requirements: reqResult?.content ?? null,
   644→    requirementsUrl: reqResult?.url ?? null,
   645→  };
   646→}
   647→
   648→/**
   649→ * GPU parsing for IDENTITY READMEs only — every candidate line must carry
   650→ * requirement semantics (require/minimum/supported/validated/hardware/node),
   651→ * so a passing mention of an H100 in a blog-style paragraph never becomes a
   652→ * fake GPU requirement. Strict context wins over loose; lines about OTHER
   653→ * subnets ("helped Chutes onboard B200/B300 nodes") are excluded so a
   654→ * subnet's own fleet is described, not its neighbours'.
   655→ */
   656→const GPU_STRICT_CONTEXT =
   657→  /(require|minimum|supported|validated|must|need|provision|fleet|runs? on|topolog)/i;
   658→const GPU_LOOSE_CONTEXT =
   659→  /(require|minimum|supported|validated|must|need|provision|hardware|cluster|node|topolog|fleet|gpu)/i;
   660→const OTHER_SUBNET_MENTION =
   661→  /\b(subnet\s?\d{1,3}|chutes|targon|lium|gradients|compute.?horde)\b/i;
   662→
   663→function parseGpuWithContext(readme: string): GpuMatch | null {
   664→  for (const strict of [true, false]) {
   665→    const ctx = strict ? GPU_STRICT_CONTEXT : GPU_LOOSE_CONTEXT;
   666→    for (const line of readme.split("\n")) {
   667→      if (!ctx.test(line)) continue;
   668→      if (OTHER_SUBNET_MENTION.test(line)) continue;
   669→      const m = parseGpuRequirement(line);
   670→      if (m) return m;
   671→    }
   672→  }
   673→  return null;
   674→}

   676→export async function scrapeGithubMetadata(
   677→  githubUrl: string,
   678→  opts?: { netuid?: number; subnetName?: string | null }
   679→): Promise<ScrapedMetadata> {
   680→  const NO_MECHANICS: ScrapedMetadata = { description: null, minVramGb: null, recommendedGpu: null, gpuCount: null, gpuModelRaw: null, hosting: null, mechanics: null, infra: null, requirementsSource: null, readmeUrl: null, requirementsUrl: null, rawReadmeSnippet: null, source: "error" };
   681→  try {
   682→    let info = parseGithubUrl(githubUrl);
   683→    if (!info) {
   684→      return { ...NO_MECHANICS, error: "Invalid GitHub URL" };
   685→    }
   686→    // Org-only identity ("github.com/Org") — resolve to the org's most
   687→    // likely subnet/miner repo via the repositories page; try candidates
   688→    // in order until one has a README.
   689→    let identityRepos: RepoScrape;
   690→    if (!info.repo) {
   691→      const candidates = await resolveOrgRepos(info.owner);
   692→      let resolvedInfo: RepoInfo | null = null;
   693→      let identityScrape: RepoScrape | null = null;
   694→      for (const cand of candidates) {
   695→        const sc = await scrapeRepo(cand);
   696→        if (sc.readme) {
   697→          resolvedInfo = cand;
   698→          identityScrape = sc;
   699→          break;
   700→        }
   701→        if (!identityScrape) {
   702→          resolvedInfo = cand;
   703→          identityScrape = sc;
   704→        }
   705→      }
   706→      if (!resolvedInfo || !identityScrape) {
   707→        return { ...NO_MECHANICS, error: `No repo resolvable for org ${info.owner}` };
   708→      }
   709→      info = resolvedInfo;
   710→      identityRepos = identityScrape;
   711→    } else {
   712→      identityRepos = await scrapeRepo(info);
   713→    }
   714→
   715→    const identity = identityRepos;
   716→
   717→    // --- Second repo: the miner repo (curated override > README discovery) --
   718→    let minerUrl = opts?.netuid != null ? CURATED_MINER_REPOS[opts.netuid] ?? null : null;
   719→    if (!minerUrl && identity.readme) {
   720→      minerUrl = discoverMinerRepoUrl(identity.readme, info);
   721→    }
   722→    let miner: RepoScrape | null = null;
   723→    let minerRepoInfo: RepoInfo | null = null;
   724→    if (minerUrl) {
   725→      const mi = parseGithubUrl(minerUrl);
   726→      if (mi) {
   727→        minerRepoInfo = mi;
   728→        miner = await scrapeRepo(mi);
   729→      }
   730→    }
   731→
   732→    // --- Extra curated requirement repos (sek8s-style host docs) ----------
   733→    const extraTexts: string[] = [];
   734→    const extraUrls = opts?.netuid != null ? CURATED_EXTRA_REPOS[opts.netuid] ?? [] : [];
   735→    for (const extraUrl of extraUrls) {
   736→      const ei = parseGithubUrl(extraUrl);
   737→      if (!ei) continue;
   738→      const er = await fetchReadme(ei);
   739→      if (er) extraTexts.push(er.content);
   740→      for (const docPath of EXTRA_DOC_PATHS) {
   741→        const [do_, dp] = [ei.owner, ei.repo];
   742→        const branchDoc = await fetchRaw(do_, dp, "HEAD", docPath);
   743→        if (branchDoc) extraTexts.push(branchDoc);
   744→      }
   745→    }
   746→
   747→    if (!identity.readme && !miner?.readme) {
   748→      return { ...NO_MECHANICS, error: "No README or requirements found" };
   749→    }
   750→
   751→    // Mining requirements prefer the MINER repo text (that's where operators
   752→    // document hardware + hosting rules); description prefers the identity repo.
   753→    // The IDENTITY README is the last-resort hosting source: many subnets
   754→    // document TEE/hosting rules only there (SN90 KubeTEE, SN4 Targon,
   755→    // SN28 SayGM, SN51 lium, SN58 greevils) — but GPU models are parsed
   756→    // from it only with requirement-context, to avoid feature-prose
   757→    // false positives (the SN64 class of bug).
   758→    const hostingTexts = [
   759→      miner?.readme,
   760→      ...extraTexts,
   761→      miner?.requirements,
   762→      identity.requirements,
   763→      identity.readme,
   764→    ].filter(Boolean) as string[];
   765→    const hostingCombined = hostingTexts.join("\n\n") || null;
   766→    const reqTexts = [
   767→      miner?.readme,
   768→      ...extraTexts,
   769→      miner?.requirements,
   770→      identity.requirements,
   771→    ].filter(Boolean) as string[];
   772→    const reqCombined = reqTexts.join("\n\n") || null;
   773→
   774→    const gpu = reqCombined ? parseGpuRequirementSmart(reqCombined) : null;
   775→    // Identity-README GPU fallback — only lines with requirement context.
   776→    const gpuFromIdentity = !gpu && identity.readme ? parseGpuWithContext(identity.readme) : null;
   777→    const gpuFinal = gpu ?? gpuFromIdentity;
   778→    const hosting = hostingCombined ? parseHosting(hostingCombined) : null;
   779→    const vram = reqCombined ? parseVram(reqCombined) : null;
   780→
   781→    // MECHANICS-ALL: derive mechanics from the same combined README text the
   782→    // hosting/GPU parsers read. Sparse by design — null when nothing hit.
   783→    const extractedMechanics = hostingCombined
   784→      ? extractMechanicsFromText(hostingCombined)
   785→      : null;
   786→    const mechanics = extractedMechanics
   787→      ? buildDerivedMechanics({
   788→          netuid: opts?.netuid ?? null,
   789→          subnetName: opts?.subnetName ?? null,
   790→          sourceUrl: miner?.readme && minerRepoInfo
   791→            ? `https://github.com/${minerRepoInfo.owner}/${minerRepoInfo.repo}`
   792→            : `https://github.com/${info.owner}/${info.repo}`,
   793→          extracted: extractedMechanics,
   794→          hosting,
   795→        })
   796→      : null;
   797→
   798→    // INFRA-STACK: service components from the same combined text.
   799→    const infra = hostingCombined ? parseInfraStack(hostingCombined) : null;
   800→
   801→    const requirementsSource = miner?.readme && minerRepoInfo
   802→      ? `https://github.com/${minerRepoInfo.owner}/${minerRepoInfo.repo}`
   803→      : `https://github.com/${info.owner}/${info.repo}`;
   804→
   805→    return {
   806→      description: identity.readme ? parseDescription(identity.readme) : null,
   807→      minVramGb: vram,
   808→      recommendedGpu: gpuFinal ? gpuFinal.raw : null,
   809→      gpuCount: gpuFinal ? gpuFinal.count : null,
   810→      gpuModelRaw: gpuFinal ? gpuFinal.raw : null,
   811→      hosting,
   812→      mechanics,
   813→      infra,
   814→      requirementsSource: gpuFinal || hosting ? requirementsSource : null,
   815→      readmeUrl: identity.readmeUrl,
   816→      requirementsUrl: miner?.requirementsUrl ?? identity.requirementsUrl,
   817→      rawReadmeSnippet: identity.readme?.slice(0, 500) ?? null,
   818→      source: "github",
   819→    };
   820→  } catch (e) {
   821→    return { ...NO_MECHANICS, error: e instanceof Error ? e.message : String(e) };
   822→  }
   823→}
